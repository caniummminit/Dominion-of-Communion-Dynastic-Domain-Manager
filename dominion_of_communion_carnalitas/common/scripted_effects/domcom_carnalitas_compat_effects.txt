# A Dominion Pet is not Carnalitas slavery. If both characters are mutually attracted and Carnalitas
# allows the BDSM relation, the Pet is additionally recognized as submissive and the Patron as dominant.
domcom_carn_sync_pet_relation_effect = {
    if = {
        limit = {
            has_character_flag = domcom_pet
            has_variable = domcom_pet_patron
            var:domcom_pet_patron = { is_alive = yes age >= 18 }
        }
        var:domcom_pet_patron = { save_scope_as = domcom_carn_pet_patron }
        if = {
            limit = {
                can_set_relation_submissive_trigger = { CHARACTER = scope:domcom_carn_pet_patron }
            }
            set_relation_submissive = scope:domcom_carn_pet_patron
        }
    }
}
