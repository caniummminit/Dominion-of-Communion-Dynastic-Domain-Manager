domcom.200 = {
    type = character_event
    title = domcom.200.t
    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { exists = scope:domcom_crush_source }
                desc = domcom.200.desc.kin
            }
            desc = domcom.200.desc
        }
    }
    theme = seduction
    left_portrait = { character = root animation = personality_rational }
    right_portrait = { character = scope:domcom_admirer animation = flirtation_left }

    immediate = {
        if = {
            limit = {
                exists = scope:domcom_admirer
                scope:domcom_admirer = { has_variable = domcom_crush_source }
                scope:domcom_admirer.var:domcom_crush_source = { is_alive = yes }
            }
            scope:domcom_admirer.var:domcom_crush_source = { save_scope_as = domcom_crush_source }
        }
    }

    trigger = {
        exists = scope:domcom_admirer
        scope:domcom_admirer = {
            is_alive = yes
            age >= 18
            has_variable = domcom_crush_patron
            var:domcom_crush_patron = root
            OR = { has_relation_crush = root has_relation_lover = root has_relation_soulmate = root }
        }
        root = { age >= 18 is_alive = yes }
    }

    option = {
        name = domcom.200.a
        scope:domcom_admirer = {
            root = { save_scope_as = domcom_admirer_patron }
            domcom_consummate_admirer_effect = yes

            # Receiving a prospect is a one-time resolution. Keep any resulting Crush/Lover/Soulmate
            # relationship, but retire the prospect marker so the same person does not remain forever in
            # the queue.
            if = {
                limit = { has_variable = domcom_crush_source }
                var:domcom_crush_source = {
                    remove_list_variable = {
                        name = domcom_family_admirers
                        target = scope:domcom_admirer
                    }
                }
                remove_variable = domcom_crush_source
            }
            remove_variable = domcom_crush_patron
            add_character_flag = domcom_prospect_received
        }
        remove_list_variable = { name = domcom_crush_candidates target = scope:domcom_admirer }
        ai_chance = { base = 80 }
    }

    option = {
        name = domcom.200.b
        scope:domcom_admirer = {
            if = {
                limit = { has_variable = domcom_crush_source }
                var:domcom_crush_source = {
                    remove_list_variable = {
                        name = domcom_family_admirers
                        target = scope:domcom_admirer
                    }
                }
                remove_variable = domcom_crush_source
            }
            remove_relation_crush = root
            remove_variable = domcom_crush_patron
        }
        remove_list_variable = { name = domcom_crush_candidates target = scope:domcom_admirer }
        ai_chance = { base = 20 }
    }
}

    immediate = {
        scope:domcom_attraction_patron = { save_scope_as = domcom_compatibility_target }

            domcom_register_communion_admirer_effect = yes
        
            add_character_flag = domcom_rng_attraction

        # CK3 has no else on random; a failed attraction check must explicitly clear its pending flag.

        }
        remove_character_flag = domcom_rng_attraction
    }
}
