# 1.0.0-rc4.4 — the AI values are what the character sheet's two-word
# personality summary is computed from, so shifting them here is what turns a
# Blackguard into something servile. The summary is only a readout; there is no
# label to set.
#
# The profile is a useful servant, not a ruined one. Skills and rationality are
# preserved or raised, because the entire point of taking a collapsed character
# into court instead of killing him is that he keeps working and keeps breeding.
# What changes is what he wants, not what he can do:
#
#   boldness     floored - he cannot bring himself to act against a superior.
#                This is the single biggest factor in whether a character joins
#                a faction, so it is what actually stops the rebelling.
#   vengefulness floored - he keeps no grievance, so nothing accumulates.
#   honor        raised  - he keeps his word once given, including to you.
#   rationality  raised  - he does what is objectively best, and rebelling
#                against the man who broke him never is.
#   energy       raised  - he stays industrious. A lackey who will not work is
#                no use as a councillor.
#   greed, zeal  lowered - the two motives that most often override loyalty.
#
# Sociability is left alone deliberately. He should still function at court and
# in a marriage; withdrawing him would defeat the purpose.
domcom_absolute_devotion_modifier = {
    icon = love_positive
    diplomacy = 2
    attraction_opinion = 100

    ai_boldness = -100
    ai_vengefulness = -100
    ai_honor = 80
    ai_rationality = 60
    ai_energy = 40
    ai_greed = -50
    ai_zeal = -60
}
