domcom_alpha_check_interaction = {
    icon = wrathful
    category = interaction_category_hostile
    common_interaction = yes
    # 0.9.6 — respect CK3 diplomatic range; right-click can no longer bypass geography.
    use_diplomatic_range = yes
    interface_priority = 57
    desc = domcom_alpha_check_interaction_desc
    cooldown_against_recipient = { years = 1 }

    is_shown = {
        scope:actor = { age >= 18 is_alive = yes }
        scope:recipient != scope:actor
        scope:recipient = { domcom_general_alpha_target_trigger = yes }
    }

    is_valid_showing_failures_only = {
        scope:actor = {
            is_imprisoned = no
            # 0.9.9 — the per-quarter allowance. The Alpha Gauntlet is exempt and
            # is not gated by this.
            custom_tooltip = {
                text = domcom_alpha_budget_spent_tt
                domcom_alpha_budget_available_trigger = yes
            }
        }
        scope:recipient = {
            is_imprisoned = no
            custom_tooltip = {
                text = domcom_alpha_recent_challenge_tt
                NOT = {
                    AND = {
                        has_character_flag = domcom_recent_alpha_challenge
                        has_variable = domcom_last_alpha_patron
                        var:domcom_last_alpha_patron = scope:actor
                    }
                }
            }
            # 0.9.8 — these three used to hide the interaction outright. Shown
            # as greyed-out reasons instead, so a rival who cannot be challenged
            # says so rather than vanishing from the menu.
            custom_tooltip = {
                text = domcom_alpha_collapse_resolved_tt
                NOT = { has_character_flag = domcom_collapse_resolved }
            }
            custom_tooltip = {
                text = domcom_alpha_already_outclassed_tt
                NOT = {
                    AND = {
                        has_character_flag = domcom_dominated
                        has_variable = domcom_dominance_patron
                        var:domcom_dominance_patron = scope:actor
                        NOT = {
                            OR = {
                                domcom_cuck_evidence_trigger = yes
                                AND = {
                                    has_character_flag = domcom_cucked
                                    has_variable = domcom_cuck_patron
                                    var:domcom_cuck_patron = scope:actor
                                }
                            }
                        }
                    }
                }
            }
            custom_tooltip = {
                text = domcom_alpha_terminal_xp_tt
                NOT = {
                    AND = {
                        has_character_flag = domcom_owned_cuck
                        has_variable = domcom_patron
                        var:domcom_patron = scope:actor
                        has_variable = domcom_cuck_xp
                        var:domcom_cuck_xp >= 1000
                    }
                }
            }
        }
    }

    on_accept = {
        # 0.8.4 — the outcome branches are gated on domcom_alpha_check_succeeded,
        # a flag that does not exist while CK3 is building the tooltip, so the
        # old version rendered "97% chance of:" followed by nothing at all. The
        # opinion penalty rendered as certain for the same reason. Scopes are
        # saved before the tooltip so the chance is computed from the real
        # numbers rather than from a missing patron.
        scope:actor = { save_scope_as = domcom_owner }

        # Let CK3 render the percentage itself. A tooltip-only random block
        # produces the same native "X% chance of:" presentation used by
        # vanilla interactions instead of relying on localization scope parsing.
        # 0.9.7 — a bare random prints one "X% chance of:" header and then lists
        # everything under it, which is why success and failure both appeared to
        # sit inside the 18%. A random_list prints a percentage against each
        # branch instead, which is the presentation Send An Insult uses.
        # Base weight 0 plus a modifier is the reliable way to weight a
        # random_list by a script value.
        show_as_tooltip = {
            scope:recipient = {
                random_list = {
                    0 = {
                        modifier = { add = domcom_alpha_check_chance_value }
                        custom_tooltip = domcom_alpha_check_tt_success
                    }
                    0 = {
                        modifier = { add = domcom_alpha_check_failure_weight_value }
                        custom_tooltip = domcom_alpha_check_tt_failure
                    }
                }
            }
        }

        hidden_effect = {
        scope:actor = {
            domcom_initialize_profile_effect = yes
            domcom_initialize_ledger_counters_effect = yes
            change_variable = { name = domcom_alpha_attempts add = 1 }
            domcom_spend_alpha_budget_effect = yes
            save_scope_as = domcom_owner
        }
        scope:recipient = {
            domcom_initialize_profile_effect = yes
            add_character_flag = { flag = domcom_recent_alpha_challenge days = 365 }
            set_variable = { name = domcom_last_alpha_patron value = scope:actor }
        }

        scope:recipient = {
            random = {
                chance = domcom_alpha_check_chance_value
                add_character_flag = domcom_alpha_check_succeeded
            }
        }

        if = {
            limit = { scope:recipient = { has_character_flag = domcom_alpha_check_succeeded } }
            scope:recipient = { remove_character_flag = domcom_alpha_check_succeeded }
            scope:actor = { domcom_dominion_success_stress_effect = yes }

            if = {
                # After cuck evidence exists, Alpha victory no longer creates the
                # pre-cuck Outclassed state. It registers/owns the cuck if needed
                # and directly pushes their Cuck XP toward terminal submission.
                limit = { scope:recipient = { domcom_is_cuck_target_of_actor_trigger = yes } }
                scope:recipient = {
                    if = {
                        limit = { NOT = { has_character_flag = domcom_cucked } }
                        domcom_register_cuck_generic_effect = yes
                    }
                    if = {
                        limit = {
                            has_character_flag = domcom_dominated
                            has_variable = domcom_dominance_patron
                            var:domcom_dominance_patron = scope:actor
                        }
                        remove_character_flag = domcom_dominated
                        remove_character_modifier = domcom_dominated_modifier
                        remove_variable = domcom_dominance_patron
                    }
                    if = {
                        limit = { NOT = { domcom_is_owned_by_actor_trigger = yes } }
                        domcom_own_cuck_generic_effect = yes
                    }
                    if = { limit = { NOT = { has_variable = domcom_cuck_xp } } set_variable = { name = domcom_cuck_xp value = 0 } }
                    change_variable = { name = domcom_cuck_xp add = 100 }
                    if = { limit = { var:domcom_cuck_xp > 1000 } set_variable = { name = domcom_cuck_xp value = 1000 } }
                    domcom_refresh_cuck_stage_effect = yes
                }
            }
            else = {
                # Before any cuck evidence exists, victory grants leverage/access
                # only. It does NOT create obedience, ownership, or a loyalty hook.
                scope:recipient = { domcom_apply_dominated_effect = yes }
            }

            # Reverse Alpha: personal dominance over someone above your landed status.
            if = {
                limit = {
                    OR = {
                        AND = { scope:recipient = { highest_held_title_tier = tier_county } scope:actor = { is_landed = no } }
                        AND = { scope:recipient = { highest_held_title_tier = tier_duchy } scope:actor = { OR = { is_landed = no highest_held_title_tier = tier_county } } }
                        AND = { scope:recipient = { highest_held_title_tier = tier_kingdom } scope:actor = { OR = { is_landed = no highest_held_title_tier = tier_county highest_held_title_tier = tier_duchy } } }
                        AND = { scope:recipient = { highest_held_title_tier = tier_empire } scope:actor = { NOT = { highest_held_title_tier = tier_empire } } }
                    }
                }
                scope:actor = { domcom_apply_reverse_alpha_reward_effect = yes }
                scope:recipient = { add_opinion = { target = scope:actor modifier = domcom_reverse_alpha_shock_opinion } }
            }

            scope:actor = {
                change_variable = { name = domcom_alpha_successes add = 1 }
                if = {
                    limit = { is_ai = no }
                    send_interface_toast = {
                        type = event_toast_effect_good
                        title = domcom_alpha_check_success_toast
                        left_icon = scope:actor
                        right_icon = scope:recipient
                    }
                }
            }
        }
        else = {
            scope:actor = { domcom_dominion_rejection_stress_effect = yes }
            # Before cuck evidence, failure simply means the rival resisted you.
            # After cuck evidence, a failed reassertion restores some resistance
            # instead of perversely increasing their Cuck XP as older builds did.
            if = {
                limit = { scope:recipient = { domcom_is_cuck_target_of_actor_trigger = yes } }
                scope:recipient = {
                    if = { limit = { NOT = { has_character_flag = domcom_cucked } } domcom_register_cuck_generic_effect = yes }
                    if = { limit = { NOT = { has_variable = domcom_cuck_xp } } set_variable = { name = domcom_cuck_xp value = 100 } }
                    change_variable = { name = domcom_cuck_xp add = -50 }
                    if = { limit = { var:domcom_cuck_xp < 100 } set_variable = { name = domcom_cuck_xp value = 100 } }
                    domcom_refresh_cuck_stage_effect = yes
                }
            }
            scope:recipient = { add_opinion = { target = scope:actor modifier = domcom_resisted_alpha_opinion } }
            scope:actor = {
                change_variable = { name = domcom_alpha_failures add = 1 }
                if = {
                    limit = { is_ai = no }
                    send_interface_toast = {
                        type = event_toast_effect_bad
                        title = domcom_alpha_check_failure_toast
                        left_icon = scope:actor
                        right_icon = scope:recipient
                    }
                }
            }
        }
        }
    }

    auto_accept = yes
    ai_accept = { base = 100 }
    ai_frequency = 0
}

domcom_tantalize_interaction = {
    icon = lover
    category = interaction_category_friendly
    common_interaction = yes
    # 0.9.6 — respect CK3 diplomatic range; right-click can no longer bypass geography.
    use_diplomatic_range = yes
    interface_priority = 56
    desc = domcom_tantalize_interaction_desc
    cooldown_against_recipient = { years = 1 }

    is_shown = {
        scope:actor = { age >= 18 is_alive = yes }
        scope:recipient != scope:actor
        scope:recipient = { domcom_tantalize_target_trigger = yes }
    }

    is_valid_showing_failures_only = {
        scope:actor = { is_imprisoned = no }
        scope:recipient = { is_imprisoned = no }
    }

    on_accept = {
        # 0.7.3 — the tooltip is now written explicitly instead of being dry-run
        # from the logic. CK3 builds an interaction tooltip by evaluating the
        # effect block before anything has happened, so the old version showed
        # "31% chance of:" with nothing under it (the success branch is gated on
        # a character flag that does not exist yet, and add_character_flag has no
        # tooltip form), and then rendered the else branch's opinion penalty as
        # though it were certain. show_as_tooltip states the real outcomes and
        # hidden_effect runs the real logic.
        # The chance value reads scope:domcom_tantalize_patron, which the old
        # code only saved inside the branch below. At tooltip time that scope
        # did not exist, so the patron's diplomacy, intrigue, prestige and dread
        # contributions all evaluated as nothing and the displayed percentage
        # was computed from the base alone. Saving it up front fixes the number.
        scope:actor = {
            save_scope_as = domcom_tantalize_patron
            save_scope_as = domcom_compatibility_target
        }

        # Use CK3's native random-effect tooltip so the actual calculated
        # percentage is rendered as "X% chance of:" instead of a blank %.
        show_as_tooltip = {
            scope:recipient = {
                random_list = {
                    0 = {
                        modifier = { add = domcom_tantalize_chance_v063_value }
                        custom_tooltip = domcom_tantalize_tt_success
                    }
                    0 = {
                        modifier = { add = domcom_tantalize_failure_weight_value }
                        custom_tooltip = domcom_tantalize_tt_failure
                    }
                }
            }
        }

        hidden_effect = {
        scope:actor = {
            domcom_initialize_profile_effect = yes
            change_variable = { name = domcom_tantalize_attempts add = 1 }
            save_scope_as = domcom_tantalize_patron
            save_scope_as = domcom_compatibility_target
        }
        scope:recipient = {
            domcom_initialize_attraction_profile_effect = yes
            set_variable = { name = domcom_last_compatibility value = domcom_pair_compatibility_value }
        }

        scope:recipient = {
            random = {
                chance = domcom_tantalize_chance_v063_value
                add_character_flag = domcom_tantalize_succeeded
            }
        }

        if = {
            limit = { scope:recipient = { has_character_flag = domcom_tantalize_succeeded } }
            scope:actor = { domcom_dominion_success_stress_effect = yes }
            scope:recipient = {
                remove_character_flag = domcom_tantalize_succeeded
                domcom_apply_tantalize_success_effect = yes
                domcom_try_tantalize_intimacy_effect = yes
            }
            scope:actor = {
                change_variable = { name = domcom_tantalize_successes add = 1 }
                if = {
                    limit = { is_ai = no }
                    send_interface_toast = {
                        type = event_toast_effect_good
                        title = domcom_tantalize_success_toast
                        left_icon = scope:actor
                        right_icon = scope:recipient
                    }
                }
            }
            if = {
                limit = { scope:recipient = { has_character_flag = domcom_new_pet_created } }
                scope:recipient = { remove_character_flag = domcom_new_pet_created }
                scope:actor = {
                    if = {
                        limit = { is_ai = no }
                        send_interface_toast = {
                            type = event_toast_effect_good
                            title = domcom_pet_created_toast
                            left_icon = scope:actor
                            right_icon = scope:recipient
                        }
                    }
                }
            }
        }
        else = {
            scope:recipient = { add_opinion = { target = scope:actor modifier = domcom_rebuffed_tantalize_opinion } }
            scope:actor = {
                domcom_dominion_rejection_stress_effect = yes
                change_variable = { name = domcom_tantalize_failures add = 1 }
                if = {
                    limit = { is_ai = no }
                    if = {
                        limit = {
                            scope:recipient = {
                                OR = {
                                    has_relation_lover = scope:actor
                                    has_relation_soulmate = scope:actor
                                }
                            }
                        }
                        send_interface_toast = {
                            type = event_toast_effect_bad
                            title = domcom_tantalize_influence_resisted_toast
                            left_icon = scope:actor
                            right_icon = scope:recipient
                        }
                    }
                    else = {
                        send_interface_toast = {
                            type = event_toast_effect_bad
                            title = domcom_tantalize_failure_toast
                            left_icon = scope:actor
                            right_icon = scope:recipient
                        }
                    }
                }
            }
        }
        }
    }

    auto_accept = yes
    ai_accept = { base = 100 }
    ai_frequency = 0
}
