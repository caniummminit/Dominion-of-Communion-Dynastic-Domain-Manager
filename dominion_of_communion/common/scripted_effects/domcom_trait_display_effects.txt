# Dominion of Communion 1.2.4 — native Endowment trait display.
#
# This file remains an override point for the optional Carnalitas bridge, but
# both versions apply the same Dominion trait. With Carnalitas loaded, its
# native DT trait is synchronized and appears beside this rose-based Dominion
# trait rather than replacing it.
#
# 1.2.4 — each branch now prunes the other five tiers before checking whether
# the correct one is already held. The old shape only cleared when it was about
# to add, so a character who somehow carried two Endowment traits — a ruler
# designer pick plus a rolled one, a console add, an interrupted rebuild — kept
# both forever, because the branch for the tier he already had did nothing at
# all. Pruning first makes exactly one Endowment trait an invariant, and the
# effect stays a true no-op for the overwhelming majority who are already right.

domcom_apply_native_endowment_trait_effect = {
    if = {
        limit = { var:domcom_endowment_category = flag:micro }
        if = { limit = { has_trait = domcom_endowment_small } remove_trait = domcom_endowment_small }
        if = { limit = { has_trait = domcom_endowment_average } remove_trait = domcom_endowment_average }
        if = { limit = { has_trait = domcom_endowment_big } remove_trait = domcom_endowment_big }
        if = { limit = { has_trait = domcom_endowment_hung } remove_trait = domcom_endowment_hung }
        if = { limit = { has_trait = domcom_endowment_massive } remove_trait = domcom_endowment_massive }
        if = { limit = { NOT = { has_trait = domcom_endowment_micro } } add_trait = domcom_endowment_micro }
    }
    else_if = {
        limit = { var:domcom_endowment_category = flag:small }
        if = { limit = { has_trait = domcom_endowment_micro } remove_trait = domcom_endowment_micro }
        if = { limit = { has_trait = domcom_endowment_average } remove_trait = domcom_endowment_average }
        if = { limit = { has_trait = domcom_endowment_big } remove_trait = domcom_endowment_big }
        if = { limit = { has_trait = domcom_endowment_hung } remove_trait = domcom_endowment_hung }
        if = { limit = { has_trait = domcom_endowment_massive } remove_trait = domcom_endowment_massive }
        if = { limit = { NOT = { has_trait = domcom_endowment_small } } add_trait = domcom_endowment_small }
    }
    else_if = {
        limit = { var:domcom_endowment_category = flag:average }
        if = { limit = { has_trait = domcom_endowment_micro } remove_trait = domcom_endowment_micro }
        if = { limit = { has_trait = domcom_endowment_small } remove_trait = domcom_endowment_small }
        if = { limit = { has_trait = domcom_endowment_big } remove_trait = domcom_endowment_big }
        if = { limit = { has_trait = domcom_endowment_hung } remove_trait = domcom_endowment_hung }
        if = { limit = { has_trait = domcom_endowment_massive } remove_trait = domcom_endowment_massive }
        if = { limit = { NOT = { has_trait = domcom_endowment_average } } add_trait = domcom_endowment_average }
    }
    else_if = {
        limit = { var:domcom_endowment_category = flag:big }
        if = { limit = { has_trait = domcom_endowment_micro } remove_trait = domcom_endowment_micro }
        if = { limit = { has_trait = domcom_endowment_small } remove_trait = domcom_endowment_small }
        if = { limit = { has_trait = domcom_endowment_average } remove_trait = domcom_endowment_average }
        if = { limit = { has_trait = domcom_endowment_hung } remove_trait = domcom_endowment_hung }
        if = { limit = { has_trait = domcom_endowment_massive } remove_trait = domcom_endowment_massive }
        if = { limit = { NOT = { has_trait = domcom_endowment_big } } add_trait = domcom_endowment_big }
    }
    else_if = {
        limit = { var:domcom_endowment_category = flag:hung }
        if = { limit = { has_trait = domcom_endowment_micro } remove_trait = domcom_endowment_micro }
        if = { limit = { has_trait = domcom_endowment_small } remove_trait = domcom_endowment_small }
        if = { limit = { has_trait = domcom_endowment_average } remove_trait = domcom_endowment_average }
        if = { limit = { has_trait = domcom_endowment_big } remove_trait = domcom_endowment_big }
        if = { limit = { has_trait = domcom_endowment_massive } remove_trait = domcom_endowment_massive }
        if = { limit = { NOT = { has_trait = domcom_endowment_hung } } add_trait = domcom_endowment_hung }
    }
    else_if = {
        limit = { var:domcom_endowment_category = flag:massive }
        if = { limit = { has_trait = domcom_endowment_micro } remove_trait = domcom_endowment_micro }
        if = { limit = { has_trait = domcom_endowment_small } remove_trait = domcom_endowment_small }
        if = { limit = { has_trait = domcom_endowment_average } remove_trait = domcom_endowment_average }
        if = { limit = { has_trait = domcom_endowment_big } remove_trait = domcom_endowment_big }
        if = { limit = { has_trait = domcom_endowment_hung } remove_trait = domcom_endowment_hung }
        if = { limit = { NOT = { has_trait = domcom_endowment_massive } } add_trait = domcom_endowment_massive }
    }
}
