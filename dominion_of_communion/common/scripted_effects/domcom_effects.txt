
# Endowment has three permanent pieces of state: length, girth, and tier.
# The tier is stored separately from the derived working category so ordinary
# profile refreshes cannot reclassify a character during Alpha Check/Tantalize.
domcom_restore_locked_endowment_category_effect = {
    if = { limit = { var:domcom_locked_endowment_category = flag:micro } set_variable = { name = domcom_endowment_category value = flag:micro } }
    else_if = { limit = { var:domcom_locked_endowment_category = flag:small } set_variable = { name = domcom_endowment_category value = flag:small } }
    else_if = { limit = { var:domcom_locked_endowment_category = flag:average } set_variable = { name = domcom_endowment_category value = flag:average } }
    else_if = { limit = { var:domcom_locked_endowment_category = flag:big } set_variable = { name = domcom_endowment_category value = flag:big } }
    else_if = { limit = { var:domcom_locked_endowment_category = flag:hung } set_variable = { name = domcom_endowment_category value = flag:hung } }
    else_if = { limit = { var:domcom_locked_endowment_category = flag:massive } set_variable = { name = domcom_endowment_category value = flag:massive } }
}

domcom_lock_current_endowment_category_effect = {
    if = { limit = { var:domcom_endowment_category = flag:micro } set_variable = { name = domcom_locked_endowment_category value = flag:micro } }
    else_if = { limit = { var:domcom_endowment_category = flag:small } set_variable = { name = domcom_locked_endowment_category value = flag:small } }
    else_if = { limit = { var:domcom_endowment_category = flag:average } set_variable = { name = domcom_locked_endowment_category value = flag:average } }
    else_if = { limit = { var:domcom_endowment_category = flag:big } set_variable = { name = domcom_locked_endowment_category value = flag:big } }
    else_if = { limit = { var:domcom_endowment_category = flag:hung } set_variable = { name = domcom_locked_endowment_category value = flag:hung } }
    else_if = { limit = { var:domcom_endowment_category = flag:massive } set_variable = { name = domcom_locked_endowment_category value = flag:massive } }
}

# 0.5.0 — stable character profiles. Measurements are generated once and persist.
domcom_assign_endowment_effect = {
    # 1.0.0-rc4.2 — exact measurements are immutable after their first valid
    # assignment. The locked copies are authoritative even if another mod or a
    # later profile refresh changes/removes the working variables. Alpha,
    # Tantalize, birthdays and compatibility synchronization therefore cannot
    # reroll anatomy. Only the explicit destructive reroll clears the lock.
    if = {
        limit = {
            is_male = yes
            age >= 18
            has_variable = domcom_locked_length_tenths
            has_variable = domcom_locked_girth_tenths
        }
        set_variable = { name = domcom_length_tenths value = var:domcom_locked_length_tenths }
        set_variable = { name = domcom_girth_tenths value = var:domcom_locked_girth_tenths }
    }
    if = {
        limit = { is_male = yes age >= 18 has_variable = domcom_locked_length_tenths NOT = { has_variable = domcom_locked_girth_tenths } }
        set_variable = { name = domcom_length_tenths value = var:domcom_locked_length_tenths }
    }
    if = {
        limit = { is_male = yes age >= 18 has_variable = domcom_locked_girth_tenths NOT = { has_variable = domcom_locked_length_tenths } }
        set_variable = { name = domcom_girth_tenths value = var:domcom_locked_girth_tenths }
    }

    # Give an already-present compatibility source one chance to seed a brand
    # new profile. This never runs once Dominion has a locked measurement pair.
    if = {
        limit = {
            is_male = yes
            age >= 18
            NOT = { has_variable = domcom_locked_length_tenths }
            NOT = { has_variable = domcom_locked_girth_tenths }
            NOT = { has_variable = domcom_length_tenths }
            NOT = { has_variable = domcom_girth_tenths }
        }
        domcom_preseed_endowment_effect = yes
    }

    # 1.0.0-rc4.3 — heredity, and the reason the Endowment traits can honestly
    # call themselves congenital.
    #
    # The traits themselves are pinned to inherit_chance = 0, because a trait
    # handed down by the engine would arrive without measurements behind it and
    # read 0.00 in x 0.00 in until adulthood. Inheritance is applied to the
    # numbers instead, one level below the trait, so the tier a son ends up in
    # is inherited in fact and always agrees with his own figures.
    #
    # The sire's measurements are normally captured at birth while he is
    # certainly still around; this call is the catch-up for old saves and for
    # characters the game generates already adult.
    domcom_capture_endowment_heredity_effect = yes

    # Sired roll. Regression to the mean is 40% per generation, so a dynasty
    # can drift toward an extreme over several generations but does not lock
    # itself there, and no single birth is a copy of the father.
    if = {
        limit = {
            is_male = yes
            age >= 18
            NOT = { has_variable = domcom_length_tenths }
            has_variable = domcom_sire_length_tenths
        }
        set_variable = { name = domcom_length_tenths value = domcom_sire_length_seed_value }
        random_list = {
            6  = { change_variable = { name = domcom_length_tenths add = -20 } }
            10 = { change_variable = { name = domcom_length_tenths add = -15 } }
            15 = { change_variable = { name = domcom_length_tenths add = -10 } }
            19 = { change_variable = { name = domcom_length_tenths add = -5 } }
            20 = { change_variable = { name = domcom_length_tenths add = 0 } }
            19 = { change_variable = { name = domcom_length_tenths add = 5 } }
            15 = { change_variable = { name = domcom_length_tenths add = 10 } }
            10 = { change_variable = { name = domcom_length_tenths add = 15 } }
            6  = { change_variable = { name = domcom_length_tenths add = 20 } }
        }
        if = { limit = { var:domcom_length_tenths < 20 } set_variable = { name = domcom_length_tenths value = 20 } }
        if = { limit = { var:domcom_length_tenths > 130 } set_variable = { name = domcom_length_tenths value = 130 } }
    }
    if = {
        limit = {
            is_male = yes
            age >= 18
            NOT = { has_variable = domcom_girth_tenths }
            has_variable = domcom_sire_girth_tenths
        }
        set_variable = { name = domcom_girth_tenths value = domcom_sire_girth_seed_value }
        random_list = {
            6  = { change_variable = { name = domcom_girth_tenths add = -15 } }
            12 = { change_variable = { name = domcom_girth_tenths add = -10 } }
            20 = { change_variable = { name = domcom_girth_tenths add = -5 } }
            24 = { change_variable = { name = domcom_girth_tenths add = 0 } }
            20 = { change_variable = { name = domcom_girth_tenths add = 5 } }
            12 = { change_variable = { name = domcom_girth_tenths add = 10 } }
            6  = { change_variable = { name = domcom_girth_tenths add = 15 } }
        }
        if = { limit = { var:domcom_girth_tenths < 20 } set_variable = { name = domcom_girth_tenths value = 20 } }
        if = { limit = { var:domcom_girth_tenths > 85 } set_variable = { name = domcom_girth_tenths value = 85 } }
    }

    # Roll only the component that is actually missing. This is deliberately
    # split so a partial old-save profile cannot cause its existing dimension
    # to be replaced while repairing the other one. A character with a sired
    # value already set above falls straight through both blocks.
    if = {
        limit = { is_male = yes age >= 18 NOT = { has_variable = domcom_length_tenths } }
        random_list = {
            2 = { set_variable = { name = domcom_length_tenths value = 20 } }
            6 = { set_variable = { name = domcom_length_tenths value = 25 } }
            22 = { set_variable = { name = domcom_length_tenths value = 30 } }
            70 = { set_variable = { name = domcom_length_tenths value = 35 } }
            170 = { set_variable = { name = domcom_length_tenths value = 40 } }
            500 = { set_variable = { name = domcom_length_tenths value = 45 } }
            950 = { set_variable = { name = domcom_length_tenths value = 50 } }
            1150 = { set_variable = { name = domcom_length_tenths value = 55 } }
            800 = { set_variable = { name = domcom_length_tenths value = 60 } }
            520 = { set_variable = { name = domcom_length_tenths value = 65 } }
            340 = { set_variable = { name = domcom_length_tenths value = 70 } }
            210 = { set_variable = { name = domcom_length_tenths value = 75 } }
            130 = { set_variable = { name = domcom_length_tenths value = 80 } }
            85 = { set_variable = { name = domcom_length_tenths value = 85 } }
            55 = { set_variable = { name = domcom_length_tenths value = 90 } }
            35 = { set_variable = { name = domcom_length_tenths value = 95 } }
            22 = { set_variable = { name = domcom_length_tenths value = 100 } }
            12 = { set_variable = { name = domcom_length_tenths value = 110 } }
            6 = { set_variable = { name = domcom_length_tenths value = 120 } }
            3 = { set_variable = { name = domcom_length_tenths value = 130 } }
        }
    }
    if = {
        limit = { is_male = yes age >= 18 NOT = { has_variable = domcom_girth_tenths } }
        random_list = {
            2 = { set_variable = { name = domcom_girth_tenths value = 20 } }
            6 = { set_variable = { name = domcom_girth_tenths value = 25 } }
            28 = { set_variable = { name = domcom_girth_tenths value = 30 } }
            90 = { set_variable = { name = domcom_girth_tenths value = 35 } }
            340 = { set_variable = { name = domcom_girth_tenths value = 40 } }
            780 = { set_variable = { name = domcom_girth_tenths value = 45 } }
            1000 = { set_variable = { name = domcom_girth_tenths value = 50 } }
            780 = { set_variable = { name = domcom_girth_tenths value = 55 } }
            460 = { set_variable = { name = domcom_girth_tenths value = 60 } }
            250 = { set_variable = { name = domcom_girth_tenths value = 65 } }
            130 = { set_variable = { name = domcom_girth_tenths value = 70 } }
            60 = { set_variable = { name = domcom_girth_tenths value = 75 } }
            26 = { set_variable = { name = domcom_girth_tenths value = 80 } }
            10 = { set_variable = { name = domcom_girth_tenths value = 85 } }
        }
    }

    # Adopt the first complete pair as the permanent profile. This also
    # migrates existing saves without changing their current measurements.
    if = {
        limit = {
            is_male = yes
            age >= 18
            has_variable = domcom_length_tenths
            has_variable = domcom_girth_tenths
            OR = {
                NOT = { has_variable = domcom_locked_length_tenths }
                NOT = { has_variable = domcom_locked_girth_tenths }
            }
        }
        if = { limit = { NOT = { has_variable = domcom_locked_length_tenths } } set_variable = { name = domcom_locked_length_tenths value = var:domcom_length_tenths } }
        if = { limit = { NOT = { has_variable = domcom_locked_girth_tenths } } set_variable = { name = domcom_locked_girth_tenths value = var:domcom_girth_tenths } }
        add_character_flag = domcom_endowment_locked
    }

    # Restore the permanent measurements before display work. The tier itself is
    # also permanent once established. Ordinary profile calls therefore reassert
    # state; they never classify the character again.
    if = {
        limit = {
            is_male = yes age >= 18
            has_variable = domcom_locked_length_tenths
            has_variable = domcom_locked_girth_tenths
        }
        set_variable = { name = domcom_length_tenths value = var:domcom_locked_length_tenths }
        set_variable = { name = domcom_girth_tenths value = var:domcom_locked_girth_tenths }

        if = {
            limit = { has_variable = domcom_locked_endowment_category }
            domcom_restore_locked_endowment_category_effect = yes
            if = { limit = { NOT = { has_variable = domcom_endowment_index } } set_variable = { name = domcom_endowment_index value = domcom_endowment_index_value } }
            domcom_apply_native_endowment_trait_effect = yes
            domcom_post_category_effect = yes
        }
        else = {
            # First initialization (or an old save): classify once from the now
            # permanent measurements, lock that tier, and never recalculate it
            # during normal gameplay again.
            domcom_sync_endowment_trait_effect = yes
        }
    }
}

# Converts the hidden measurements into one visible physical trait. This is run
# even for old saves so the former character-modifier display migrates cleanly.
domcom_sync_endowment_trait_effect = {
    if = {
        limit = { is_male = yes age >= 18 has_variable = domcom_length_tenths has_variable = domcom_girth_tenths }

        # Legacy 0.5.x display modifiers, guarded so this is a true no-op once
        # a character has already been migrated.
        if = { limit = { has_character_modifier = domcom_endowment_modest_modifier } remove_character_modifier = domcom_endowment_modest_modifier }
        if = { limit = { has_character_modifier = domcom_endowment_average_modifier } remove_character_modifier = domcom_endowment_average_modifier }
        if = { limit = { has_character_modifier = domcom_endowment_noteworthy_modifier } remove_character_modifier = domcom_endowment_noteworthy_modifier }
        if = { limit = { has_character_modifier = domcom_endowment_enviable_modifier } remove_character_modifier = domcom_endowment_enviable_modifier }
        if = { limit = { has_character_modifier = domcom_endowment_legendary_modifier } remove_character_modifier = domcom_endowment_legendary_modifier }

        # 0.7.1 — the category variable is the single source of truth. It is set
        # here unconditionally, because setting a variable is silent and every
        # other system (Ledger, sorting, Carnalitas sync) reads it.
        #
        # 1.0.0-rc4.4 — the index is recomputed HERE, from the measurements this
        # character currently holds, instead of trusting whatever the caller
        # last wrote into domcom_endowment_index.
        #
        # That trust was a real bug. Every band below reads the variable, so any
        # path that reached this effect with an index older than the numbers put
        # the character in the wrong tier - and because the tooltip prints the
        # measurements live while the trait is only rewritten on sync, the two
        # halves of the tooltip openly contradicted each other. Two characters
        # both indexing 56 came out as big and small respectively. Recomputing
        # first makes this effect self-contained and removes the whole class.
        set_variable = { name = domcom_endowment_index value = domcom_endowment_index_value }

        # Bands for the 0.6 length / 0.4 girth weighting: 52 / 57 / 63 / 70.
        #
        # These are tuned to a target shape rather than left where the raw
        # distribution falls. The natural boundaries put 41% of men in average
        # and only 8.5% in small, which made small a rarity it has no reason to
        # be. Moving the small line down to 52 and the big line up to 57 gives
        # small 27%, average 28% and big 23% - three comparable middle bands
        # instead of one dominant one, with small slightly ahead of big.
        #
        # Band edges are whole numbers on the index, so these are the closest
        # available fit; there is no finer adjustment short of re-weighting the
        # roll tables themselves.
        #
        # Micro stays length-led rather than index-led, approximating the
        # real-life definition: girth cannot lift a character out of it.
        if = { limit = { var:domcom_length_tenths <= 40 } set_variable = { name = domcom_endowment_category value = flag:micro } }
        else_if = { limit = { var:domcom_endowment_index < 52 } set_variable = { name = domcom_endowment_category value = flag:small } }
        else_if = { limit = { var:domcom_endowment_index < 57 } set_variable = { name = domcom_endowment_category value = flag:average } }
        else_if = { limit = { var:domcom_endowment_index < 63 } set_variable = { name = domcom_endowment_category value = flag:big } }
        else_if = { limit = { var:domcom_endowment_index < 70 } set_variable = { name = domcom_endowment_category value = flag:hung } }
        else = { set_variable = { name = domcom_endowment_category value = flag:massive } }

        # Classification is a one-time operation. Save the result separately so
        # later Alpha Checks, Tantalize calls and profile refreshes cannot move a
        # character between tiers. Rebuild/Reroll explicitly clear this lock.
        if = {
            limit = { NOT = { has_variable = domcom_locked_endowment_category } }
            domcom_lock_current_endowment_category_effect = yes
        }

        # The Dominion physical trait is always retained. When the optional
        # Carnalitas bridge is loaded, Carnalitas' DT trait sits alongside it:
        # Carnalitas supplies its own artwork/mechanics while this Dominion
        # trait supplies exact measurements, range, and preference context.
        domcom_apply_native_endowment_trait_effect = yes

        # 0.8.3 — anything that owns endowment display elsewhere gets told the
        # category changed, every time Dominion computes one. Previously the
        # Carnalitas sync only ran from Carnalitas' own initialization hooks, so
        # a character first measured by an Alpha Check or Tantalize received a
        # Dominion trait while keeping whatever Carnalitas trait they already
        # had. That is the doubling, and it is why the pairing looked wrong
        # numerically: the two traits were set at different times from
        # different data.
        domcom_post_category_effect = yes
    }
}

# A character can arrive already carrying a tier: a Dominion Endowment trait
# picked in the ruler designer, or a Carnalitas dick trait picked there when the
# compatibility module is loaded. The trait by itself carries no measurements,

# touch a profile that already exists.

# 1.0.0-rc4.3 — record the sire's locked measurements on a male character so
# his own roll at eighteen can descend from them.
#
# Called from on_birth_child, which is the honest moment for it: the father is
# alive and measured there, whereas at the son's eighteenth birthday the father
# is frequently dead and his variables are no longer safe to read. It is also
# called defensively at the top of the roll for old saves and for characters
# the game generates already adult.
#
# real_father wins over father, so a bastard descends from whoever actually
# sired him rather than from the husband who thinks he did. That is consistent
# with the rest of the mod, and it is invisible: nothing displays this variable,
# so it leaks no more than the hidden parentage the base game already tracks.
#
# The whole effect short-circuits on one variable check once a sire is on
# record, and it never overwrites an existing one.
domcom_capture_endowment_heredity_effect = {
    if = {
        limit = {
            is_male = yes
            NOT = { has_variable = domcom_sire_length_tenths }
        }
        if = {
            limit = {
                exists = real_father
                real_father = { has_variable = domcom_locked_length_tenths }
            }
            real_father = { save_temporary_scope_as = domcom_heredity_sire }
        }
        else_if = {
            limit = {
                exists = father
                father = { has_variable = domcom_locked_length_tenths }
            }
            father = { save_temporary_scope_as = domcom_heredity_sire }
        }
        if = {
            limit = { exists = scope:domcom_heredity_sire }
            set_variable = {
                name = domcom_sire_length_tenths
                value = scope:domcom_heredity_sire.var:domcom_locked_length_tenths
            }
            if = {
                limit = { scope:domcom_heredity_sire = { has_variable = domcom_locked_girth_tenths } }
                set_variable = {
                    name = domcom_sire_girth_tenths
                    value = scope:domcom_heredity_sire.var:domcom_locked_girth_tenths
                }
            }
        }
    }
}

# Characters attracted to men receive three distinct ranked size preferences.
# Primary is strongest, secondary moderate, tertiary mild. Anything outside all
# three is a moderate mismatch. Preferences are hidden and randomized; bigger is
# therefore not inherently better. The old 0.5.x single preference is discarded.
domcom_assign_size_preference_effect = {
    if = { limit = { has_variable = domcom_size_preference } remove_variable = domcom_size_preference }

    if = {
        limit = {
            age >= 18
            is_attracted_to_men = yes
            NOT = { has_variable = domcom_size_preference_primary }
        }

        # Primary preference: centered on ordinary sizes, but every tier is possible.
        random_list = {
            5  = { set_variable = { name = domcom_size_preference_primary value = flag:micro } }
            17 = { set_variable = { name = domcom_size_preference_primary value = flag:small } }
            30 = { set_variable = { name = domcom_size_preference_primary value = flag:average } }
            25 = { set_variable = { name = domcom_size_preference_primary value = flag:big } }
            16 = { set_variable = { name = domcom_size_preference_primary value = flag:hung } }
            7  = { set_variable = { name = domcom_size_preference_primary value = flag:massive } }
        }

        # Secondary cannot duplicate the primary.
        random_list = {
            5  = { trigger = { NOT = { var:domcom_size_preference_primary = flag:micro } } set_variable = { name = domcom_size_preference_secondary value = flag:micro } }
            17 = { trigger = { NOT = { var:domcom_size_preference_primary = flag:small } } set_variable = { name = domcom_size_preference_secondary value = flag:small } }
            30 = { trigger = { NOT = { var:domcom_size_preference_primary = flag:average } } set_variable = { name = domcom_size_preference_secondary value = flag:average } }
            25 = { trigger = { NOT = { var:domcom_size_preference_primary = flag:big } } set_variable = { name = domcom_size_preference_secondary value = flag:big } }
            16 = { trigger = { NOT = { var:domcom_size_preference_primary = flag:hung } } set_variable = { name = domcom_size_preference_secondary value = flag:hung } }
            7  = { trigger = { NOT = { var:domcom_size_preference_primary = flag:massive } } set_variable = { name = domcom_size_preference_secondary value = flag:massive } }
        }

        # Tertiary cannot duplicate either higher-ranked preference.
        random_list = {
            5  = { trigger = { NOT = { var:domcom_size_preference_primary = flag:micro } NOT = { var:domcom_size_preference_secondary = flag:micro } } set_variable = { name = domcom_size_preference_tertiary value = flag:micro } }
            17 = { trigger = { NOT = { var:domcom_size_preference_primary = flag:small } NOT = { var:domcom_size_preference_secondary = flag:small } } set_variable = { name = domcom_size_preference_tertiary value = flag:small } }
            30 = { trigger = { NOT = { var:domcom_size_preference_primary = flag:average } NOT = { var:domcom_size_preference_secondary = flag:average } } set_variable = { name = domcom_size_preference_tertiary value = flag:average } }
            25 = { trigger = { NOT = { var:domcom_size_preference_primary = flag:big } NOT = { var:domcom_size_preference_secondary = flag:big } } set_variable = { name = domcom_size_preference_tertiary value = flag:big } }
            16 = { trigger = { NOT = { var:domcom_size_preference_primary = flag:hung } NOT = { var:domcom_size_preference_secondary = flag:hung } } set_variable = { name = domcom_size_preference_tertiary value = flag:hung } }
            7  = { trigger = { NOT = { var:domcom_size_preference_primary = flag:massive } NOT = { var:domcom_size_preference_secondary = flag:massive } } set_variable = { name = domcom_size_preference_tertiary value = flag:massive } }
        }
    }
}

# light on raw stats: the important mechanical transition is the final Pet state.

            limit = {
                has_variable = domcom_pet_dynasty
                NOT = { has_nickname = nick_domcom_dynasty_slave }

            give_nickname = nick_domcom_dynasty_slave

        else_if = {
            limit = {
                has_variable = domcom_pet_patron
                NOT = { has_nickname = nick_domcom_personal_slave }
            }

# Run from a character who just reached 1000 Obedience, with
# scope:domcom_tantalize_patron saved. Pet numbering is stored on the Patron's
# dynasty so it survives succession and counts across the dynasty rather than one ruler.
domcom_create_pet_effect = {
    save_scope_as = domcom_pet_subject
    add_character_flag = domcom_pet
    add_character_flag = domcom_new_pet_created
    set_variable = { name = domcom_pet_patron value = scope:domcom_tantalize_patron }
    set_variable = { name = domcom_obedience_patron value = scope:domcom_tantalize_patron }
    set_variable = { name = domcom_obedience value = 1000 }
    add_character_modifier = domcom_pet_modifier
    add_opinion = { target = scope:domcom_tantalize_patron modifier = domcom_pet_obedience_opinion }

    if = {
        limit = { exists = scope:domcom_tantalize_patron.dynasty }
        scope:domcom_tantalize_patron.dynasty = {
            if = {
                limit = { NOT = { has_variable = domcom_pet_counter } }
                set_variable = { name = domcom_pet_counter value = 0 }
            }
            change_variable = { name = domcom_pet_counter add = 1 }
            save_scope_as = domcom_pet_dynasty_scope
        }
        set_variable = { name = domcom_pet_dynasty value = scope:domcom_pet_dynasty_scope }
        set_variable = { name = domcom_pet_number value = scope:domcom_pet_dynasty_scope.var:domcom_pet_counter }
        give_nickname = nick_domcom_dynasty_slave
    }
    else = {
        scope:domcom_tantalize_patron = {
            if = {
                limit = { NOT = { has_variable = domcom_personal_pet_counter } }
                set_variable = { name = domcom_personal_pet_counter value = 0 }
            }
            change_variable = { name = domcom_personal_pet_counter add = 1 }
        }
        set_variable = { name = domcom_pet_number value = scope:domcom_tantalize_patron.var:domcom_personal_pet_counter }
        give_nickname = nick_domcom_personal_slave
    }

    scope:domcom_tantalize_patron = {
        add_to_variable_list = { name = domcom_pet_registry target = scope:domcom_pet_subject }
        if = {
            limit = { can_add_hook = { target = scope:domcom_pet_subject type = loyalty_hook } }
            add_hook_no_toast = { target = scope:domcom_pet_subject type = loyalty_hook }
        }
        save_scope_as = domcom_owner
    }

    # Full obedience finally grants the same kind of household access that an
    # Outclassed rival can expose, but Pet itself remains distinct from Cucked/Owned.
    domcom_schedule_family_attraction_checks_effect = yes
    domcom_refresh_obedience_stage_effect = yes
    trigger_event = { on_action = domcom_pet_created }
}

# Run from the Tantalize recipient with scope:domcom_tantalize_patron saved. A
# base success now scales with compatibility; established intimate bonds accelerate the
# track, and a prior same-sex Alpha victory against this target adds extra leverage.
domcom_apply_tantalize_success_effect = {
    if = {
        limit = { NOT = { has_variable = domcom_obedience_patron } }
        set_variable = { name = domcom_obedience_patron value = scope:domcom_tantalize_patron }
    }
    if = { limit = { NOT = { has_variable = domcom_obedience } } set_variable = { name = domcom_obedience value = 0 } }

    # 0.9.7 — enter the Patron's Tantalized registry on the FIRST success, not at
    # Pet. The old code only ever registered characters who reached 1000, so
    # everyone between 1 and 999 Obedience existed in state but nowhere in the UI.
    if = {
        limit = { NOT = { has_character_flag = domcom_tantalized_registered } }
        add_character_flag = domcom_tantalized_registered
        save_temporary_scope_as = domcom_tantalize_subject
        scope:domcom_tantalize_patron = {
            add_to_variable_list = { name = domcom_tantalized_registry target = scope:domcom_tantalize_subject }
        }
    }

    change_variable = { name = domcom_obedience add = domcom_tantalize_obedience_gain_value }
    if = { limit = { has_relation_lover = scope:domcom_tantalize_patron } change_variable = { name = domcom_obedience add = 50 } }
    if = { limit = { has_relation_soulmate = scope:domcom_tantalize_patron } change_variable = { name = domcom_obedience add = 100 } }
    if = {
        limit = {
            has_character_flag = domcom_dominated
            has_variable = domcom_dominance_patron
            var:domcom_dominance_patron = scope:domcom_tantalize_patron
        }
        change_variable = { name = domcom_obedience add = 25 }
    }

    # A first successful Tantalize creates interest, not an affair. Vanilla
    # Seduce/Romance or later Dominion mechanics can still determine intimacy.
    if = {
        limit = {
            NOR = {
                has_relation_crush = scope:domcom_tantalize_patron
                has_relation_lover = scope:domcom_tantalize_patron
                has_relation_soulmate = scope:domcom_tantalize_patron
            }
        }
        set_relation_crush = scope:domcom_tantalize_patron
    }

    if = { limit = { var:domcom_obedience > 1000 } set_variable = { name = domcom_obedience value = 1000 } }
    domcom_refresh_obedience_stage_effect = yes

    if = {
        limit = {
            var:domcom_obedience >= 1000
            NOT = { has_character_flag = domcom_pet }
        }
        domcom_create_pet_effect = yes
    }
}

# Transfer a broken ruler's realm to the patron. scope:domcom_collapse_patron saved.
domcom_ai_resolve_collapse_effect = {
    save_scope_as = domcom_surrendering_subject
    domcom_surrender_prestige_if_lower_effect = yes
    if = { limit = { gold > 0 } pay_short_term_gold = { target = scope:domcom_collapse_patron gold = gold } }
    if = {
        limit = { is_landed = yes }
        # 0.8.7 — was type = abdication. CK3 models abdication as a ruler standing
        # down in favour of their own successor, so using it to hand a realm to an
        # unrelated patron could resolve to nothing at all, silently: the event
        # played, the option was taken, and no title moved.
        # granted is the type that actually describes this — the subject grants
        # their holdings away — and it carries no relationship requirement.
        create_title_and_vassal_change = {
            type = granted
            save_scope_as = domcom_surrender_change
            add_claim_on_loss = yes
        }
        every_held_title = {
            limit = {
                NOT = { tier = tier_barony }
                NOT = { is_landless_type_title = yes }
            }
            change_title_holder = {
                holder = scope:domcom_collapse_patron
                change = scope:domcom_surrender_change
                government_base = scope:domcom_collapse_patron
            }
        }
        resolve_title_and_vassal_change = scope:domcom_surrender_change
    }

    # 1.0.0-rc4.4 — destroy whatever refused to move, then retry.
    #
    # CK3 silently rejects a scripted transfer it considers illegal for the
    # receiving holder; Paradox's own title documentation gives the canonical
    # case, that a Muslim cannot be made the Catholic Pope. A head of faith
    # therefore kept his title through the surrender above, stayed landed, and
    # held on to the county beneath it - which also blocked him from entering
    # the patron's court further down.
    #
    # Rather than test for head-of-faith specifically, this destroys anything of
    # duchy tier or above still held after the surrender resolved. Whatever the
    # reason it would not move, it cannot move, and the patron is not getting
    # it. Counties and baronies are excluded because CK3 does not support
    # destroying them.
    #
    # The list is built first and destroyed second, rather than destroying
    # inside every_held_title, because mutating a list while iterating it is how
    # you end up operating on a scope the game has already removed.
    if = {
        limit = { is_landed = yes }
        every_held_title = {
            limit = {
                tier >= tier_duchy
                NOT = { is_landless_type_title = yes }
            }
            add_to_temporary_list = domcom_stuck_titles
        }
        every_in_list = {
            list = domcom_stuck_titles
            save_temporary_scope_as = domcom_stuck_title
            destroy_title = scope:domcom_stuck_title
        }
    }

    # Second pass. Counties orphaned by the destruction above are now free to be
    # granted, so the patron still gains the land even though the religious
    # title itself is gone.
    if = {
        limit = { is_landed = yes }
        create_title_and_vassal_change = {
            type = granted
            save_scope_as = domcom_surrender_change_freed
            add_claim_on_loss = yes
        }
        every_held_title = {
            limit = {
                NOT = { tier = tier_barony }
                NOT = { is_landless_type_title = yes }
            }
            change_title_holder = {
                holder = scope:domcom_collapse_patron
                change = scope:domcom_surrender_change_freed
                government_base = scope:domcom_collapse_patron
            }
        }
        resolve_title_and_vassal_change = scope:domcom_surrender_change_freed
    }

    # 1.0.0-rc4.4 — a surrendered subject enters the patron's court rather than
    # drifting into the character pool. This is the other half of taking him
    # alive: he is only useful as a councillor, a commander or a marriage if he
    # is physically present.
    #
    # Guarded on is_landed = no because set_employer only applies to unlanded
    # characters. Anyone still holding a title after the surrender above falls
    # through untouched - which today includes a head of faith, whose title CK3
    # refuses to move to a holder of another faith.
    if = {
        limit = {
            is_landed = no
            exists = scope:domcom_collapse_patron
            NOT = { is_courtier_of = scope:domcom_collapse_patron }
        }
        set_employer = scope:domcom_collapse_patron
    }

    remove_character_flag = domcom_collapse_pending

    # 0.8.7 — only declare this resolved if it actually happened. Previously the
    # flag was set unconditionally, so a failed seizure left the subject in a
    # terminal state that could never be retried: Final Submission on the
    # character, Collapse greyed out in the Ledger, and their titles still theirs.
    if = {
        limit = { is_landed = yes }
        add_character_flag = domcom_collapse_seizure_failed
        # 0.8.9 — these two lines used to sit outside the branch, so a failed
        # seizure still stamped the subject with the terminal modifier and still
        # incremented the patron's resolved-collapse count. Retryable, but
        # already wearing Final Submission and already counted.
        if = {
            limit = { has_character_modifier = domcom_final_submission_modifier }
            remove_character_modifier = domcom_final_submission_modifier
        }
    }
    else = {
        add_character_flag = domcom_collapse_resolved
        add_character_modifier = domcom_final_submission_modifier
        scope:domcom_collapse_patron = { change_variable = { name = domcom_collapses_resolved add = 1 } }
    }
}

domcom_check_final_collapse_effect = {
    if = {
        limit = {
            has_character_flag = domcom_owned_cuck
            has_variable = domcom_cuck_xp
            var:domcom_cuck_xp >= 1000
            NOT = { has_character_flag = domcom_collapse_pending }
            NOT = { has_character_flag = domcom_collapse_resolved }
            has_variable = domcom_patron
            var:domcom_patron = { is_alive = yes }
        }
        save_scope_as = domcom_collapsing_subject
        var:domcom_patron = { save_scope_as = domcom_collapse_patron }

        # A fully collapsed Household Dominion resolves as a household once.
        if = {
            limit = {
                is_married = yes
                any_spouse = {
                    is_alive = yes
                    has_character_flag = domcom_owned_cuck
                    has_variable = domcom_patron
                    var:domcom_patron = scope:domcom_collapse_patron
                    has_variable = domcom_cuck_xp
                    var:domcom_cuck_xp >= 1000
                    NOT = { has_character_flag = domcom_collapse_pending }
                    NOT = { has_character_flag = domcom_collapse_resolved }
                }
            }
            every_spouse = {
                limit = {
                    is_alive = yes
                    has_character_flag = domcom_owned_cuck
                    has_variable = domcom_patron
                    var:domcom_patron = scope:domcom_collapse_patron
                    has_variable = domcom_cuck_xp
                    var:domcom_cuck_xp >= 1000

            }
            add_character_flag = domcom_collapse_pending
            scope:domcom_collapsing_spouse = { add_character_flag = domcom_collapse_pending }
            if = {
                limit = { scope:domcom_collapse_patron = { is_ai = no } }
                scope:domcom_collapse_patron = { trigger_event = { id = domcom.105 days = 1 } }
            }
            else = { domcom_resolve_household_collapse_effect = yes }
        }

        # A Patron's own intimate partner ends in permanent devotion, not destruction.
        else_if = {
            limit = {
                OR = {
                    is_consort_of = scope:domcom_collapse_patron
                    has_relation_lover = scope:domcom_collapse_patron
                    has_relation_soulmate = scope:domcom_collapse_patron
                }
            }
            add_character_flag = domcom_collapse_pending
            if = {
                limit = { scope:domcom_collapse_patron = { is_ai = no } }
                scope:domcom_collapse_patron = { trigger_event = { id = domcom.101 days = 1 } }
            }
            else = { domcom_resolve_devotion_collapse_effect = yes }
        }

        else = {
            add_character_flag = domcom_collapse_pending
            if = {
                limit = { scope:domcom_collapse_patron = { is_ai = no } }
                scope:domcom_collapse_patron = { trigger_event = { id = domcom.100 days = 1 } }
            }
            else = { domcom_ai_resolve_collapse_effect = yes }
        }
    }

# Resolve an admirer encounter. Run from admirer with scope:domcom_admirer_patron.
# rc4.7 — durable Dominion intimate protection.
#
# The earlier rc4.4 implementation tried to suppress the Unfaithful chain by
# deleting `secret_adultery`. Current CK3's infidelity system actually tracks the
# lover secret as `secret_lover`, and the engine's own confrontation bootstrap
# explicitly respects the character flag `block_infidelity_flag`. The old effect
# therefore neither removed the relevant secret nor prevented a later generic
# breakup. Dominion now uses the engine-supported block and remembers the exact
# intimate so Enforce Dominion can repair a relationship if another event still
# removes it. No vanilla/Carnalitas event file is overwritten.

# Run from a Dominion intimate with scope:domcom_communion_patron saved.
# Only the secret between this exact pair is removed; unrelated affairs keep
# vanilla/Carnalitas consequences.
domcom_clear_communion_pair_secrets_effect = {
    save_scope_as = domcom_communion_partner
    if = {
        limit = {
            exists = scope:domcom_communion_patron
            NOT = { has_character_flag = domcom_allow_adultery_secrets }
        }
        every_secret = {
            type = secret_lover
            limit = { secret_target = scope:domcom_communion_patron }
            remove_secret = yes
        }
        scope:domcom_communion_patron = {
            every_secret = {
                type = secret_lover
                limit = { secret_target = scope:domcom_communion_partner }
                remove_secret = yes
            }
        }
    }
}

# Run from a character whose Lover/Soulmate bond was established by Dominion,
# with scope:domcom_communion_patron saved.
domcom_register_communion_intimate_effect = {
    save_scope_as = domcom_communion_partner
    if = {
        limit = {
            is_alive = yes
            exists = scope:domcom_communion_patron
            NOT = { this = scope:domcom_communion_patron }
        }
        add_character_flag = domcom_communion_intimate_bound
        # Vanilla and Carnalitas' retained vanilla infidelity bootstrap both check
        # this flag before starting the Unfaithful confrontation story. It is put

        add_character_flag = block_infidelity_flag

                    NOT = {

                            variable = domcom_intimate_registry
                            this = scope:domcom_communion_partner
                        }
                    }
                }
                add_to_variable_list = { name = domcom_intimate_registry target = scope:domcom_communion_partner }
            }
        }
        domcom_clear_communion_pair_secrets_effect = yes
    }
}

                    has_relation_soulmate = scope:domcom_communion_patron
                }
            }
            domcom_register_communion_intimate_effect = yes
        }
    }

    # Dominion admirer has lost every romantic relation, but is still married to

        }
    }

    # to the same ring the quarterly pass uses, and domcom_needs_profile_init_trigger
    # makes it nearly free once a court has already been covered. Characters outside
    # the ring remain lazily initialized when they enter a roster or interaction, and
    # new adults are still initialized at 18 through on_birthday.
    domcom_initialize_local_population_effect = yes
    domcom_fast_recover_records_effect = yes
    domcom_repair_patron_registries_effect = yes

    # Cucked rosters now read Dominion's authoritative global registry. Personal
    # Cucked means this Patron actually established cuck evidence; merely later
    # Owning an AI-created Cuck does not make them one of mine.
    if = {
        limit = { has_global_variable_list = domcom_all_cucked }
        every_in_global_list = {
            variable = domcom_all_cucked
            limit = { is_alive = yes age >= 18 has_character_flag = domcom_cucked }
            domcom_initialize_profile_effect = yes
            # Keep the displayed 0–1000 progression authoritative even if an old
            # save or an unusual AI path accumulated excess XP before this refresh.
            domcom_refresh_cuck_stage_effect = yes
            domcom_refresh_cuck_reputation_effect = yes
            save_temporary_scope_as = domcom_ui_roster_subject

            if = {
                limit = {
                    OR = {
                        any_in_list = {
                            variable = domcom_cuck_patrons
                            this = scope:domcom_ui_patron
                        }
                        AND = {
                            NOT = { has_variable_list = domcom_cuck_patrons }
                            has_variable = domcom_cuck_origin_patron
                            var:domcom_cuck_origin_patron = scope:domcom_ui_patron
                        }
                    }
                }
                set_variable = { name = domcom_ui_alpha_chance value = domcom_ui_alpha_check_chance_value }
                domcom_prepare_cucked_link_effect = yes
                scope:domcom_ui_patron = {
                    add_to_variable_list = { name = domcom_ui_cucked_raw target = scope:domcom_ui_roster_subject }
                    change_variable = { name = domcom_ui_cucked_count add = 1 }
                }
            }
            else = {
                # Observational world list: only people this player has never
                # personally cuckified. Legacy records with unrecoverable origin
                # remain visible and display "Unknown" rather than inventing one.
                scope:domcom_ui_patron = {
                    add_to_variable_list = { name = domcom_ui_world_cucks_raw target = scope:domcom_ui_roster_subject }
                    change_variable = { name = domcom_ui_world_cuck_count add = 1 }
                }
            }
        }
    }

    # Owned roster uses the repaired Patron-local registry. Dynasty storage is
    # still maintained for inheritance but is no longer a single point of failure.
    if = {
        limit = { has_variable_list = domcom_owned_registry }
        every_in_list = {
            variable = domcom_owned_registry
            limit = { is_alive = yes has_character_flag = domcom_owned_cuck has_variable = domcom_patron var:domcom_patron = scope:domcom_ui_patron }
            domcom_initialize_profile_effect = yes
            domcom_refresh_cuck_reputation_effect = yes
            save_temporary_scope_as = domcom_ui_owned_subject
            scope:domcom_ui_patron = {
                add_to_variable_list = { name = domcom_ui_owned_raw target = scope:domcom_ui_owned_subject }

            if = { limit = { highest_held_title_tier >= tier_kingdom } scope:domcom_ui_patron = { change_variable = { name = domcom_ui_royal_owned_count add = 1 } } }

        }
    }

    # Prospects were already stored on the patron at creation; no world scan needed.
    if = {
        limit = { has_variable_list = domcom_crush_candidates }

            variable = domcom_crush_candidates
            limit = {
                is_alive = yes age >= 18
                has_variable = domcom_crush_patron var:domcom_crush_patron = scope:domcom_ui_patron

                change_variable = { name = domcom_ui_prospect_count add = 1 }
            }
        }
    }

    # 0.9.7 — Children of Communion is now driven by the persistent blood
    # registry that the deep scan and every acknowledgment path maintain, not by
    # every_child. Two old assumptions were wrong:
    #   every_child returns only LEGAL children, so a hidden bastard whose
    #     real_father is me but whose father is the cuck never appeared unless
    #     that cuck's spouse happened to still be a current lover/soulmate; and
    #   the "not in my house" filter deleted every child I sired inside my own
    #     dynasty, which is precisely the inbred-dynasty case.
    # Eligibility is now a property of the child (mine by blood, not yet legally
    # mine), never a property of which house they were born into.
    if = {
        limit = { has_variable_list = domcom_blood_registry }
        every_in_list = {
            variable = domcom_blood_registry
            limit = { is_alive = yes domcom_is_unacknowledged_blood_trigger = yes }
            domcom_initialize_profile_effect = yes
            save_temporary_scope_as = domcom_ui_blood_child
            scope:domcom_ui_patron = {
                add_to_variable_list = { name = domcom_ui_blood_raw target = scope:domcom_ui_blood_child }
                change_variable = { name = domcom_ui_blood_count add = 1 }
            }
        }
    }

    # Cheap incremental catch for anyone born since the last scan: legal children
    # and the children of current intimates. The deep scan is what guarantees
    # completeness; this only keeps an open ledger current between scans.
    every_child = {
        limit = {
            is_alive = yes
            domcom_is_unacknowledged_blood_trigger = yes
            NOT = { has_character_flag = domcom_blood_registered }
        }
        save_temporary_scope_as = domcom_ui_blood_child
        add_character_flag = domcom_blood_registered
        domcom_initialize_profile_effect = yes
        scope:domcom_ui_patron = {
            add_to_variable_list = { name = domcom_blood_registry target = scope:domcom_ui_blood_child }
            add_to_variable_list = { name = domcom_ui_blood_raw target = scope:domcom_ui_blood_child }
            change_variable = { name = domcom_ui_blood_count add = 1 }
        }
    }
    every_relation = {
        type = lover
        type = soulmate
        limit = { is_alive = yes }
        every_child = {
            limit = {
                is_alive = yes
                domcom_is_unacknowledged_blood_trigger = yes
                NOT = { has_character_flag = domcom_blood_registered }
            }
            save_temporary_scope_as = domcom_ui_blood_child
            add_character_flag = domcom_blood_registered
            domcom_initialize_profile_effect = yes
            scope:domcom_ui_patron = {
                add_to_variable_list = { name = domcom_blood_registry target = scope:domcom_ui_blood_child }
                add_to_variable_list = { name = domcom_ui_blood_raw target = scope:domcom_ui_blood_child }
                change_variable = { name = domcom_ui_blood_count add = 1 }
            }
        }
    }

    # ISSUE 5 — the Tantalized roster. Obedience has always been tracked on the
    # subject; nothing ever collected it for display.
    set_variable = { name = domcom_ui_tantalized_count value = 0 }
    set_variable = { name = domcom_ui_tantalized_women value = 0 }
    set_variable = { name = domcom_ui_tantalized_men value = 0 }
    set_variable = { name = domcom_ui_pet_count value = 0 }
    clear_variable_list = domcom_ui_tantalized_raw
    clear_variable_list = domcom_ui_tantalized
    if = {
        limit = { has_variable_list = domcom_tantalized_registry }
        every_in_list = {
            variable = domcom_tantalized_registry
            limit = {
                is_alive = yes
                has_variable = domcom_obedience_patron
                var:domcom_obedience_patron = scope:domcom_ui_patron
            }
            domcom_initialize_profile_effect = yes
            save_temporary_scope_as = domcom_ui_tantalized_subject
            scope:domcom_ui_patron = {
                add_to_variable_list = { name = domcom_ui_tantalized_raw target = scope:domcom_ui_tantalized_subject }
                change_variable = { name = domcom_ui_tantalized_count add = 1 }
            }
            if = { limit = { is_female = yes } scope:domcom_ui_patron = { change_variable = { name = domcom_ui_tantalized_women add = 1 } } }
            else_if = { limit = { is_male = yes } scope:domcom_ui_patron = { change_variable = { name = domcom_ui_tantalized_men add = 1 } } }
            if = { limit = { has_character_flag = domcom_pet } scope:domcom_ui_patron = { change_variable = { name = domcom_ui_pet_count add = 1 } } }
        }
    }

    domcom_rebuild_ui_rosters_effect = yes
}

# Collect one year's tribute immediately. The subject pays first; a shortfall is
# collected from wealthy adult vassals, courtiers, and close/extended family.
domcom_demand_annual_tribute_effect = {
    save_scope_as = domcom_tribute_advance_subject
    var:domcom_patron = { save_scope_as = domcom_tribute_advance_patron }
    set_variable = { name = domcom_tribute_advance_shortfall value = domcom_annual_tribute_due_value }
    set_variable = { name = domcom_tribute_advance_collected value = 0 }

    if = {

    if = {

# Run from the hegemon. Builds a one-pass roster of direct landed vassals.
domcom_begin_imperial_tribute_effect = {
    clear_variable_list = domcom_tribute_pending
    set_variable = { name = domcom_tribute_pending_count value = 0 }
    set_variable = { name = domcom_tribute_summoned_count value = 0 }
    # Per-season counters. These are the ones the Tribute page reports and the
    # only ones a new season may reset.
    set_variable = { name = domcom_tribute_gold_count value = 0 }
    set_variable = { name = domcom_tribute_prestige_count value = 0 }
    set_variable = { name = domcom_tribute_military_count value = 0 }
    set_variable = { name = domcom_tribute_dynastic_count value = 0 }
    set_variable = { name = domcom_tribute_household_season_count value = 0 }
    set_variable = { name = domcom_tribute_refusal_season_count value = 0 }
    # domcom_tribute_household_count and domcom_tribute_refusal_count are LIFETIME
    # ledger counters and are never reset here; zeroing them on every summons was
    # wiping the career record the Ledger page reads.
    domcom_initialize_ledger_counters_effect = yes
    add_character_flag = domcom_tribute_season_active
    every_vassal = {
        limit = { is_alive = yes is_landed = yes }
        save_temporary_scope_as = domcom_tribute_vassal_to_add
        root = {
            add_to_variable_list = { name = domcom_tribute_pending target = scope:domcom_tribute_vassal_to_add }
            change_variable = { name = domcom_tribute_pending_count add = 1 }
        }
    }
    set_variable = { name = domcom_tribute_total_count value = var:domcom_tribute_pending_count }
    trigger_event = { id = domcom.500 days = 5 }
}

# Run from the patron when advancing the season.
domcom_advance_imperial_tribute_effect = {
    if = {
        limit = { has_character_flag = domcom_tribute_season_active has_variable_list = domcom_tribute_pending }

        # Drop petitioners who can no longer appear (dead, unlanded, no longer my
        # vassal) and recount from what is actually left, so the page never
        # advertises an audience that will never happen.
        save_scope_as = domcom_tribute_prune_patron
        every_in_list = {
            variable = domcom_tribute_pending
            limit = { NOT = { AND = { is_alive = yes is_landed = yes is_vassal_of = scope:domcom_tribute_prune_patron } } }
            save_temporary_scope_as = domcom_tribute_stale
            scope:domcom_tribute_prune_patron = {
                remove_list_variable = { name = domcom_tribute_pending target = scope:domcom_tribute_stale }
            }
        }
        set_variable = { name = domcom_tribute_pending_count value = 0 }
        every_in_list = {
            variable = domcom_tribute_pending
            scope:domcom_tribute_prune_patron = { change_variable = { name = domcom_tribute_pending_count add = 1 } }
        }

        random_in_list = {
            variable = domcom_tribute_pending
            limit = { is_alive = yes is_landed = yes is_vassal_of = root }
            save_scope_as = domcom_tribute_vassal
        }
        if = {
            limit = { exists = scope:domcom_tribute_vassal }
            remove_list_variable = { name = domcom_tribute_pending target = scope:domcom_tribute_vassal }
            change_variable = { name = domcom_tribute_pending_count subtract = 1 }
            change_variable = { name = domcom_tribute_summoned_count add = 1 }
            save_scope_as = domcom_tribute_patron
            scope:domcom_tribute_vassal = { trigger_event = { id = domcom.510 days = 1 } }
        }
        else = { domcom_close_tribute_season_effect = yes }
    }
    else_if = {
        limit = { has_character_flag = domcom_tribute_season_active }
        domcom_close_tribute_season_effect = yes
    }
}

# the player's orbit without a synchronous world scan. Each quarterly
# player pulse attempts a handful of still-unprofiled adult men.
# 0.7.0 — one-shot world seed for saves that did not start with Dominion.
#
# 0.5.7 removed the world scan to kill the Ledger-open hitch, and the random
# sampler that replaced it only advanced while the game was unpaused, so an
# existing save filled in at a trickle and most characters showed nothing.
# The optional compatibility module handles equivalent initialization without a duplicate living-character pass
# at startup, so this now does the same: once per save, gated on a global
# variable set before the scan runs so a re-entrant call cannot double it.
# Endowment only, no hidden preference work, which is the expensive half.
#
# Players who would rather keep the old lazy behaviour can set the
# "Existing Save Seeding" game rule to Off.
# 0.8.2 — the gated wrapper. Runs at most once per save and obeys the
# Existing Save Seeding rule. A rebuild must NOT go through this: it would be
# refused outright for anyone who set that rule to Off, leaving the save with
# every measurement wiped and nothing put back.
domcom_seed_world_endowments_once_effect = {
    if = {
        limit = {
            NOT = { has_global_variable = domcom_world_endowments_seeded_070 }
            NOT = { has_game_rule = domcom_world_seed_off }
        }
        domcom_seed_world_endowments_now_effect = yes
    }
}

# The scan itself, ungated. Only the wrapper above and the rebuild tools call it.
domcom_seed_world_endowments_now_effect = {
    set_global_variable = domcom_world_endowments_seeded_070
    every_living_character = {
        limit = {
            is_male = yes
            age >= 18
            NOT = { has_variable = domcom_length_tenths }
        }
        domcom_assign_endowment_effect = yes
    }
}

############################################################################
# 0.5.9 — BOUNDED LOCAL POPULATION INITIALIZATION
#
# 0.5.7 removed the every_living_character scan, which fixed the Ledger-open
# hitch but left every adult who existed at game start without an Endowment
# trait. Only players, characters turning 18 in-game, and characters pulled
# into a Dominion roster were ever initialized.
#
# This pass restores coverage without restoring the scan. It walks the
# patron's own orbit — court, close family, intimates, direct vassals and
# their spouses — which is O(court) rather than O(world). Characters the
# player can actually open in a character window get a profile; characters on
# the far side of the map do not, and are still initialized lazily if they
# ever enter a roster or interaction.
#
# domcom_needs_profile_init_trigger makes repeat passes nearly free.
############################################################################

domcom_try_initialize_profile_effect = {
    if = {
        limit = { domcom_needs_profile_init_trigger = yes }
        domcom_initialize_profile_effect = yes
    }
}

domcom_initialize_local_population_effect = {
    # The patron themselves.
    domcom_try_initialize_profile_effect = yes

    # Own court. This is the single largest source of visible characters.
    every_courtier = {
        limit = { domcom_needs_profile_init_trigger = yes }
        domcom_initialize_profile_effect = yes
    }

    # Prisoners are held locally and are inspectable.
    every_prisoner = {
        limit = { domcom_needs_profile_init_trigger = yes }
        domcom_initialize_profile_effect = yes
    }

    # Spouses and close family. Cheap and almost always inspected.
    every_spouse = {
        limit = { domcom_needs_profile_init_trigger = yes }
        domcom_initialize_profile_effect = yes
    }
    every_close_family_member = {
        limit = { domcom_needs_profile_init_trigger = yes }
        domcom_initialize_profile_effect = yes
    }

    # Existing intimates, which are exactly the characters Dominion evidence
    # is built from.
    every_relation = {
        type = lover
        type = soulmate
        type = crush
        limit = { domcom_needs_profile_init_trigger = yes }
        domcom_initialize_profile_effect = yes
    }

    # Direct vassals and their spouses. Not every_vassal_or_below — that would
    # walk the whole realm for a high-tier ruler and reintroduce the hitch.
    every_vassal = {
        limit = { domcom_needs_profile_init_trigger = yes }
        domcom_initialize_profile_effect = yes
    }
    every_vassal = {
        every_spouse = {
            limit = { domcom_needs_profile_init_trigger = yes }
            domcom_initialize_profile_effect = yes
        }
    }

    # Immediate liege and their spouses, so a vassal player's most visible
    # superiors are covered too.
    if = {
        limit = { exists = liege NOT = { liege = this } }
        liege = {
            domcom_try_initialize_profile_effect = yes
            every_spouse = {
                limit = { domcom_needs_profile_init_trigger = yes }
                domcom_initialize_profile_effect = yes
            }
        }
    }
}


############################################################################
# 0.7.7 — REBUILD TOOLS
#
# Almost every expensive thing in Dominion is gated on a run-once flag, which
# is what keeps it cheap at runtime. The cost is that the same flags block a
# changed version of the mod from ever re-applying to a save that already ran
# the old one, which is why testing appeared to need a new game every time.
#
# The gates in question:
#   domcom_world_endowments_seeded_070          global, blocks the world seed
#   domcom_ruler_endowments_initialized_061_r2  global, blocks the ruler pass
#   domcom_length_tenths / domcom_girth_tenths  per character, block the roll
#   domcom_carn_synced_<tier>                   legacy compatibility flags;
#                                               rc4.2 clears but no longer trusts them
############################################################################

# Non-destructive. Keeps every measurement, clears the derived state, and
# recomputes categories and traits from the numbers already in the save.
# This is the one to use after changing tier thresholds, trait display, icons
# or the Carnalitas mapping.
domcom_rebuild_display_effect = {
    every_living_character = {
        limit = { is_male = yes age >= 18 has_variable = domcom_length_tenths }
        remove_variable = domcom_endowment_category
        remove_variable = domcom_endowment_index
        remove_variable = domcom_locked_endowment_category
        domcom_rebuild_clear_sync_flags_effect = yes
        domcom_assign_endowment_effect = yes
    }
}

# Destructive. Wipes the measurements themselves so they are rolled again from
# scratch. Use after changing the random_list weights or the preseed mapping.
# Every character's size will change.
domcom_rebuild_measurements_effect = {
    remove_global_variable = domcom_world_endowments_seeded_070
    remove_global_variable = domcom_ruler_endowments_initialized_061_r2
    every_living_character = {
        limit = { is_male = yes age >= 18 }
        remove_variable = domcom_length_tenths
        remove_variable = domcom_girth_tenths
        remove_variable = domcom_locked_length_tenths
        remove_variable = domcom_locked_girth_tenths
        remove_variable = domcom_locked_endowment_category
        # 1.0.0-rc4.3 — the sire record is a copy of the father's figures taken
        # at birth, not a live reference, so leaving it in place would seed the
        # new roll from pre-rebuild numbers and quietly defeat "from scratch".
        # Cleared here; domcom_capture_endowment_heredity_effect re-takes it
        # from whatever the father ends up with, and heredity applies normally
        # to every birth after this point.
        remove_variable = domcom_sire_length_tenths
        remove_variable = domcom_sire_girth_tenths
        remove_character_flag = domcom_endowment_locked
        remove_variable = domcom_endowment_category
        remove_variable = domcom_endowment_index
        domcom_rebuild_clear_sync_flags_effect = yes
        domcom_clear_endowment_traits_effect = yes
        # 0.8.2 — must also drop traits owned by an integration, or
        # domcom_preseed_endowment_effect reads one straight back and seeds the
        # character from their old tier instead of rolling. With Carnalitas
        # loaded that turned "reroll everything" into "pin everyone to their
        # existing tier's midpoint", which looks like the button did nothing
        # while quietly destroying all variety within each tier.
        domcom_rebuild_clear_external_traits_effect = yes
    }
    # Ungated on purpose. See domcom_seed_world_endowments_once_effect.
    domcom_seed_world_endowments_now_effect = yes
    set_global_variable = domcom_ruler_endowments_initialized_061_r2
}

# 0.8.7 — repair for subjects stranded by the pre-0.8.7 seizure bug. Their
# collapse was recorded as resolved while they kept every title, and nothing in
# the mod could reopen it. This clears the terminal state for anyone still
# landed and still owned by the patron, so Collapse becomes available again.
domcom_repair_failed_collapses_effect = {
    save_scope_as = domcom_repair_patron
    every_living_character = {
        limit = {
            is_landed = yes
            OR = {
                has_character_flag = domcom_collapse_seizure_failed
                AND = {
                    has_character_flag = domcom_collapse_resolved
                    # 0.8.9 — was domcom_owner, which is only ever a saved scope
                    # and is never stored on a character. The condition could
                    # therefore never match, so the repair found nobody.
                    has_variable = domcom_patron
                    var:domcom_patron = scope:domcom_repair_patron
                }
            }
        }
        remove_character_flag = domcom_collapse_resolved
        remove_character_flag = domcom_collapse_seizure_failed
        if = {
            limit = { has_character_modifier = domcom_final_submission_modifier }
            remove_character_modifier = domcom_final_submission_modifier
        }
    }
}

############################################################################
# 0.9.8 — THE MONTHLY DIGEST
#
# Registration used to be a push model: every admirer that passed its check
# fired its own approach event, so working on two households in an afternoon
# produced a dozen popups, and Enforce Dominion existed largely to mop up
# afterwards. The player was acting as the scheduler.
#
# Now interest is recorded and reported once a month. Nobody is consumed by the
# digest: everyone named in it stays on Dynastic Prospects with their interest
# intact, so the digest is a notification rather than a queue that has to be
# drained before the next one arrives.
############################################################################

domcom_digest_note_admirer_effect = {
    if = {
        limit = { NOT = { has_variable = domcom_digest_pending } }
        set_variable = { name = domcom_digest_pending value = 0 }
    }
    change_variable = { name = domcom_digest_pending add = 1 }
}

domcom_monthly_digest_effect = {
    if = {
        limit = {
            is_ai = no
            has_variable = domcom_digest_pending
            var:domcom_digest_pending > 0
        }
        trigger_event = { id = domcom.910 }
    }
}



# 1.0.0-rc4 — complete a Crowned Union. Current scope is the Queen/Empress;
# scope:domcom_crowned_partner is the prospective spouse.
domcom_form_crowned_union_effect = {
    save_scope_as = domcom_crowned_queen
    remove_character_flag = domcom_crowned_union_completed

    if = {
        limit = {
            exists = primary_spouse
            primary_spouse = {
                is_alive = yes
                NOT = { this = scope:domcom_crowned_partner }
            }
        }
        primary_spouse = {
            save_scope_as = domcom_crowned_old_spouse
            divorce = scope:domcom_crowned_queen
        }
    }

    if = {
        limit = {
            can_marry_trigger = yes
            allowed_to_marry_character_gender_trigger = { CHARACTER = scope:domcom_crowned_partner }
            scope:domcom_crowned_partner = {
                can_marry_trigger = yes
                allowed_to_marry_character_gender_trigger = { CHARACTER = scope:domcom_crowned_queen }
            }
        }
        marry_on_realm_law = {
            SPOUSE_1 = scope:domcom_crowned_partner
            SPOUSE_2 = scope:domcom_crowned_queen
        }
        add_character_flag = domcom_crowned_union_completed
        add_opinion = { target = scope:domcom_crowned_partner modifier = domcom_crowned_union_devotion_opinion }

        if = {
            limit = { exists = scope:domcom_crowned_old_spouse }
            scope:domcom_crowned_old_spouse = {
                add_opinion = { target = scope:domcom_crowned_queen modifier = domcom_crowned_union_betrayed_opinion }
                add_opinion = { target = scope:domcom_crowned_partner modifier = domcom_crowned_union_betrayed_opinion }
            }
        }
    }
    else = {
        # State can change between an AI offer and the player's answer. If the
        # new marriage is no longer legal, restore the old primary marriage
        # where possible instead of leaving the Queen divorced by a stale event.
        if = {
            limit = {
                exists = scope:domcom_crowned_old_spouse
                scope:domcom_crowned_old_spouse = { is_alive = yes can_marry_trigger = yes }
                can_marry_trigger = yes
            }
            marry_on_realm_law = {
                SPOUSE_1 = scope:domcom_crowned_queen
                SPOUSE_2 = scope:domcom_crowned_old_spouse
            }
        }
    }
}

# AI Queen/Empress yearly check. It only considers human players, requires the
# custom five-county-border proximity gate, and needs strong compatibility plus
# an action-oriented personality. The random offer chance remains deliberately
# low so this feels exceptional rather than like inbox spam.
domcom_ai_try_crowned_heart_effect = {
    save_scope_as = domcom_crowned_queen

    every_player = {
        limit = {
            is_ai = no
            is_alive = yes
            age >= 18
            is_imprisoned = no
            can_marry_trigger = yes
            scope:domcom_crowned_queen = {
                NOT = { has_character_flag = domcom_crowned_offer_pending }
                NOT = { has_character_flag = domcom_crowned_heart_rejected }
                domcom_crowned_heart_actor_trigger = yes
                domcom_crowned_union_pair_trigger = { PARTNER = prev }
                is_attracted_to_gender_of = prev
            }
        }

        save_scope_as = domcom_crowned_partner
        scope:domcom_crowned_queen = {
            domcom_initialize_attraction_profile_effect = yes
            scope:domcom_crowned_partner = { save_scope_as = domcom_compatibility_target }
            set_variable = { name = domcom_last_crowned_compatibility value = domcom_pair_compatibility_value }

            if = {
                limit = {
                    OR = {
                        var:domcom_last_crowned_compatibility >= 75
                        AND = {
                            var:domcom_last_crowned_compatibility >= 50
                            OR = {
                                has_relation_crush = scope:domcom_crowned_partner
                                has_relation_lover = scope:domcom_crowned_partner
                                has_relation_soulmate = scope:domcom_crowned_partner
                                AND = {
                                    has_character_flag = domcom_pet
                                    has_variable = domcom_pet_patron
                                    var:domcom_pet_patron = scope:domcom_crowned_partner
                                }
                            }
                        }
                    }
                }
                random = {
                    chance = domcom_crowned_heart_offer_chance_value
                    add_character_flag = { flag = domcom_crowned_offer_pending years = 2 }
                    scope:domcom_crowned_partner = {
                        trigger_event = { id = domcom.920 days = 1 }
                    }
                }
            }
        }
    }
}

# ---------------------------------------------------------------------------
# 0.9.7 — DEEP SAVE SCAN (issue 4)
# ---------------------------------------------------------------------------
# Everything else in this file is deliberately bounded to the Patron's visible
# orbit for performance. That is correct for the quarterly pass and wrong for a
# save picked up mid-game, where the evidence the mod cares about (hidden
# paternity, dead partners, households compromised generations ago) sits outside
# any orbit. This walks the whole living world once, on explicit request.
#
# Cost is one cheap test per living character. It is a button, never a pulse.
domcom_deep_scan_effect = {
    save_scope_as = domcom_ui_patron
    save_scope_as = domcom_owner
    save_scope_as = domcom_scan_patron

    # --- 1. BLOOD -------------------------------------------------------
    # The authoritative question is biological, and it is asked of the child.
    # House membership is irrelevant: a child sired on my own dynasty is still
    # unacknowledged blood, which is exactly what the old house filter dropped.
    every_living_character = {
        limit = {
            domcom_is_unacknowledged_blood_trigger = yes
            NOT = { has_character_flag = domcom_blood_registered }
        }
        add_character_flag = domcom_blood_registered
        save_temporary_scope_as = domcom_scan_child
        scope:domcom_scan_patron = {
            add_to_variable_list = { name = domcom_blood_registry target = scope:domcom_scan_child }
        }
    }

    # --- 2. CUCK EVIDENCE ------------------------------------------------
    # A household is compromised if its married member is (or ever was recorded
    # as) my intimate, or if a child of that marriage is mine by blood. Both
    # tests are run from the spouse who was wronged, so no lover relation needs
    # to still exist for the evidence to be recoverable.
    every_living_character = {
        limit = {
            is_alive = yes
            age >= 18
            is_married = yes
            NOT = { this = scope:domcom_scan_patron }
            any_spouse = {
                is_alive = yes
                OR = {
                    has_relation_lover = scope:domcom_scan_patron
                    has_relation_soulmate = scope:domcom_scan_patron
                    AND = {
                        has_variable = domcom_obedience_patron
                        var:domcom_obedience_patron = scope:domcom_scan_patron
                        has_variable = domcom_obedience
                        var:domcom_obedience > 0
                    }
                    any_child = { real_father ?= scope:domcom_scan_patron }
                }
            }
        }
        domcom_register_cuck_generic_effect = yes
    }

    # --- 3. TANTALIZED ---------------------------------------------------
    # Obedience was already persisted per character; it simply was never
    # collected anywhere the Patron could read it.
    every_living_character = {
        limit = {
            is_alive = yes
            age >= 18
            has_variable = domcom_obedience_patron
            var:domcom_obedience_patron = scope:domcom_scan_patron
            has_variable = domcom_obedience
            var:domcom_obedience > 0
            NOT = { has_character_flag = domcom_tantalized_registered }
        }
        add_character_flag = domcom_tantalized_registered
        save_temporary_scope_as = domcom_scan_tantalized
        scope:domcom_scan_patron = {
            add_to_variable_list = { name = domcom_tantalized_registry target = scope:domcom_scan_tantalized }
        }
    }

    # --- 4. OWNED --------------------------------------------------------
    # Repair the Patron-local Owned registry from world state, so a save whose
    # dynasty storage was lost or never written still reports correctly.
    every_living_character = {
        limit = {
            is_alive = yes
            has_character_flag = domcom_owned_cuck
            has_variable = domcom_patron
            var:domcom_patron = scope:domcom_scan_patron
        }
        save_temporary_scope_as = domcom_scan_owned
        scope:domcom_scan_patron = {
            if = {
                limit = {
                    OR = {
                        NOT = { has_variable_list = domcom_owned_registry }
                        NOT = { any_in_list = { variable = domcom_owned_registry this = scope:domcom_scan_owned } }
                    }
                }
                add_to_variable_list = { name = domcom_owned_registry target = scope:domcom_scan_owned }
            }
        }
    }
}
