
# Endowment has three permanent pieces of state: length, girth, and tier. The tier is stored separately
# from the derived working category so ordinary profile refreshes cannot reclassify a character during
# Alpha Check/Tantalize.
domcom_restore_locked_endowment_category_effect = {
    if = { limit = { var:domcom_locked_endowment_category = flag:micro } set_variable = { name = domcom_endowment_category value = flag:micro } }
    else_if = { limit = { var:domcom_locked_endowment_category = flag:small } set_variable = { name = domcom_endowment_category value = flag:small } }
    else_if = { limit = { var:domcom_locked_endowment_category = flag:average } set_variable = { name = domcom_endowment_category value = flag:average } }
    else_if = { limit = { var:domcom_locked_endowment_category = flag:big } set_variable = { name = domcom_endowment_category value = flag:big } }
    else_if = { limit = { var:domcom_locked_endowment_category = flag:hung } set_variable = { name = domcom_endowment_category value = flag:hung } }
    else_if = { limit = { var:domcom_locked_endowment_category = flag:massive } set_variable = { name = domcom_endowment_category value = flag:massive } }
}

domcom_lock_current_endowment_category_effect = {
    if = { limit = { var:domcom_endowment_category = flag:micro } set_variable = { name = domcom_locked_endowment_category value = flag:micro } }
    else_if = { limit = { var:domcom_endowment_category = flag:small } set_variable = { name = domcom_locked_endowment_category value = flag:small } }
    else_if = { limit = { var:domcom_endowment_category = flag:average } set_variable = { name = domcom_locked_endowment_category value = flag:average } }
    else_if = { limit = { var:domcom_endowment_category = flag:big } set_variable = { name = domcom_locked_endowment_category value = flag:big } }
    else_if = { limit = { var:domcom_endowment_category = flag:hung } set_variable = { name = domcom_locked_endowment_category value = flag:hung } }
    else_if = { limit = { var:domcom_endowment_category = flag:massive } set_variable = { name = domcom_locked_endowment_category value = flag:massive } }
}

domcom_assign_endowment_effect = {
    if = {
        limit = {
            is_male = yes
            age >= 18
            has_variable = domcom_locked_length_tenths
            has_variable = domcom_locked_girth_tenths
        }
        set_variable = { name = domcom_length_tenths value = var:domcom_locked_length_tenths }
        set_variable = { name = domcom_girth_tenths value = var:domcom_locked_girth_tenths }
    }
    if = {
        limit = { is_male = yes age >= 18 has_variable = domcom_locked_length_tenths NOT = { has_variable = domcom_locked_girth_tenths } }
        set_variable = { name = domcom_length_tenths value = var:domcom_locked_length_tenths }
    }
    if = {
        limit = { is_male = yes age >= 18 has_variable = domcom_locked_girth_tenths NOT = { has_variable = domcom_locked_length_tenths } }
        set_variable = { name = domcom_girth_tenths value = var:domcom_locked_girth_tenths }
    }

    # Give an already-present compatibility source one chance to seed a brand new profile. This never
    # runs once Dominion has a locked measurement pair.
    if = {
        limit = {
            is_male = yes
            age >= 18
            NOT = { has_variable = domcom_locked_length_tenths }
            NOT = { has_variable = domcom_locked_girth_tenths }
            NOT = { has_variable = domcom_length_tenths }
            NOT = { has_variable = domcom_girth_tenths }
        }
        domcom_preseed_endowment_effect = yes
    }

    domcom_capture_endowment_heredity_effect = yes

    # Sired roll. Regression to the mean is 40% per generation, so a dynasty can drift toward an extreme
    # over several generations but does not lock itself there, and no single birth is a copy of the
    # father.
    if = {
        limit = {
            is_male = yes
            age >= 18
            NOT = { has_variable = domcom_length_tenths }
            has_variable = domcom_sire_length_tenths
        }
        set_variable = { name = domcom_length_tenths value = domcom_sire_length_seed_value }
        random_list = {
            6  = { change_variable = { name = domcom_length_tenths add = -20 } }
            10 = { change_variable = { name = domcom_length_tenths add = -15 } }
            15 = { change_variable = { name = domcom_length_tenths add = -10 } }
            19 = { change_variable = { name = domcom_length_tenths add = -5 } }
            20 = { change_variable = { name = domcom_length_tenths add = 0 } }
            19 = { change_variable = { name = domcom_length_tenths add = 5 } }
            15 = { change_variable = { name = domcom_length_tenths add = 10 } }
            10 = { change_variable = { name = domcom_length_tenths add = 15 } }
            6  = { change_variable = { name = domcom_length_tenths add = 20 } }
        }
        if = { limit = { var:domcom_length_tenths < 20 } set_variable = { name = domcom_length_tenths value = 20 } }
        if = { limit = { var:domcom_length_tenths > 130 } set_variable = { name = domcom_length_tenths value = 130 } }
    }
    if = {
        limit = {
            is_male = yes
            age >= 18
            NOT = { has_variable = domcom_girth_tenths }
            has_variable = domcom_sire_girth_tenths
        }
        set_variable = { name = domcom_girth_tenths value = domcom_sire_girth_seed_value }
        random_list = {
            6  = { change_variable = { name = domcom_girth_tenths add = -15 } }
            12 = { change_variable = { name = domcom_girth_tenths add = -10 } }
            20 = { change_variable = { name = domcom_girth_tenths add = -5 } }
            24 = { change_variable = { name = domcom_girth_tenths add = 0 } }
            20 = { change_variable = { name = domcom_girth_tenths add = 5 } }
            12 = { change_variable = { name = domcom_girth_tenths add = 10 } }
            6  = { change_variable = { name = domcom_girth_tenths add = 15 } }
        }
        if = { limit = { var:domcom_girth_tenths < 20 } set_variable = { name = domcom_girth_tenths value = 20 } }
        if = { limit = { var:domcom_girth_tenths > 85 } set_variable = { name = domcom_girth_tenths value = 85 } }
    }

    if = {
        limit = { is_male = yes age >= 18 NOT = { has_variable = domcom_length_tenths } }
        random_list = {
            2 = { set_variable = { name = domcom_length_tenths value = 20 } }
            6 = { set_variable = { name = domcom_length_tenths value = 25 } }
            22 = { set_variable = { name = domcom_length_tenths value = 30 } }
            70 = { set_variable = { name = domcom_length_tenths value = 35 } }
            170 = { set_variable = { name = domcom_length_tenths value = 40 } }
            500 = { set_variable = { name = domcom_length_tenths value = 45 } }
            950 = { set_variable = { name = domcom_length_tenths value = 50 } }
            1150 = { set_variable = { name = domcom_length_tenths value = 55 } }
            800 = { set_variable = { name = domcom_length_tenths value = 60 } }
            520 = { set_variable = { name = domcom_length_tenths value = 65 } }
            340 = { set_variable = { name = domcom_length_tenths value = 70 } }
            210 = { set_variable = { name = domcom_length_tenths value = 75 } }
            130 = { set_variable = { name = domcom_length_tenths value = 80 } }
            85 = { set_variable = { name = domcom_length_tenths value = 85 } }
            55 = { set_variable = { name = domcom_length_tenths value = 90 } }
            35 = { set_variable = { name = domcom_length_tenths value = 95 } }
            22 = { set_variable = { name = domcom_length_tenths value = 100 } }
            12 = { set_variable = { name = domcom_length_tenths value = 110 } }
            6 = { set_variable = { name = domcom_length_tenths value = 120 } }
            3 = { set_variable = { name = domcom_length_tenths value = 130 } }
        }
    }
    if = {
        limit = { is_male = yes age >= 18 NOT = { has_variable = domcom_girth_tenths } }
        random_list = {
            2 = { set_variable = { name = domcom_girth_tenths value = 20 } }
            6 = { set_variable = { name = domcom_girth_tenths value = 25 } }
            28 = { set_variable = { name = domcom_girth_tenths value = 30 } }
            90 = { set_variable = { name = domcom_girth_tenths value = 35 } }
            340 = { set_variable = { name = domcom_girth_tenths value = 40 } }
            780 = { set_variable = { name = domcom_girth_tenths value = 45 } }
            1000 = { set_variable = { name = domcom_girth_tenths value = 50 } }
            780 = { set_variable = { name = domcom_girth_tenths value = 55 } }
            460 = { set_variable = { name = domcom_girth_tenths value = 60 } }
            250 = { set_variable = { name = domcom_girth_tenths value = 65 } }
            130 = { set_variable = { name = domcom_girth_tenths value = 70 } }
            60 = { set_variable = { name = domcom_girth_tenths value = 75 } }
            26 = { set_variable = { name = domcom_girth_tenths value = 80 } }
            10 = { set_variable = { name = domcom_girth_tenths value = 85 } }
        }
    }

    # Adopt the first complete pair as the permanent profile. This also migrates existing saves without
    # changing their current measurements.
    if = {
        limit = {
            is_male = yes
            age >= 18
            has_variable = domcom_length_tenths
            has_variable = domcom_girth_tenths
            OR = {
                NOT = { has_variable = domcom_locked_length_tenths }
                NOT = { has_variable = domcom_locked_girth_tenths }
            }
        }
        if = { limit = { NOT = { has_variable = domcom_locked_length_tenths } } set_variable = { name = domcom_locked_length_tenths value = var:domcom_length_tenths } }
        if = { limit = { NOT = { has_variable = domcom_locked_girth_tenths } } set_variable = { name = domcom_locked_girth_tenths value = var:domcom_girth_tenths } }
        add_character_flag = domcom_endowment_locked
    }

    # Restore the permanent measurements before display work. The tier itself is also permanent once
    # established. Ordinary profile calls therefore reassert state; they never classify the character
    # again.
    if = {
        limit = {
            is_male = yes age >= 18
            has_variable = domcom_locked_length_tenths
            has_variable = domcom_locked_girth_tenths
        }
        set_variable = { name = domcom_length_tenths value = var:domcom_locked_length_tenths }
        set_variable = { name = domcom_girth_tenths value = var:domcom_locked_girth_tenths }

        if = {
            limit = { has_variable = domcom_locked_endowment_category }
            domcom_restore_locked_endowment_category_effect = yes
            if = { limit = { NOT = { has_variable = domcom_endowment_index } } set_variable = { name = domcom_endowment_index value = domcom_endowment_index_value } }
            domcom_apply_native_endowment_trait_effect = yes
            domcom_post_category_effect = yes
        }
        else = {
            # First initialization (or an old save): classify once from the now permanent measurements,
            # lock that tier, and never recalculate it during normal gameplay again.
            domcom_sync_endowment_trait_effect = yes
        }
    }
}

# Converts the hidden measurements into one visible physical trait. This is run even for old saves so the
# former character-modifier display migrates cleanly.
domcom_sync_endowment_trait_effect = {
    if = {
        limit = { is_male = yes age >= 18 has_variable = domcom_length_tenths has_variable = domcom_girth_tenths }

        if = { limit = { has_character_modifier = domcom_endowment_modest_modifier } remove_character_modifier = domcom_endowment_modest_modifier }
        if = { limit = { has_character_modifier = domcom_endowment_average_modifier } remove_character_modifier = domcom_endowment_average_modifier }
        if = { limit = { has_character_modifier = domcom_endowment_noteworthy_modifier } remove_character_modifier = domcom_endowment_noteworthy_modifier }
        if = { limit = { has_character_modifier = domcom_endowment_enviable_modifier } remove_character_modifier = domcom_endowment_enviable_modifier }
        if = { limit = { has_character_modifier = domcom_endowment_legendary_modifier } remove_character_modifier = domcom_endowment_legendary_modifier }

        set_variable = { name = domcom_endowment_index value = domcom_endowment_index_value }

        if = { limit = { var:domcom_length_tenths <= 40 } set_variable = { name = domcom_endowment_category value = flag:micro } }
        else_if = { limit = { var:domcom_endowment_index < 52 } set_variable = { name = domcom_endowment_category value = flag:small } }
        else_if = { limit = { var:domcom_endowment_index < 57 } set_variable = { name = domcom_endowment_category value = flag:average } }
        else_if = { limit = { var:domcom_endowment_index < 63 } set_variable = { name = domcom_endowment_category value = flag:big } }
        else_if = { limit = { var:domcom_endowment_index < 70 } set_variable = { name = domcom_endowment_category value = flag:hung } }
        else = { set_variable = { name = domcom_endowment_category value = flag:massive } }

        # Classification is a one-time operation. Save the result separately so later Alpha Checks,
        # Tantalize calls and profile refreshes cannot move a character between tiers. Rebuild/Reroll
        # explicitly clear this lock.
        if = {
            limit = { NOT = { has_variable = domcom_locked_endowment_category } }
            domcom_lock_current_endowment_category_effect = yes
        }

        domcom_apply_native_endowment_trait_effect = yes

        domcom_post_category_effect = yes
    }
}

# Reads a Dominion Endowment trait the character is already carrying and seeds from it. Used by the
# standalone preseed and, as a fallback, by the Carnalitas one. Highest tier wins if a designer somehow
# produced more than one.
domcom_seed_from_own_trait_effect = {
    if = { limit = { has_trait = domcom_endowment_massive } domcom_seed_tier_massive_effect = yes }
    else_if = { limit = { has_trait = domcom_endowment_hung } domcom_seed_tier_hung_effect = yes }
    else_if = { limit = { has_trait = domcom_endowment_big } domcom_seed_tier_big_effect = yes }
    else_if = { limit = { has_trait = domcom_endowment_average } domcom_seed_tier_average_effect = yes }
    else_if = { limit = { has_trait = domcom_endowment_small } domcom_seed_tier_small_effect = yes }
    else_if = { limit = { has_trait = domcom_endowment_micro } domcom_seed_tier_micro_effect = yes }
}

# The sanity pass. Cheap, idempotent, and safe to run on anybody.
domcom_refresh_endowment_expression_effect = {
    if = {
        limit = {
            domcom_should_express_endowment_trigger = yes
            is_adult = yes
        }
        domcom_activate_endowment_traits_effect = yes
    }
    else = {
        domcom_deactivate_endowment_traits_effect = yes
    }
    domcom_carn_suppress_dt_traits_effect = yes
}

domcom_refresh_endowment_expression_once_effect = {
    if = {
        limit = { NOT = { has_global_variable = domcom_endowment_expression_pass_101 } }
        every_living_character = {
            limit = {
                OR = {
                    has_trait = domcom_endowment_micro
                    has_trait = domcom_endowment_small
                    has_trait = domcom_endowment_average
                    has_trait = domcom_endowment_big
                    has_trait = domcom_endowment_hung
                    has_trait = domcom_endowment_massive
                }
                NOT = { domcom_should_express_endowment_trigger = yes }
            }
            domcom_deactivate_endowment_traits_effect = yes
        }
        set_global_variable = domcom_endowment_expression_pass_101
    }
}

domcom_clear_endowment_traits_effect = {
    if = { limit = { has_trait = domcom_endowment_micro } remove_trait = domcom_endowment_micro }
    if = { limit = { has_trait = domcom_endowment_small } remove_trait = domcom_endowment_small }
    if = { limit = { has_trait = domcom_endowment_average } remove_trait = domcom_endowment_average }
    if = { limit = { has_trait = domcom_endowment_big } remove_trait = domcom_endowment_big }
    if = { limit = { has_trait = domcom_endowment_hung } remove_trait = domcom_endowment_hung }
    if = { limit = { has_trait = domcom_endowment_massive } remove_trait = domcom_endowment_massive }
}

domcom_capture_endowment_heredity_effect = {
    if = {
        limit = {
            is_male = yes
            NOT = { has_variable = domcom_sire_length_tenths }
        }
        if = {
            limit = {
                exists = real_father
                real_father = { has_variable = domcom_locked_length_tenths }
            }
            real_father = { save_temporary_scope_as = domcom_heredity_sire }
        }
        else_if = {
            limit = {
                exists = father
                father = { has_variable = domcom_locked_length_tenths }
            }
            father = { save_temporary_scope_as = domcom_heredity_sire }
        }
        if = {
            limit = { exists = scope:domcom_heredity_sire }
            set_variable = {
                name = domcom_sire_length_tenths
                value = scope:domcom_heredity_sire.var:domcom_locked_length_tenths
            }
            if = {
                limit = { scope:domcom_heredity_sire = { has_variable = domcom_locked_girth_tenths } }
                set_variable = {
                    name = domcom_sire_girth_tenths
                    value = scope:domcom_heredity_sire.var:domcom_locked_girth_tenths
                }
            }
        }
    }
}

domcom_assign_size_preference_effect = {
    if = { limit = { has_variable = domcom_size_preference } remove_variable = domcom_size_preference }

    if = {
        limit = {
            age >= 18
            is_attracted_to_men = yes

# Run from a character who just reached 1000 Obedience, with scope:domcom_tantalize_patron saved. Pet
# numbering is stored on the Patron's dynasty so it survives succession and counts across the dynasty
# rather than one ruler.
domcom_create_pet_effect = {
    save_scope_as = domcom_pet_subject
    add_character_flag = domcom_pet
    add_character_flag = domcom_new_pet_created
    set_variable = { name = domcom_pet_patron value = scope:domcom_tantalize_patron }
    set_variable = { name = domcom_obedience_patron value = scope:domcom_tantalize_patron }
    set_variable = { name = domcom_obedience value = 1000 }
    add_character_modifier = domcom_pet_modifier
    add_opinion = { target = scope:domcom_tantalize_patron modifier = domcom_pet_obedience_opinion }

    if = {
        limit = { exists = scope:domcom_tantalize_patron.dynasty }
        scope:domcom_tantalize_patron.dynasty = {
            if = {
                limit = { NOT = { has_variable = domcom_pet_counter } }
                set_variable = { name = domcom_pet_counter value = 0 }
            }
            change_variable = { name = domcom_pet_counter add = 1 }
            save_scope_as = domcom_pet_dynasty_scope
        }
        set_variable = { name = domcom_pet_dynasty value = scope:domcom_pet_dynasty_scope }
        set_variable = { name = domcom_pet_number value = scope:domcom_pet_dynasty_scope.var:domcom_pet_counter }
        give_nickname = nick_domcom_dynasty_slave
    }
    else = {
        scope:domcom_tantalize_patron = {
            if = {
                limit = { NOT = { has_variable = domcom_personal_pet_counter } }
                set_variable = { name = domcom_personal_pet_counter value = 0 }
            }
            change_variable = { name = domcom_personal_pet_counter add = 1 }
        }
        set_variable = { name = domcom_pet_number value = scope:domcom_tantalize_patron.var:domcom_personal_pet_counter }
        give_nickname = nick_domcom_personal_slave
    }

    scope:domcom_tantalize_patron = {
        add_to_variable_list = { name = domcom_pet_registry target = scope:domcom_pet_subject }
        if = {
            limit = { can_add_hook = { target = scope:domcom_pet_subject type = loyalty_hook } }
            add_hook_no_toast = { target = scope:domcom_pet_subject type = loyalty_hook }
        }
        save_scope_as = domcom_owner
    }

    # Full obedience finally grants the same kind of household access that an Outclassed rival can
    # expose, but Pet itself remains distinct from Cucked/Owned.
    domcom_schedule_family_attraction_checks_effect = yes
    domcom_refresh_obedience_stage_effect = yes
    trigger_event = { on_action = domcom_pet_created }
}

# Run from the Tantalize recipient with scope:domcom_tantalize_patron saved. A base success now scales
# with compatibility; established intimate bonds accelerate the track, and a prior same-sex Alpha victory
# against this target adds extra leverage.
domcom_apply_tantalize_success_effect = {
    if = {
        limit = { NOT = { has_variable = domcom_obedience_patron } }
        set_variable = { name = domcom_obedience_patron value = scope:domcom_tantalize_patron }
    }
    if = { limit = { NOT = { has_variable = domcom_obedience } } set_variable = { name = domcom_obedience value = 0 } }

    if = {
        limit = { NOT = { has_character_flag = domcom_tantalized_registered } }
        add_character_flag = domcom_tantalized_registered
        save_temporary_scope_as = domcom_tantalize_subject
        scope:domcom_tantalize_patron = {
            add_to_variable_list = { name = domcom_tantalized_registry target = scope:domcom_tantalize_subject }
        }
    }

    change_variable = { name = domcom_obedience add = domcom_tantalize_obedience_gain_value }
    if = { limit = { has_relation_lover = scope:domcom_tantalize_patron } change_variable = { name = domcom_obedience add = 50 } }
    if = { limit = { has_relation_soulmate = scope:domcom_tantalize_patron } change_variable = { name = domcom_obedience add = 100 } }
    if = {
        limit = {
            has_character_flag = domcom_dominated
            has_variable = domcom_dominance_patron
            var:domcom_dominance_patron = scope:domcom_tantalize_patron
        }
        change_variable = { name = domcom_obedience add = 25 }
    }

    # A first successful Tantalize creates interest, not an affair. Vanilla Seduce/Romance or later
    # Dominion mechanics can still determine intimacy.
    if = {
        limit = {
            NOR = {
                has_relation_crush = scope:domcom_tantalize_patron
                has_relation_lover = scope:domcom_tantalize_patron
                has_relation_soulmate = scope:domcom_tantalize_patron
            }
        }
        set_relation_crush = scope:domcom_tantalize_patron
    }

    if = { limit = { var:domcom_obedience > 1000 } set_variable = { name = domcom_obedience value = 1000 } }
    domcom_refresh_obedience_stage_effect = yes

    if = {
        limit = {
            var:domcom_obedience >= 1000
            NOT = { has_character_flag = domcom_pet }
        }
        domcom_create_pet_effect = yes
    }
}

# If a lower-status challenger reverses the expected hierarchy, the victory itself becomes a source of
# standing.
domcom_apply_reverse_alpha_reward_effect = {
    # Any victory upward in the hierarchy creates relationship-scheme momentum.
    add_prestige = 300
    add_character_modifier = { modifier = domcom_reverse_alpha_victory_modifier years = 10 }

    # Major upset: roughly two or more social tiers above the challenger.
    if = {
        limit = {
            OR = {
                AND = { scope:recipient = { highest_held_title_tier = tier_duchy } is_landed = no }
                AND = { scope:recipient = { highest_held_title_tier = tier_kingdom } OR = { is_landed = no highest_held_title_tier = tier_county } }
                AND = { scope:recipient = { highest_held_title_tier = tier_empire } OR = { is_landed = no highest_held_title_tier = tier_county highest_held_title_tier = tier_duchy } }
            }
        }
        add_prestige = 300
        add_character_modifier = { modifier = domcom_reverse_alpha_major_upset_modifier years = 10 }
    }

    # Legendary upset: a landless challenger topples a King/Emperor, or a Count overcomes an Emperor.
    # This stacks with the base and Major Upset rewards.
    if = {
        limit = {
            OR = {
                AND = { scope:recipient = { highest_held_title_tier = tier_kingdom } is_landed = no }
                AND = { scope:recipient = { highest_held_title_tier = tier_empire } OR = { is_landed = no highest_held_title_tier = tier_county } }
            }
        }
        add_prestige = 400
        add_character_modifier = { modifier = domcom_reverse_alpha_legendary_upset_modifier years = 10 }
    }

    change_variable = { name = domcom_reverse_alpha_victories add = 1 }
}

# Fast upkeep from the patron-local registry. This prevents vanilla/random nickname systems from
# permanently replacing the Dominion reputation without performing any world scan.
domcom_reassert_cuck_reputations_for_patron_effect = {
    save_scope_as = domcom_reputation_patron
    if = {
        limit = { has_variable_list = domcom_cucked_registry }
        every_in_list = {
            variable = domcom_cucked_registry
            limit = {
                is_alive = yes
                has_character_flag = domcom_cucked
                OR = {
                    AND = { has_variable = domcom_cuck_patron var:domcom_cuck_patron = scope:domcom_reputation_patron }
                    AND = { has_variable = domcom_patron var:domcom_patron = scope:domcom_reputation_patron }
                }
            }
            domcom_refresh_cuck_reputation_effect = yes
        }
    }
}
domcom_refresh_cuck_stage_effect = {
    # Cuck XP is a bounded 0–1000 progression.
    if = {
        limit = { has_variable = domcom_cuck_xp var:domcom_cuck_xp > 1000 }
        set_variable = { name = domcom_cuck_xp value = 1000 }
    }
    if = {
        limit = { has_variable = domcom_cuck_xp var:domcom_cuck_xp < 0 }
        set_variable = { name = domcom_cuck_xp value = 0 }
    }

    # Transition-only stage maintenance avoids losing and immediately regaining the same modifier every
    # monthly repair pulse.
    if = {
        limit = { has_variable = domcom_cuck_xp var:domcom_cuck_xp >= 750 }
        if = { limit = { has_character_modifier = domcom_humiliated_stage_modifier } remove_character_modifier = domcom_humiliated_stage_modifier }
        if = { limit = { has_character_modifier = domcom_submissive_stage_modifier } remove_character_modifier = domcom_submissive_stage_modifier }
        if = { limit = { has_character_modifier = domcom_broken_stage_modifier } remove_character_modifier = domcom_broken_stage_modifier }
        if = { limit = { NOT = { has_character_modifier = domcom_abject_stage_modifier } } add_character_modifier = domcom_abject_stage_modifier }
    }
    else_if = {
        limit = { has_variable = domcom_cuck_xp var:domcom_cuck_xp >= 500 }
        if = { limit = { has_character_modifier = domcom_humiliated_stage_modifier } remove_character_modifier = domcom_humiliated_stage_modifier }
        if = { limit = { has_character_modifier = domcom_submissive_stage_modifier } remove_character_modifier = domcom_submissive_stage_modifier }
        if = { limit = { has_character_modifier = domcom_abject_stage_modifier } remove_character_modifier = domcom_abject_stage_modifier }
        if = { limit = { NOT = { has_character_modifier = domcom_broken_stage_modifier } } add_character_modifier = domcom_broken_stage_modifier }
    }
    else_if = {
        limit = { has_variable = domcom_cuck_xp var:domcom_cuck_xp >= 250 }
        if = { limit = { has_character_modifier = domcom_humiliated_stage_modifier } remove_character_modifier = domcom_humiliated_stage_modifier }
        if = { limit = { has_character_modifier = domcom_broken_stage_modifier } remove_character_modifier = domcom_broken_stage_modifier }
        if = { limit = { has_character_modifier = domcom_abject_stage_modifier } remove_character_modifier = domcom_abject_stage_modifier }
        if = { limit = { NOT = { has_character_modifier = domcom_submissive_stage_modifier } } add_character_modifier = domcom_submissive_stage_modifier }
    }
    else = {
        if = { limit = { has_character_modifier = domcom_submissive_stage_modifier } remove_character_modifier = domcom_submissive_stage_modifier }
        if = { limit = { has_character_modifier = domcom_broken_stage_modifier } remove_character_modifier = domcom_broken_stage_modifier }
        if = { limit = { has_character_modifier = domcom_abject_stage_modifier } remove_character_modifier = domcom_abject_stage_modifier }
        if = { limit = { NOT = { has_character_modifier = domcom_humiliated_stage_modifier } } add_character_modifier = domcom_humiliated_stage_modifier }
    }
}

# Register a successful post-Alpha attraction check. Run from candidate with
# scope:domcom_attraction_patron saved.
domcom_register_communion_admirer_effect = {
    save_scope_as = domcom_admirer_candidate
    if = {
        limit = {
            is_alive = yes
            age >= 18
            NOT = { this = scope:domcom_attraction_patron }
            NOT = { is_close_or_extended_family_of = scope:domcom_attraction_patron }
            scope:domcom_attraction_patron = { is_attracted_to_gender_of = prev }
            is_attracted_to_gender_of = scope:domcom_attraction_patron
        }
        if = {
            limit = {
                NOR = {
                    has_relation_crush = scope:domcom_attraction_patron
                    has_relation_lover = scope:domcom_attraction_patron
                    has_relation_soulmate = scope:domcom_attraction_patron
                }
            }
            set_relation_crush = scope:domcom_attraction_patron
        }
        set_variable = { name = domcom_crush_patron value = scope:domcom_attraction_patron }

        # Preserve the exact controlled relative whose post-Alpha check produced this admirer. Later
        # family humiliation must match this source; merely being related to some other Owned character
        # is not enough.
        if = {
            limit = { exists = scope:domcom_attraction_source }
            set_variable = { name = domcom_crush_source value = scope:domcom_attraction_source }
            scope:domcom_attraction_source = {
                add_to_variable_list = {
                    name = domcom_family_admirers
                    target = scope:domcom_admirer_candidate
                }
            }
        }

        add_character_flag = domcom_communion_admirer_registered
        remove_character_flag = domcom_communion_attraction_pending
        scope:domcom_attraction_patron = {
            add_to_variable_list = { name = domcom_crush_candidates target = scope:domcom_admirer_candidate }
            domcom_digest_note_admirer_effect = yes
        }
    }
}

# Transfer a broken ruler's realm to the patron. scope:domcom_collapse_patron saved.
domcom_ai_resolve_collapse_effect = {
    save_scope_as = domcom_surrendering_subject
    domcom_surrender_prestige_if_lower_effect = yes
    if = { limit = { gold > 0 } pay_short_term_gold = { target = scope:domcom_collapse_patron gold = gold } }
    if = {
        limit = { is_landed = yes }
        create_title_and_vassal_change = {
            type = granted
            save_scope_as = domcom_surrender_change
            add_claim_on_loss = yes
        }
        every_held_title = {
            limit = {
                NOT = { tier = tier_barony }
                NOT = { is_landless_type_title = yes }
            }
            change_title_holder = {
                holder = scope:domcom_collapse_patron
                change = scope:domcom_surrender_change
                government_base = scope:domcom_collapse_patron
            }
        }
        resolve_title_and_vassal_change = scope:domcom_surrender_change
    }

    if = {
        limit = { is_landed = yes }
        every_held_title = {
            limit = {
                tier >= tier_duchy
                NOT = { is_landless_type_title = yes }
            }
            add_to_temporary_list = domcom_stuck_titles
        }
        every_in_list = {
            list = domcom_stuck_titles
            save_temporary_scope_as = domcom_stuck_title
            destroy_title = scope:domcom_stuck_title
        }
    }

    # Second pass. Counties orphaned by the destruction above are now free to be granted, so the patron
    # still gains the land even though the religious title itself is gone.
    if = {
        limit = { is_landed = yes }
        create_title_and_vassal_change = {
            type = granted
            save_scope_as = domcom_surrender_change_freed
            add_claim_on_loss = yes
        }
        every_held_title = {
            limit = {
                NOT = { tier = tier_barony }
                NOT = { is_landless_type_title = yes }
            }
            change_title_holder = {
                holder = scope:domcom_collapse_patron
                change = scope:domcom_surrender_change_freed
                government_base = scope:domcom_collapse_patron
            }
        }
        resolve_title_and_vassal_change = scope:domcom_surrender_change_freed
    }

    if = {
        limit = {
            is_landed = no
            exists = scope:domcom_collapse_patron
            NOT = { is_courtier_of = scope:domcom_collapse_patron }
        }
        set_employer = scope:domcom_collapse_patron
    }

    remove_character_flag = domcom_collapse_pending

    if = {
        limit = { is_landed = yes }
        add_character_flag = domcom_collapse_seizure_failed
        if = {
            limit = { has_character_modifier = domcom_final_submission_modifier }
            remove_character_modifier = domcom_final_submission_modifier
        }
    }
    else = {
        add_character_flag = domcom_collapse_resolved
        add_character_modifier = domcom_final_submission_modifier
        scope:domcom_collapse_patron = { change_variable = { name = domcom_collapses_resolved add = 1 } }
    }
}

domcom_check_final_collapse_effect = {
    if = {
        limit = {
            has_character_flag = domcom_owned_cuck
            has_variable = domcom_cuck_xp
            var:domcom_cuck_xp >= 1000
            NOT = { has_character_flag = domcom_collapse_pending }
            NOT = { has_character_flag = domcom_collapse_resolved }
            has_variable = domcom_patron
            var:domcom_patron = { is_alive = yes }
        }
        save_scope_as = domcom_collapsing_subject
        var:domcom_patron = { save_scope_as = domcom_collapse_patron }

        # A fully collapsed Household Dominion resolves as a household once.
        if = {
            limit = {
                is_married = yes
                any_spouse = {
                    is_alive = yes
                    has_character_flag = domcom_owned_cuck
                    has_variable = domcom_patron
                    var:domcom_patron = scope:domcom_collapse_patron
                    has_variable = domcom_cuck_xp
                    var:domcom_cuck_xp >= 1000
                    NOT = { has_character_flag = domcom_collapse_pending }
                    NOT = { has_character_flag = domcom_collapse_resolved }
                }
            }
            every_spouse = {
                limit = {
                    is_alive = yes
                    has_character_flag = domcom_owned_cuck
                    has_variable = domcom_patron
                    var:domcom_patron = scope:domcom_collapse_patron
                    has_variable = domcom_cuck_xp
                    var:domcom_cuck_xp >= 1000
                    NOT = { has_character_flag = domcom_collapse_pending }
                    NOT = { has_character_flag = domcom_collapse_resolved }
                }
                save_scope_as = domcom_collapsing_spouse
            }
            add_character_flag = domcom_collapse_pending
            scope:domcom_collapsing_spouse = { add_character_flag = domcom_collapse_pending }
            if = {
                limit = { scope:domcom_collapse_patron = { is_ai = no } }
                scope:domcom_collapse_patron = { trigger_event = { id = domcom.105 days = 1 } }
            }
            else = { domcom_resolve_household_collapse_effect = yes }
        }

        # A Patron's own intimate partner ends in permanent devotion, not destruction.
        else_if = {
            limit = {
                OR = {
                    is_consort_of = scope:domcom_collapse_patron
                    has_relation_lover = scope:domcom_collapse_patron
                    has_relation_soulmate = scope:domcom_collapse_patron
                }
            }
            add_character_flag = domcom_collapse_pending
            if = {
                limit = { scope:domcom_collapse_patron = { is_ai = no } }
                scope:domcom_collapse_patron = { trigger_event = { id = domcom.101 days = 1 } }
            }
            else = { domcom_resolve_devotion_collapse_effect = yes }
        }

        else = {
            add_character_flag = domcom_collapse_pending
            if = {
                limit = { scope:domcom_collapse_patron = { is_ai = no } }
                scope:domcom_collapse_patron = { trigger_event = { id = domcom.100 days = 1 } }
            }
            else = { domcom_ai_resolve_collapse_effect = yes }
        }
    }
}

        add_character_flag = domcom_blood_registered

            add_to_variable_list = { name = domcom_ui_blood_raw target = scope:domcom_ui_blood_child }
            change_variable = { name = domcom_ui_blood_count add = 1 }
        }
    }
    every_relation = {
        type = lover
        type = soulmate
        limit = { is_alive = yes }
        every_child = {
            limit = {
                is_alive = yes
                domcom_is_unacknowledged_blood_trigger = yes
                NOT = { has_character_flag = domcom_blood_registered }
            }
            save_temporary_scope_as = domcom_ui_blood_child

    set_variable = { name = domcom_ui_tantalized_men value = 0 }
    set_variable = { name = domcom_ui_pet_count value = 0 }

    clear_variable_list = domcom_ui_owned_page_list
    clear_variable_list = domcom_ui_lovers_page_list
    clear_variable_list = domcom_ui_prospects_page_list
    clear_variable_list = domcom_ui_blood_page_list
    clear_variable_list = domcom_ui_tantalized_page_list
    clear_variable_list = domcom_ui_secrets_page_list

    # World Cucks is lazy: only collect the lightweight reference roster when this is the page the player

    if = {

# Collect one year's tribute immediately. The subject pays first; a shortfall is collected from wealthy
# adult vassals, courtiers, and close/extended family.
domcom_demand_annual_tribute_effect = {
    save_scope_as = domcom_tribute_advance_subject
    var:domcom_patron = { save_scope_as = domcom_tribute_advance_patron }
    set_variable = { name = domcom_tribute_advance_shortfall value = domcom_annual_tribute_due_value }
    set_variable = { name = domcom_tribute_advance_collected value = 0 }

    if = {
        limit = { var:domcom_tribute_advance_shortfall > 0 gold > 0 }
        set_variable = { name = domcom_tribute_piece value = gold }
        if = { limit = { var:domcom_tribute_piece > var:domcom_tribute_advance_shortfall } set_variable = { name = domcom_tribute_piece value = var:domcom_tribute_advance_shortfall } }
        pay_short_term_gold = { target = scope:domcom_tribute_advance_patron gold = var:domcom_tribute_piece }
        change_variable = { name = domcom_tribute_advance_collected add = var:domcom_tribute_piece }
        change_variable = { name = domcom_tribute_advance_shortfall subtract = var:domcom_tribute_piece }
    }

    every_vassal = {
        limit = { is_alive = yes age >= 18 gold > 0 }
        add_to_temporary_list = domcom_tribute_advance_network
    }
    every_courtier_or_guest = {
        limit = { is_alive = yes age >= 18 gold > 0 NOT = { is_in_list = domcom_tribute_advance_network } }
        add_to_temporary_list = domcom_tribute_advance_network
    }
    every_close_or_extended_family_member = {
        limit = { is_alive = yes age >= 18 gold > 0 NOT = { is_in_list = domcom_tribute_advance_network } }
        add_to_temporary_list = domcom_tribute_advance_network
    }

    ordered_in_list = {
        list = domcom_tribute_advance_network
        order_by = gold
        max = 100
        check_range_bounds = no
        limit = { scope:domcom_tribute_advance_subject = { var:domcom_tribute_advance_shortfall > 0 } gold > 0 }
        save_temporary_scope_as = domcom_network_payer
        set_variable = { name = domcom_network_tribute_piece value = gold }
        if = {
            limit = { var:domcom_network_tribute_piece > scope:domcom_tribute_advance_subject.var:domcom_tribute_advance_shortfall }
            set_variable = { name = domcom_network_tribute_piece value = scope:domcom_tribute_advance_subject.var:domcom_tribute_advance_shortfall }
        }
        pay_short_term_gold = { target = scope:domcom_tribute_advance_patron gold = var:domcom_network_tribute_piece }
        add_opinion = { target = scope:domcom_tribute_advance_subject modifier = domcom_forced_to_cover_tribute_opinion }
        scope:domcom_tribute_advance_subject = {
            change_variable = { name = domcom_tribute_advance_collected add = scope:domcom_network_payer.var:domcom_network_tribute_piece }
            change_variable = { name = domcom_tribute_advance_shortfall subtract = scope:domcom_network_payer.var:domcom_network_tribute_piece }
        }
    }
    # If the subject and their network still cannot cover the annual demand, the subject is driven into
    # debt rather than the patron silently forgiving it.
    if = {
        limit = { var:domcom_tribute_advance_shortfall > 0 }
        scope:domcom_tribute_advance_patron = { add_gold = { value = scope:domcom_tribute_advance_subject.var:domcom_tribute_advance_shortfall } }
        add_gold = { value = var:domcom_tribute_advance_shortfall multiply = -1 }
        change_variable = { name = domcom_tribute_advance_collected add = var:domcom_tribute_advance_shortfall }
        set_variable = { name = domcom_tribute_advance_shortfall value = 0 }
    }
    add_character_flag = { flag = domcom_tribute_prepaid years = 1 }
}

# AI succession pressure: if an Owned ruler's primary heir is the AI patron's acknowledged child, the AI
# may press the same abdication demand available to players.
domcom_ai_manage_succession_effect = {
    save_scope_as = domcom_ai_succession_patron
    if = {
        limit = { exists = dynasty dynasty = { has_variable_list = domcom_owned_subjects } }
        dynasty = {
            random_in_list = {
                variable = domcom_owned_subjects
                limit = {
                    is_alive = yes is_landed = yes
                    has_character_flag = domcom_owned_cuck
                    has_variable = domcom_patron
                    var:domcom_patron = scope:domcom_ai_succession_patron
                    domcom_can_demand_abdication_for_blood_trigger = { PATRON = scope:domcom_ai_succession_patron }
                }
                save_scope_as = domcom_ai_abdication_target
            }
        }
        if = {
            limit = { exists = scope:domcom_ai_abdication_target }
            scope:domcom_ai_succession_patron = { save_scope_as = domcom_abdication_patron }
            scope:domcom_ai_abdication_target = {
                add_character_flag = domcom_ai_abdication_failed_roll
                random = {
                    chance = domcom_abdication_chance_value

# Run from the hegemon. Builds a one-pass roster of direct landed vassals.
domcom_begin_imperial_tribute_effect = {
    clear_variable_list = domcom_tribute_pending
    set_variable = { name = domcom_tribute_pending_count value = 0 }
    set_variable = { name = domcom_tribute_summoned_count value = 0 }
    # Per-season counters. These are the ones the Tribute page reports and the only ones a new season may
    # reset.
    set_variable = { name = domcom_tribute_gold_count value = 0 }
    set_variable = { name = domcom_tribute_prestige_count value = 0 }
    set_variable = { name = domcom_tribute_military_count value = 0 }
    set_variable = { name = domcom_tribute_dynastic_count value = 0 }
    set_variable = { name = domcom_tribute_household_season_count value = 0 }
    set_variable = { name = domcom_tribute_refusal_season_count value = 0 }
    # domcom_tribute_household_count and domcom_tribute_refusal_count are LIFETIME ledger counters and
    # are never reset here; zeroing them on every summons was wiping the career record the Ledger page
    # reads.
    domcom_initialize_ledger_counters_effect = yes

    every_vassal = {
        limit = { is_alive = yes is_landed = yes }

            add_to_variable_list = { name = domcom_tribute_pending target = scope:domcom_tribute_vassal_to_add }
            change_variable = { name = domcom_tribute_pending_count add = 1 }
        }
    }
    set_variable = { name = domcom_tribute_total_count value = var:domcom_tribute_pending_count }
    trigger_event = { id = domcom.500 days = 5 }
}

# Run from the patron when advancing the season.
domcom_advance_imperial_tribute_effect = {
    if = {
        limit = { has_character_flag = domcom_tribute_season_active has_variable_list = domcom_tribute_pending }

        # Drop petitioners who can no longer appear (dead, unlanded, no longer my vassal) and recount
        # from what is actually left, so the page never advertises an audience that will never happen.
        save_scope_as = domcom_tribute_prune_patron
        every_in_list = {
            variable = domcom_tribute_pending
            limit = { NOT = { AND = { is_alive = yes is_landed = yes is_vassal_of = scope:domcom_tribute_prune_patron } } }
            save_temporary_scope_as = domcom_tribute_stale
            scope:domcom_tribute_prune_patron = {
                remove_list_variable = { name = domcom_tribute_pending target = scope:domcom_tribute_stale }
            }
        }
        set_variable = { name = domcom_tribute_pending_count value = 0 }
        every_in_list = {
            variable = domcom_tribute_pending
            scope:domcom_tribute_prune_patron = { change_variable = { name = domcom_tribute_pending_count add = 1 } }
        }

        random_in_list = {
            variable = domcom_tribute_pending
            limit = { is_alive = yes is_landed = yes is_vassal_of = root }
            save_scope_as = domcom_tribute_vassal
        }
        if = {
            limit = { exists = scope:domcom_tribute_vassal }
            remove_list_variable = { name = domcom_tribute_pending target = scope:domcom_tribute_vassal }
            change_variable = { name = domcom_tribute_pending_count subtract = 1 }
            change_variable = { name = domcom_tribute_summoned_count add = 1 }
            save_scope_as = domcom_tribute_patron
            scope:domcom_tribute_vassal = { trigger_event = { id = domcom.510 days = 1 } }
        }
        else = { domcom_close_tribute_season_effect = yes }
    }
    else_if = {
        limit = { has_character_flag = domcom_tribute_season_active }
        domcom_close_tribute_season_effect = yes
    }
}

domcom_intimacy_discovered_effect = {
    scope:domcom_intimacy_patron = {
        add_prestige = scope:domcom_intimacy_prestige_loss
    }
    scope:domcom_intimacy_partner = { add_stress = domcom_intimacy_discovered_stress_value }

    if = {
        limit = { exists = scope:domcom_intimacy_cuckold }
        scope:domcom_intimacy_cuckold = {
            add_opinion = { target = scope:domcom_intimacy_patron modifier = domcom_intimacy_discovered_partner_opinion }
            every_close_family_member = {
                limit = { is_alive = yes }
                add_opinion = { target = scope:domcom_intimacy_patron modifier = domcom_intimacy_discovered_court_opinion }
            }

            random = {
                chance = domcom_intimacy_escape_chance_value
                add_character_flag = domcom_intimacy_patron_prevailed_roll
            }
            if = {
                limit = { has_character_flag = domcom_intimacy_patron_prevailed_roll }
                remove_character_flag = domcom_intimacy_patron_prevailed_roll
                # Seen, but the Patron is too dangerous/clever to corner.
                scope:domcom_intimacy_patron = {
                    add_character_modifier = { modifier = domcom_intimacy_prevailed_modifier years = 1 }
                    add_dread = 5
                }
                if = {
                    limit = { has_variable = domcom_cuck_xp }
                    change_variable = { name = domcom_cuck_xp add = 50 }
                    domcom_refresh_cuck_stage_effect = yes
                }
                scope:domcom_intimacy_patron = { trigger_event = { id = domcom.801 days = 1 } }
            }
            else = {
                # The spouse wins the confrontation. They gain the confidence and intimidation reward
                # that the Patron would have gained by getting away cleanly; imprisonment still requires
                # authority.
                add_character_modifier = { modifier = domcom_intimacy_enforcer_modifier years = 1 }
                if = { limit = { scope:domcom_intimacy_style = flag:discreet } add_dread = 2 }
                else_if = { limit = { scope:domcom_intimacy_style = flag:open } add_dread = 5 }
                else_if = { limit = { scope:domcom_intimacy_style = flag:brazen } add_dread = 10 }
                scope:domcom_intimacy_patron = {
                    domcom_dominion_rejection_stress_effect = yes
                    add_character_modifier = { modifier = domcom_intimacy_cornered_modifier years = 1 }
                }

                # Partner imprisonment chance scales with how brazen the choice was.
                if = {
                    limit = {
                        scope:domcom_intimacy_partner = { OR = { is_courtier_of = scope:domcom_intimacy_cuckold is_vassal_of = scope:domcom_intimacy_cuckold } }
                    }
                    if = { limit = { scope:domcom_intimacy_style = flag:discreet } random = { chance = 25 imprison = { target = scope:domcom_intimacy_partner type = house_arrest } } }
                    else_if = { limit = { scope:domcom_intimacy_style = flag:open } random = { chance = 50 imprison = { target = scope:domcom_intimacy_partner type = house_arrest } } }
                    else_if = { limit = { scope:domcom_intimacy_style = flag:brazen } random = { chance = 80 imprison = { target = scope:domcom_intimacy_partner type = house_arrest } } }
                }

                # The Patron can only be jailed if the spouse actually rules over them.
                if = {
                    limit = {
                        scope:domcom_intimacy_patron = { OR = { is_courtier_of = scope:domcom_intimacy_cuckold is_vassal_of = scope:domcom_intimacy_cuckold } }
                    }
                    if = { limit = { scope:domcom_intimacy_style = flag:discreet } random = { chance = 10 imprison = { target = scope:domcom_intimacy_patron type = house_arrest } } }
                    else_if = { limit = { scope:domcom_intimacy_style = flag:open } random = { chance = 25 imprison = { target = scope:domcom_intimacy_patron type = house_arrest } } }
                    else_if = { limit = { scope:domcom_intimacy_style = flag:brazen } random = { chance = 50 imprison = { target = scope:domcom_intimacy_patron type = house_arrest } } }
                }

                scope:domcom_intimacy_patron = { trigger_event = { id = domcom.802 days = 1 } }
            }
        }
    }
    else = {
        # No spouse to confront the pair; discovery is reputational, but there is nobody with standing to
        # corner the Patron.
        scope:domcom_intimacy_patron = {
            add_character_modifier = { modifier = domcom_intimacy_prevailed_modifier years = 1 }
            add_dread = 3
            trigger_event = { id = domcom.801 days = 1 }
        }
    }
}

            domcom_clear_obedience_hard_modifiers_effect = yes
            add_character_modifier = domcom_obedience_attached_hard_modifier

            domcom_clear_obedience_hard_modifiers_effect = yes
            add_character_modifier = domcom_obedience_tempted_hard_modifier

}

# Tolerance keyed to Obedience rather than to the ownership flags. Attached (500) is the line: below it a
# bond is exposed, at or above it the bond is protected and restorable exactly as an Owned subject's
# always was.
domcom_obedience_grants_tolerance_trigger_effect = {
    if = {
        limit = {
            NOT = { has_game_rule = domcom_obedience_weight_off }
            has_variable = domcom_obedience
            var:domcom_obedience >= 500
            has_variable = domcom_obedience_patron
        }
        set_variable = { name = domcom_accepts_communion value = yes }
        add_character_flag = block_infidelity_flag
    }
}

# --- the jealousy pass ------------------------------------------------------ Called once when the
# Patron takes someone new, against every existing intimate except the newcomer. Bounded by harem size,
# so no world scan.
domcom_run_jealousy_pass_effect = {
    if = {
        limit = { NOT = { has_game_rule = domcom_obedience_weight_off } }
        save_temporary_scope_as = domcom_jealousy_patron
        every_in_list = {
            variable = domcom_intimate_registry
            limit = {
                is_alive = yes
                age >= 18
                NOT = { this = scope:domcom_jealousy_rival }
            }
            domcom_score_jealousy_effect = yes
        }
    }
}

domcom_score_jealousy_effect = {
    # Precompute both attraction readings; the score value reads them back.
    scope:domcom_jealousy_patron = { save_temporary_scope_as = domcom_compatibility_target }
    set_variable = { name = domcom_jealousy_attr_patron value = domcom_pair_compatibility_value }
    scope:domcom_jealousy_rival = { save_temporary_scope_as = domcom_compatibility_target }
    set_variable = { name = domcom_jealousy_attr_rival value = domcom_pair_compatibility_value }

    set_variable = { name = domcom_jealousy_tier value = domcom_jealousy_tier_value }
    remove_variable = domcom_jealousy_attr_patron
    remove_variable = domcom_jealousy_attr_rival

    domcom_apply_jealousy_tier_effect = yes
    if = { limit = { has_variable = domcom_obedience } domcom_apply_obedience_weight_effect = yes }
}

domcom_apply_jealousy_tier_effect = {
    if = { limit = { NOT = { has_variable = domcom_obedience } } set_variable = { name = domcom_obedience value = 0 } }

    # 0 — pleased. She thinks better of him for it, and of the newcomer.
    if = {
        limit = { var:domcom_jealousy_tier = 0 }
        add_opinion = { target = scope:domcom_jealousy_patron modifier = domcom_jealousy_pleased_opinion opinion = 15 }
        add_opinion = { target = scope:domcom_jealousy_rival modifier = domcom_jealousy_pleased_opinion opinion = 20 }
        change_variable = { name = domcom_obedience add = 25 }
    }
    # 1 — indifferent. Nothing happens, and that is a result.
    else_if = { limit = { var:domcom_jealousy_tier = 1 } }
    # 2 — stung.
    else_if = {
        limit = { var:domcom_jealousy_tier = 2 }
        add_opinion = { target = scope:domcom_jealousy_patron modifier = domcom_jealousy_stung_opinion opinion = -10 }
        change_variable = { name = domcom_obedience subtract = 25 }
    }
    # 3 — hurt.
    else_if = {
        limit = { var:domcom_jealousy_tier = 3 }
        add_opinion = { target = scope:domcom_jealousy_patron modifier = domcom_jealousy_stung_opinion opinion = -20 }
        add_opinion = { target = scope:domcom_jealousy_rival modifier = domcom_jealousy_stung_opinion opinion = -15 }
        change_variable = { name = domcom_obedience subtract = 75 }
        add_stress = 15
    }
    # 4 — resentful. The bond may not survive.
    else_if = {
        limit = { var:domcom_jealousy_tier = 4 }
        add_opinion = { target = scope:domcom_jealousy_patron modifier = domcom_jealousy_resentful_opinion opinion = -30 }
        add_opinion = { target = scope:domcom_jealousy_rival modifier = domcom_jealousy_resentful_opinion opinion = -25 }
        change_variable = { name = domcom_obedience subtract = 150 }
        add_stress = 25
        domcom_jealousy_maybe_break_bond_effect = { CHANCE = 40 }
    }
    # 5 — hostile.
    else_if = {
        limit = { var:domcom_jealousy_tier = 5 }
        add_opinion = { target = scope:domcom_jealousy_patron modifier = domcom_jealousy_resentful_opinion opinion = -45 }
        add_opinion = { target = scope:domcom_jealousy_rival modifier = domcom_jealousy_resentful_opinion opinion = -40 }
        change_variable = { name = domcom_obedience subtract = 250 }
        add_stress = 35
        if = {
            limit = { NOT = { has_game_rule = domcom_obedience_weight_soft } }
            set_relation_rival = scope:domcom_jealousy_rival
        }
        domcom_jealousy_maybe_break_bond_effect = { CHANCE = 70 }
    }
    # 6 — dangerous.
    else_if = {
        limit = { var:domcom_jealousy_tier >= 6 }
        add_opinion = { target = scope:domcom_jealousy_patron modifier = domcom_jealousy_hostile_opinion opinion = -60 }
        add_opinion = { target = scope:domcom_jealousy_rival modifier = domcom_jealousy_hostile_opinion opinion = -60 }
        set_variable = { name = domcom_obedience value = 0 }
        add_stress = 45
        if = {
            limit = { NOT = { has_game_rule = domcom_obedience_weight_soft } }
            set_relation_rival = scope:domcom_jealousy_rival
            add_character_flag = domcom_jealousy_hostile
            set_variable = { name = domcom_jealousy_target value = scope:domcom_jealousy_rival }
            if = {
                limit = { NOT = { has_character_modifier = domcom_jealousy_hostile_modifier } }
                add_character_modifier = { modifier = domcom_jealousy_hostile_modifier years = 10 }
            }
            domcom_jealousy_maybe_break_bond_effect = { CHANCE = 100 }
        }
    }

    if = { limit = { var:domcom_obedience < 0 } set_variable = { name = domcom_obedience value = 0 } }
    remove_variable = domcom_jealousy_tier
}

# Soft never breaks a bond. Tolerance at Attached protects it regardless.
domcom_jealousy_maybe_break_bond_effect = {
    if = {
        limit = {
            NOT = { has_game_rule = domcom_obedience_weight_soft }
            NOT = { var:domcom_obedience >= 500 }
            NOT = { has_variable = domcom_accepts_communion }
            OR = {
                has_relation_lover = scope:domcom_jealousy_patron
                has_relation_soulmate = scope:domcom_jealousy_patron
            }
        }
        random = {
            chance = $CHANCE$
            if = { limit = { has_relation_soulmate = scope:domcom_jealousy_patron } remove_relation_soulmate = scope:domcom_jealousy_patron }
            if = { limit = { has_relation_lover = scope:domcom_jealousy_patron } remove_relation_lover = scope:domcom_jealousy_patron }
            remove_character_flag = domcom_communion_intimate_bound
            if = { limit = { has_variable = domcom_communion_intimate_patron } remove_variable = domcom_communion_intimate_patron }
            scope:domcom_jealousy_patron = {
                remove_list_variable = { name = domcom_intimate_registry target = prev }
                if = { limit = { is_ai = no } trigger_event = { id = domcom_jealousy.0001 days = 2 } }
            }
        }
    }
}

domcom_jealousy_hostile_yearly_effect = {
    if = {
        limit = {
            has_character_flag = domcom_jealousy_hostile
            NOT = { has_game_rule = domcom_obedience_weight_off }
            NOT = { has_game_rule = domcom_obedience_weight_soft }
            has_variable = domcom_jealousy_target
            var:domcom_jealousy_target = { is_alive = yes }
            is_capable_adult = yes
            NOT = { any_scheme = { scheme_type = murder } }
        }
        # Reconciliation: if she has come back around, she stops being a threat.
        if = {
            limit = {
                OR = {
                    opinion = { target = var:domcom_jealousy_target value >= 0 }
                    AND = { has_variable = domcom_obedience var:domcom_obedience >= 500 }
                }
            }
            remove_character_flag = domcom_jealousy_hostile
            remove_variable = domcom_jealousy_target
            if = { limit = { has_character_modifier = domcom_jealousy_hostile_modifier } remove_character_modifier = domcom_jealousy_hostile_modifier }
        }
        else = {
            random = {
                chance = 6
                modifier = { add = 4 has_trait = vengeful }
                modifier = { add = 4 has_trait = wrathful }
                modifier = { add = 3 has_trait = deceitful }
                modifier = { add = 3 intrigue >= 14 }
                modifier = { subtract = 3 has_trait = forgiving }
                modifier = { subtract = 4 has_trait = craven }
                var:domcom_jealousy_target = { save_temporary_scope_as = domcom_jealousy_mark }
                start_scheme = { type = murder target_character = scope:domcom_jealousy_mark }
            }
            # Rarer still, and uglier: the newcomer's children by the Patron.
            random = {
                chance = 3
                modifier = { add = 3 has_trait = vengeful }
                modifier = { add = 3 has_trait = ambitious }
                modifier = { subtract = 3 has_trait = compassionate }
                modifier = { subtract = 2 has_trait = forgiving }
                if = {
                    limit = {
                        var:domcom_jealousy_target = {
                            any_child = { is_alive = yes is_adult = yes }
                        }
                    }
                    var:domcom_jealousy_target = {
                        random_child = {
                            limit = { is_alive = yes is_adult = yes }
                            save_temporary_scope_as = domcom_jealousy_mark
                        }
                    }
                    if = {
                        limit = { exists = scope:domcom_jealousy_mark }
                        start_scheme = { type = murder target_character = scope:domcom_jealousy_mark }
                    }
                }
            }
        }
    }
}
