# Full sexual-attraction profile. Physical data remains managed by the original profile code; the new
# preference categories are deliberately lazy.

}

# Add a full Carnalitas sex-evidence patron to this potential cuck and mirror the target onto the patron
# so AI patrons can later act on the evidence without a world scan. Requires scope:domcom_sex_patron.
domcom_record_full_sex_evidence_effect = {
    save_scope_as = domcom_sex_cuck_target
    if = {
        limit = {
            NOT = {
                any_in_list = {
                    variable = domcom_sex_evidence_patrons
                    this = scope:domcom_sex_patron
                }
            }
        }
        add_to_variable_list = { name = domcom_sex_evidence_patrons target = scope:domcom_sex_patron }
    }
    scope:domcom_sex_patron = {
        if = {
            limit = {
                NOT = {
                    any_in_list = {
                        variable = domcom_sex_cuck_targets
                        this = scope:domcom_sex_cuck_target
                    }
                }
            }
            add_to_variable_list = { name = domcom_sex_cuck_targets target = scope:domcom_sex_cuck_target }
        }
    }
}

# never requires Carnalitas to exist.

        limit = {
            age >= 18

        scope:carn_sex_partner = { save_scope_as = domcom_sex_char_2 }

domcom_marital_preference_yearly_effect = {
    save_scope_as = domcom_compatibility_target
    every_spouse = {
        limit = { is_alive = yes age >= 18 }
        domcom_initialize_attraction_profile_effect = yes
        if = {
            limit = { NOT = { is_attracted_to_gender_of = scope:domcom_compatibility_target } }
            set_variable = { name = domcom_last_spouse_compatibility value = -90 }
        }
        else = { set_variable = { name = domcom_last_spouse_compatibility value = domcom_pair_compatibility_value } }
        if = {
            limit = { var:domcom_last_spouse_compatibility <= -65 }
            add_character_modifier = { modifier = domcom_extreme_marital_aversion_modifier years = 2 }
            if = {
                limit = {
                    is_ai = yes
                    NOT = { is_imprisoned = yes }
                    OR = { has_trait = sadistic has_trait = callous has_trait = vengeful has_trait = deceitful has_trait = wrathful }
                    NOT = { has_character_flag = domcom_marital_murder_attempt_recent }
                }
                random = {
                    chance = 15
                    add_character_flag = { flag = domcom_marital_murder_attempt_recent years = 5 }
                    start_scheme = { type = murder target = scope:domcom_compatibility_target }
                }
            }
        }
        else_if = {
            limit = { var:domcom_last_spouse_compatibility >= 90 }
            add_character_modifier = { modifier = domcom_extreme_marital_fixation_modifier years = 2 }
        }
    }
}
