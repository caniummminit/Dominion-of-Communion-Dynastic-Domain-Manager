# 0.8.3 — defaults to ON, and this is deliberate. CK3 game rules are fixed when
# a save is created and cannot be changed afterwards, so a rule defaulting to
# Off made the rebuild decisions permanently unreachable on exactly the saves
# they exist to repair: any save started before the rule existed can never turn
# it on. Inverted so the tools are present unless a player opts out, and so a
# save with no record of this rule at all still shows them.
domcom_debug_tools = {
    categories = { gameplay }
    default = domcom_debug_tools_on

    domcom_debug_tools_on = { }
    domcom_debug_tools_off = { }
}
