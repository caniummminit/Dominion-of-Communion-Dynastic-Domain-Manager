domcom_cuck_evidence_trigger = {
    # Run from the prospective cuck. Requires scope:actor.
    is_alive = yes
    age >= 18
    NOT = { is_consort_of = scope:actor }
    OR = {
        NOT = { has_variable = domcom_cuck_patron }
        var:domcom_cuck_patron = scope:actor
    }
    OR = {
        # Optional Carnalitas bridge records actual sex with a spouse/concubine or soulmate even when no
        # persistent Lover relation or child was created.
        domcom_has_full_sex_evidence_from_actor_trigger = yes

        # Standalone Dominion evidence remains relationship/parentage based.
        any_consort = {
            is_alive = yes
            age >= 18
            OR = {
                has_relation_lover = scope:actor
                has_relation_soulmate = scope:actor
                any_child = {
                    OR = {
                        father ?= scope:actor
                        mother ?= scope:actor
                        real_father ?= scope:actor
                    }
                }
            }
        }
        any_relation = {
            type = soulmate
            is_alive = yes
            age >= 18
            OR = {
                has_relation_lover = scope:actor
                has_relation_soulmate = scope:actor
                any_child = {
                    OR = {
                        father ?= scope:actor
                        mother ?= scope:actor
                        real_father ?= scope:actor
                    }
                }
            }
        }
    }
}

    NOT = { is_close_or_extended_family_of = $PARTNER$ }

    NOT = { has_trait = celibate }
