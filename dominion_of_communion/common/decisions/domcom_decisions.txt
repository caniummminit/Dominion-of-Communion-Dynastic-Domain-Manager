domcom_open_communion_decision = {
    picture = { reference = "gfx/interface/illustrations/decisions/decision_dynasty_house.dds" }
    decision_group_type = major
    ai_check_interval = 0
    sort_order = 1

    desc = domcom_open_communion_decision_desc
    selection_tooltip = domcom_open_communion_decision_tooltip
    confirm_text = domcom_open_communion_decision_confirm

    # 0.7.9 — the decision stays visible from CK3's age of majority so a young
    # ruler can see the mod is installed and working, but it is not valid until
    # 18. Previously it opened at 16 onto a Ledger where every roster was empty
    # and every button inert, because all the mechanics behind it gate at 18.
    # That read as a broken mod rather than a gated one.
    is_shown = {
        is_playable_character = yes
        is_adult = yes
    }

    is_valid_showing_failures_only = {
        is_imprisoned = no
        custom_tooltip = {
            text = domcom_ledger_age_gate_tt
            age >= 18
        }
    }

    effect = {
        hidden_effect = {
            # rc4.7 — repair/backfill intimate state before building the Ledger,
            # so the first frame already shows restored Lovers and household links.
            domcom_enforce_communion_for_patron_effect = yes
            domcom_prepare_ui_effect = yes
            set_variable = { name = domcom_ui_page value = flag:overview }
            set_variable = { name = domcom_ui_open value = yes }
        }
    }

    ai_potential = { always = no }
    ai_will_do = { base = 0 }
}
