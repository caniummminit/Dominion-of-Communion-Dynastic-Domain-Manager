# Root/this is the interaction recipient. scope:actor and scope:recipient are supplied
# by the character interaction. The result is a percentage, capped at 5-95.
domcom_alpha_check_chance_value = {
    value = 18
    if = {
        limit = { scope:recipient = { has_variable = domcom_cuck_xp } }
        add = { value = scope:recipient.var:domcom_cuck_xp multiply = 0.05 }
    }
    # A previous pre-cuck Alpha victory gives a modest edge once the same rival
    # returns for a post-cuck reassertion.
    if = {
        limit = {
            scope:recipient = {
                has_character_flag = domcom_dominated
                has_variable = domcom_dominance_patron
                var:domcom_dominance_patron = scope:actor
            }
        }
        add = 12
    }
    if = {
        limit = { scope:recipient = { domcom_has_lover_betrayal_from_actor_trigger = yes } }
        add = 8
    }
    add = { value = scope:actor.diplomacy multiply = 0.70 }
    subtract = { value = scope:recipient.diplomacy multiply = 0.65 }
    add = { value = scope:actor.intrigue multiply = 0.75 }
    subtract = { value = scope:recipient.intrigue multiply = 0.70 }
    add = { value = scope:actor.prowess multiply = 0.45 }
    subtract = { value = scope:recipient.prowess multiply = 0.40 }
    add = { value = scope:actor.prestige_level multiply = 5 }
    subtract = { value = scope:recipient.prestige_level multiply = 5 }
    add = { value = scope:actor.dread multiply = 0.12 }

    if = { limit = { scope:actor = { highest_held_title_tier = tier_empire } } add = 18 }
    else_if = { limit = { scope:actor = { highest_held_title_tier = tier_kingdom } } add = 12 }
    else_if = { limit = { scope:actor = { highest_held_title_tier = tier_duchy } } add = 6 }
    else_if = { limit = { scope:actor = { is_landed = no } } add = -8 }

    if = { limit = { scope:recipient = { highest_held_title_tier = tier_empire } } add = -22 }
    else_if = { limit = { scope:recipient = { highest_held_title_tier = tier_kingdom } } add = -14 }
    else_if = { limit = { scope:recipient = { highest_held_title_tier = tier_duchy } } add = -7 }

    if = { limit = { scope:actor.max_military_strength > scope:recipient.max_military_strength } add = 7 }
    if = { limit = { scope:recipient.max_military_strength > scope:actor.max_military_strength } add = -7 }

    if = { limit = { scope:actor = { has_trait = brave } } add = 6 }
    if = { limit = { scope:actor = { has_trait = ambitious } } add = 6 }
    if = { limit = { scope:actor = { has_trait = arrogant } } add = 5 }
    if = { limit = { scope:actor = { has_trait = gregarious } } add = 4 }
    if = { limit = { scope:actor = { has_trait = callous } } add = 4 }
    if = { limit = { scope:actor = { has_trait = sadistic } } add = 4 }

    if = { limit = { scope:recipient = { has_trait = stubborn } } add = -8 }
    if = { limit = { scope:recipient = { has_trait = brave } } add = -8 }
    if = { limit = { scope:recipient = { has_trait = arrogant } } add = -6 }
    if = { limit = { scope:recipient = { has_trait = ambitious } } add = -5 }
    if = { limit = { scope:recipient = { has_trait = craven } } add = 12 }
    if = { limit = { scope:recipient = { has_trait = content } } add = 8 }
    if = { limit = { scope:recipient = { has_trait = trusting } } add = 6 }

    min = 3
    max = 97
    round = yes
}

# 1.0.0-rc4.4 — length-led, was girth-led.
#
# The old weighting was 0.4 length / 0.6 girth, which produced results players
# read as simply wrong: 5.0 x 6.0 and 6.5 x 5.0 both index 56, so a five-inch
# one sat in the same band as a six-and-a-half-inch one. Girth deserves real
# weight, but not more than length, because length is the dimension a player
# reads first and judges the tier against.
domcom_endowment_index_value = {
    value = 0
    if = {
        limit = { exists = root }
        if = { limit = { has_variable = domcom_length_tenths } add = { value = var:domcom_length_tenths multiply = 0.6 } }
        if = { limit = { has_variable = domcom_girth_tenths } add = { value = var:domcom_girth_tenths multiply = 0.4 } }
    }
    round = yes
}

# 1.0.0-rc4.3 — heredity seeds.
#
# A son starts from his sire's figure pulled 40% of the way back toward the
# population mean, then takes a symmetric random step in domcom_assign_endowment_effect.
# Without the regression term a single outlier ancestor would pin an entire
# dynasty to the top or bottom tier forever; with it, a line drifts toward an
# extreme over generations and can drift back out again.
#
# The divide/round/multiply tail snaps the result to a multiple of 5, matching
# the granularity of the flat population tables so sired and unsired characters
# occupy the same value space.
#
# Both values fall back to the population mean, so they are safe to evaluate on
# a character with no sire on record even though nothing currently does.
domcom_sire_length_seed_value = {
    value = 0
    if = {
        limit = { exists = root has_variable = domcom_sire_length_tenths }
        add = var:domcom_sire_length_tenths
        subtract = 55
        multiply = 0.6
        add = 55
    }
    else = { add = 55 }
    divide = 5
    round = yes
    multiply = 5
}

domcom_sire_girth_seed_value = {

        add = 50
    }

    multiply = 5
}

# 1.0.0-rc4.4 — stress became personality-dependent.
#
# Every Dominion stress change was previously a flat number, which encoded one
# mentality: a man who finds all of this shameful and is worn down by it. That
# is a real character, but it is not the only one, and forcing it on every
# playthrough made the patron read as someone who does not want what he is
# doing. These values let the same act steady one character and cost another.
#
# Traits that pull toward relief are the ones for whom this is self-expression;
# traits that pull toward cost are the ones for whom it is transgression. A
# chaste, humble patron still pays for success - that is the point, his
# conscience is the obstacle. A lustful or deviant one is genuinely eased by it.
domcom_dominion_success_stress_value = {
    value = -5
    if = { limit = { has_trait = lustful } subtract = 8 }
    if = { limit = { has_trait = deviant } subtract = 6 }
    if = { limit = { has_trait = arrogant } subtract = 4 }
    if = { limit = { has_trait = sadistic } subtract = 4 }
    if = { limit = { has_trait = callous } subtract = 2 }
    if = { limit = { has_trait = ambitious } subtract = 2 }
    if = { limit = { has_trait = chaste } add = 12 }
    if = { limit = { has_trait = humble } add = 5 }
    if = { limit = { has_trait = compassionate } add = 4 }
    if = { limit = { has_trait = honest } add = 3 }
    if = { limit = { has_trait = shy } add = 3 }
}
