# Root/this is the interaction recipient. scope:actor and scope:recipient are supplied by the character
# interaction. The result is a percentage, capped at 5-95.
domcom_alpha_check_chance_value = {
    value = 18
    if = {
        limit = { scope:recipient = { has_variable = domcom_cuck_xp } }
        add = { value = scope:recipient.var:domcom_cuck_xp multiply = 0.05 }
    }
    # A previous pre-cuck Alpha victory gives a modest edge once the same rival returns for a post-cuck
    # reassertion.
    if = {
        limit = {
            scope:recipient = {
                has_character_flag = domcom_dominated
                has_variable = domcom_dominance_patron
                var:domcom_dominance_patron = scope:actor
            }
        }
        add = 12
    }
    if = {
        limit = { scope:recipient = { domcom_has_lover_betrayal_from_actor_trigger = yes } }
        add = 8
    }
    if = {
        limit = { scope:recipient = { domcom_spouse_taken_by_actor_trigger = yes } }
        add = 14
    }
    add = { value = scope:actor.diplomacy multiply = 0.70 }
    subtract = { value = scope:recipient.diplomacy multiply = 0.65 }
    add = { value = scope:actor.intrigue multiply = 0.75 }
    subtract = { value = scope:recipient.intrigue multiply = 0.70 }
    add = { value = scope:actor.prowess multiply = 0.45 }
    subtract = { value = scope:recipient.prowess multiply = 0.40 }
    add = { value = scope:actor.prestige_level multiply = 5 }
    subtract = { value = scope:recipient.prestige_level multiply = 5 }
    add = { value = scope:actor.dread multiply = 0.12 }

    if = { limit = { scope:actor = { highest_held_title_tier = tier_empire } } add = 18 }
    else_if = { limit = { scope:actor = { highest_held_title_tier = tier_kingdom } } add = 12 }
    else_if = { limit = { scope:actor = { highest_held_title_tier = tier_duchy } } add = 6 }
    else_if = { limit = { scope:actor = { is_landed = no } } add = -8 }

    if = { limit = { scope:recipient = { highest_held_title_tier = tier_empire } } add = -22 }
    else_if = { limit = { scope:recipient = { highest_held_title_tier = tier_kingdom } } add = -14 }
    else_if = { limit = { scope:recipient = { highest_held_title_tier = tier_duchy } } add = -7 }

    if = { limit = { scope:actor.max_military_strength > scope:recipient.max_military_strength } add = 7 }
    if = { limit = { scope:recipient.max_military_strength > scope:actor.max_military_strength } add = -7 }

    if = { limit = { scope:actor = { has_trait = brave } } add = 6 }
    if = { limit = { scope:actor = { has_trait = ambitious } } add = 6 }
    if = { limit = { scope:actor = { has_trait = arrogant } } add = 5 }
    if = { limit = { scope:actor = { has_trait = gregarious } } add = 4 }
    if = { limit = { scope:actor = { has_trait = callous } } add = 4 }
    if = { limit = { scope:actor = { has_trait = sadistic } } add = 4 }

    if = { limit = { scope:recipient = { has_trait = stubborn } } add = -8 }
    if = { limit = { scope:recipient = { has_trait = brave } } add = -8 }
    if = { limit = { scope:recipient = { has_trait = arrogant } } add = -6 }
    if = { limit = { scope:recipient = { has_trait = ambitious } } add = -5 }
    if = { limit = { scope:recipient = { has_trait = craven } } add = 12 }
    if = { limit = { scope:recipient = { has_trait = content } } add = 8 }
    if = { limit = { scope:recipient = { has_trait = trusting } } add = 6 }

    min = 3
    max = 97
    round = yes
}

# ROOT                            the existing partner, doing the judging
# scope:domcom_jealousy_patron    the Patron
# scope:domcom_jealousy_rival     the person he has just taken
domcom_jealousy_tier_value = {
    value = 3

    # --- how much she likes the Patron -------------------------------------
    if = { limit = { opinion = { target = scope:domcom_jealousy_patron value >= 80 } } add = -1.5 }
    else_if = { limit = { opinion = { target = scope:domcom_jealousy_patron value >= 50 } } add = -1 }
    else_if = { limit = { opinion = { target = scope:domcom_jealousy_patron value >= 20 } } add = -0.5 }
    else_if = { limit = { opinion = { target = scope:domcom_jealousy_patron value <= -50 } } add = 1.5 }
    else_if = { limit = { opinion = { target = scope:domcom_jealousy_patron value <= -20 } } add = 1 }

    # --- how much she likes the newcomer -----------------------------------
    if = { limit = { opinion = { target = scope:domcom_jealousy_rival value >= 50 } } add = -1 }
    else_if = { limit = { opinion = { target = scope:domcom_jealousy_rival value >= 20 } } add = -0.5 }
    else_if = { limit = { opinion = { target = scope:domcom_jealousy_rival value <= -50 } } add = 1.25 }
    else_if = { limit = { opinion = { target = scope:domcom_jealousy_rival value <= -20 } } add = 0.75 }

    # --- Obedience, the heaviest term --------------------------------------
    if = {
        limit = {
            has_variable = domcom_obedience_patron
            var:domcom_obedience_patron = scope:domcom_jealousy_patron
            has_variable = domcom_obedience
        }
        if = { limit = { var:domcom_obedience >= 750 } add = -2 }
        else_if = { limit = { var:domcom_obedience >= 500 } add = -1.25 }
        else_if = { limit = { var:domcom_obedience >= 250 } add = -0.5 }
        else = { add = 0.5 }
    }
    else = { add = 0.5 }

    # scope:domcom_compatibility_target for each and a script value cannot swap
    if = { limit = { var:domcom_jealousy_attr_patron >= 40 } add = 0.75 }
    else_if = { limit = { var:domcom_jealousy_attr_patron >= 20 } add = 0.4 }

    # --- her attraction to the newcomer: she sees the appeal too ------------
    if = { limit = { var:domcom_jealousy_attr_rival >= 40 } add = -1.25 }
    else_if = { limit = { var:domcom_jealousy_attr_rival >= 20 } add = -0.6 }
    else_if = { limit = { var:domcom_jealousy_attr_rival <= -20 } add = 0.5 }

    # --- temperament -------------------------------------------------------
    if = { limit = { has_trait = lustful } add = -0.5 }
    if = { limit = { has_trait = forgiving } add = -0.5 }
    if = { limit = { has_trait = deceitful } add = -0.25 }
    if = { limit = { has_trait = content } add = -0.25 }
    if = { limit = { has_trait = vengeful } add = 0.75 }
    if = { limit = { has_trait = wrathful } add = 0.75 }
    if = { limit = { has_trait = arrogant } add = 0.5 }
    if = { limit = { has_trait = paranoid } add = 0.5 }
    if = { limit = { has_trait = chaste } add = 0.5 }
    if = { limit = { has_trait = just } add = 0.25 }

    # A spouse who is watching her own husband's Patron collect another is a different case from a
    # courtier, and takes it harder.
    if = { limit = { is_married = yes NOT = { is_spouse_of = scope:domcom_jealousy_patron } } add = 0.25 }

    min = 0
    max = 6
    round = yes
}
