
domcom_endowment_micro = {
    # One trait per group is enforced natively, which also gives the six tiers a ladder instead of thirty
    # opposites pairs. Same pattern Carnalitas uses.
    group = domcom_endowment
    level = 1

    desc = {
        first_valid = {
            triggered_desc = {
                trigger = { NOT = { exists = this } }
                desc = trait_domcom_endowment_micro_desc
            }
            triggered_desc = {
                trigger = {
                    OR = {
                        NOT = { has_variable = domcom_length_tenths }
                        NOT = { has_variable = domcom_girth_tenths }
                    }
                }
                desc = trait_domcom_endowment_micro_desc
            }
            desc = trait_domcom_endowment_micro_character_desc
        }
    }

    physical = yes
    genetic = yes
    shown_in_ruler_designer = yes
    birth = 0
    random_creation = 0
    inherit_chance = 0
    both_parent_has_trait_inherit_chance = 0

    icon = domcom_endowment_micro.dds

    diplomacy = -3
    martial = -4
    intrigue = 3
    learning = 4
    attraction_opinion = 10

}
