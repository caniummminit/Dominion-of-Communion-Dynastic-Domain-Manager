domcom_ui_alpha_check = {
    scope = character
    saved_scopes = { target }
    is_valid = {
        age >= 18 is_alive = yes is_imprisoned = no
        # 0.9.9 — same per-quarter allowance as the right-click path, so the two
        # interfaces draw on one budget rather than two. The Gauntlet is exempt.
        domcom_alpha_budget_available_trigger = yes
        scope:target = {
            NOT = { has_character_flag = domcom_ui_lock_alpha_check }
            is_alive = yes age >= 18 is_imprisoned = no
            has_character_flag = domcom_cucked
            has_variable = domcom_cuck_patron
            var:domcom_cuck_patron = root
            NOT = { AND = { has_variable = domcom_cuck_xp var:domcom_cuck_xp >= 1000 } }
        }
    }
    effect = {
        if = {
            limit = { scope:target = { NOT = { has_character_flag = domcom_ui_lock_alpha_check } } }
            scope:target = { add_character_flag = { flag = domcom_ui_lock_alpha_check days = 1 } }
            change_variable = { name = domcom_alpha_attempts add = 1 }
            domcom_spend_alpha_budget_effect = yes
            save_scope_as = domcom_ui_patron
            # 0.9.7 — a manual check spends the same yearly challenge the Gauntlet
            # spends. Without this the rival stayed "fresh" and could be counted
            # toward the four-rival minimum and then challenged twice.
            scope:target = {
                add_character_flag = { flag = domcom_recent_alpha_challenge days = 365 }
                set_variable = { name = domcom_last_alpha_patron value = root }
            }
            scope:target = {
                set_variable = { name = domcom_ui_alpha_chance value = domcom_ui_alpha_check_chance_value }
                random = {
                    chance = var:domcom_ui_alpha_chance
                    domcom_own_cuck_from_ui_effect = yes
                    add_character_flag = domcom_ui_alpha_success
                }
            }
            if = {
                limit = { scope:target = { has_character_flag = domcom_ui_alpha_success } }
                scope:target = { remove_character_flag = domcom_ui_alpha_success }
                # 0.8.6 — same stress consequences as the right-click path.
                domcom_dominion_success_stress_effect = yes
                change_variable = { name = domcom_alpha_successes add = 1 }
                send_interface_toast = { type = event_toast_effect_good title = domcom_alpha_check_success_toast left_icon = root right_icon = scope:target }
            }
            else = {
                domcom_dominion_rejection_stress_effect = yes
                scope:target = {
                    if = { limit = { NOT = { has_variable = domcom_cuck_xp } } set_variable = { name = domcom_cuck_xp value = 100 } }
                    change_variable = { name = domcom_cuck_xp add = -50 }
                    if = { limit = { var:domcom_cuck_xp < 100 } set_variable = { name = domcom_cuck_xp value = 100 } }
                    add_opinion = { target = root modifier = domcom_resisted_alpha_opinion }
                    domcom_refresh_cuck_stage_effect = yes
                }
                change_variable = { name = domcom_alpha_failures add = 1 }
                send_interface_toast = { type = event_toast_effect_bad title = domcom_alpha_check_failure_toast left_icon = root right_icon = scope:target }
            }
            domcom_prepare_ui_effect = yes
        }
    }
}

domcom_ui_alpha_gauntlet = {
    scope = character
    is_valid = {
        NOT = { has_character_flag = domcom_ui_lock_alpha_gauntlet }
        is_alive = yes age >= 18 has_variable_list = domcom_ui_cucked_raw
        # A Gauntlet is a run, not a bulk button. Four fresh rivals minimum.
        any_in_list = {
            variable = domcom_ui_cucked_raw
            count >= 4
            is_alive = yes age >= 18 has_character_flag = domcom_cucked
            NOT = { has_character_flag = domcom_owned_cuck }
            has_variable = domcom_cuck_patron var:domcom_cuck_patron = root
            NOT = {
                AND = {
                    has_character_flag = domcom_recent_alpha_challenge
                    has_variable = domcom_last_alpha_patron
                    var:domcom_last_alpha_patron = root
                }
            }
        }
    }
    effect = {
        if = {
            limit = { NOT = { has_character_flag = domcom_ui_lock_alpha_gauntlet } }
            add_character_flag = { flag = domcom_ui_lock_alpha_gauntlet days = 1 }
            domcom_initialize_ledger_counters_effect = yes
            # NOTE: the Gauntlet is deliberately exempt from the Alpha Check
            # budget. It neither checks domcom_alpha_budget_available_trigger nor
            # calls domcom_spend_alpha_budget_effect. A Gauntlet is one committed
            # run against a roster the player already built; charging it eight
            # separate checks would make the budget the only thing that matters.
            remove_character_flag = domcom_gauntlet_failed
            set_variable = { name = domcom_gauntlet_eligible value = 0 }
            set_variable = { name = domcom_gauntlet_cleared value = 0 }
            set_variable = { name = domcom_gauntlet_multiplier value = 0.90 }
            save_scope_as = domcom_gauntlet_patron
            # 0.9.7 — recount against live state before anything is spent. The
            # roster can be stale by the time the button is pressed (a rival was
            # just Owned, died, or was individually checked).
            domcom_prepare_ui_effect = yes
            save_scope_as = domcom_gauntlet_patron
            every_in_list = {
                variable = domcom_ui_cucked_raw
                limit = {
                    is_alive = yes age >= 18 has_character_flag = domcom_cucked
                    NOT = { has_character_flag = domcom_owned_cuck }
                    has_variable = domcom_cuck_patron var:domcom_cuck_patron = scope:domcom_gauntlet_patron
                    NOT = {
                        AND = {
                            has_character_flag = domcom_recent_alpha_challenge
                            has_variable = domcom_last_alpha_patron
                            var:domcom_last_alpha_patron = scope:domcom_gauntlet_patron
                        }
                    }
                }
                scope:domcom_gauntlet_patron = { change_variable = { name = domcom_gauntlet_eligible add = 1 } }
            }
            if = {
              limit = { var:domcom_gauntlet_eligible >= 4 }
              ordered_in_list = {
                variable = domcom_ui_cucked_raw
                order_by = { value = domcom_sort_alpha_value multiply = -1 }
                max = 500 check_range_bounds = no
                limit = {
                    is_alive = yes age >= 18 has_character_flag = domcom_cucked NOT = { has_character_flag = domcom_owned_cuck }
                    has_variable = domcom_cuck_patron var:domcom_cuck_patron = scope:domcom_gauntlet_patron
                    NOT = {
                        AND = {
                            has_character_flag = domcom_recent_alpha_challenge
                            has_variable = domcom_last_alpha_patron
                            var:domcom_last_alpha_patron = scope:domcom_gauntlet_patron
                        }
                    }
                }
                if = {
                    limit = { scope:domcom_gauntlet_patron = { NOT = { has_character_flag = domcom_gauntlet_failed } } }
                    save_scope_as = domcom_gauntlet_target
                    add_character_flag = { flag = domcom_recent_alpha_challenge days = 365 }
                    set_variable = { name = domcom_last_alpha_patron value = scope:domcom_gauntlet_patron }
                    add_character_flag = domcom_gauntlet_failed_roll
                    random = {
                        chance = domcom_gauntlet_alpha_chance_value
                        remove_character_flag = domcom_gauntlet_failed_roll
                        scope:domcom_gauntlet_patron = {
                            save_scope_as = domcom_owner
                            change_variable = { name = domcom_alpha_attempts add = 1 }
                            change_variable = { name = domcom_alpha_successes add = 1 }
                            change_variable = { name = domcom_gauntlet_cleared add = 1 }
                            change_variable = { name = domcom_gauntlet_multiplier add = -0.03 }
                        }
                        domcom_own_cuck_generic_effect = yes
                    }
                    if = {
                        limit = { has_character_flag = domcom_gauntlet_failed_roll }
                        remove_character_flag = domcom_gauntlet_failed_roll
                        # Match the normal post-cuck Alpha rule: a resisted check
                        # restores 50 XP of resistance rather than rewarding failure.
                        if = { limit = { NOT = { has_variable = domcom_cuck_xp } } set_variable = { name = domcom_cuck_xp value = 0 } }
                        if = { limit = { var:domcom_cuck_xp >= 50 } change_variable = { name = domcom_cuck_xp subtract = 50 } }
                        else = { set_variable = { name = domcom_cuck_xp value = 0 } }
                        add_opinion = { target = scope:domcom_gauntlet_patron modifier = domcom_resisted_alpha_opinion }
                        domcom_refresh_cuck_stage_effect = yes
                        scope:domcom_gauntlet_patron = { add_character_flag = domcom_gauntlet_failed change_variable = { name = domcom_alpha_attempts add = 1 } change_variable = { name = domcom_alpha_failures add = 1 } }
                        save_scope_as = domcom_gauntlet_failed_target
                    }
                }
              }
              if = {
                  limit = { NOT = { has_character_flag = domcom_gauntlet_failed } var:domcom_gauntlet_cleared >= var:domcom_gauntlet_eligible }
                  add_character_modifier = { modifier = domcom_dominion_unchallenged_modifier years = 5 }
                  change_variable = { name = domcom_perfect_gauntlets add = 1 }
                  send_interface_toast = { type = event_toast_effect_good title = DOMCOM_UI_GAUNTLET_PERFECT_TOAST left_icon = root }
              }
              else_if = {
                  limit = { has_character_flag = domcom_gauntlet_failed exists = scope:domcom_gauntlet_failed_target }
                  send_interface_toast = { type = event_toast_effect_bad title = DOMCOM_UI_GAUNTLET_FAILED_TOAST left_icon = root right_icon = scope:domcom_gauntlet_failed_target }
              }
            }
            else = {
              # Too few fresh rivals. Nothing is spent and no one is challenged.
              send_interface_toast = { type = event_toast_effect_neutral title = DOMCOM_UI_GAUNTLET_TOO_FEW_TOAST left_icon = root }
            }
            remove_character_flag = domcom_gauntlet_failed
            remove_variable = domcom_gauntlet_multiplier
            domcom_prepare_ui_effect = yes
        }
    }
}

# 0.8.5 — receive every registered prospect in one action. Individual Receive
# Them buttons are kept: the per-character choice is the interesting decision,
# this is for players who have twenty of them queued and do not want twenty
# clicks. Locked for a day like the other bulk actions so it cannot be spammed.
domcom_ui_receive_all_prospects = {
    scope = character
    is_valid = {
        NOT = { has_character_flag = domcom_ui_lock_receive_all }
        age >= 18 is_alive = yes
        has_variable_list = domcom_ui_prospects_raw
        any_in_list = {
            variable = domcom_ui_prospects_raw
            is_alive = yes age >= 18
            has_variable = domcom_crush_patron
            var:domcom_crush_patron = root
        }
    }
    effect = {
        add_character_flag = { flag = domcom_ui_lock_receive_all days = 1 }
        remove_variable = domcom_ui_open
        save_scope_as = domcom_receive_all_patron
        every_in_list = {
            variable = domcom_ui_prospects_raw
            limit = {
                is_alive = yes age >= 18
                has_variable = domcom_crush_patron
                var:domcom_crush_patron = scope:domcom_receive_all_patron
            }
            add_character_flag = { flag = domcom_ui_lock_receive_prospect days = 1 }
            save_scope_as = domcom_admirer
            # Staggered rather than fired together. domcom.200 is one popup per
            # prospect, so releasing twenty at once would just relocate the
            # flood the player is trying to escape.
            scope:domcom_receive_all_patron = { trigger_event = { id = domcom.200 days = { 1 20 } } }
        }
        domcom_prepare_ui_effect = yes
    }
}
