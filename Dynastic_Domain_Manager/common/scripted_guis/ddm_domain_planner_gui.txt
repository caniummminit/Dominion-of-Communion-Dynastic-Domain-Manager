ddm_ui_run_settlement = {
    scope = character
    is_valid = { is_landed = yes is_at_war = no }
    effect = {
        ddm_initialize_settings_effect = yes
        if = {
            limit = { OR = { ddm_domain_plan_limit_changed_trigger = yes ddm_domain_plan_target_mismatch_trigger = yes } }
            set_variable = { name = ddm_pending_domain_action value = 1 }
            ddm_prepare_domain_planner_effect = yes
            set_variable = { name = ddm_ui_page value = flag:estate }
        }
        else = {
            # The office is intentionally paused while open. A full settlement
            # needs one simulation day so the post-settlement alliance phase can
            # settle cleanly. The GUI hides the office and resumes the player's
            # prior simulation speed; a hidden completion event reopens it after
            # one day and the office pauses the simulation again.
            set_variable = { name = ddm_ui_processing_settlement value = 1 }
            remove_variable = ddm_ui_open
            ddm_run_full_settlement_effect = yes
            set_variable = { name = ddm_ui_records_mode value = 2 }
            set_variable = { name = ddm_ui_page value = flag:records }
            trigger_event = { id = ddm_manager.0111 days = 1 }
        }
    }
}

ddm_ui_queue_settlement = {

    scope = landed_title
    is_valid = {
        holder = { has_variable = ddm_show_domain_planner }
        this != holder.capital_county
        OR = { has_variable = ddm_manual_core holder = { var:ddm_planner_selected_count < var:ddm_planner_target_count } }
    }
    effect = {
        if = { limit = { has_variable = ddm_manual_core } remove_variable = ddm_manual_core }
        else = { set_variable = ddm_manual_core }
        remove_variable = ddm_manual_shed
        remove_variable = ddm_manual_neutral
        holder = { ddm_refresh_domain_planner_counts_effect = yes }
    }
}
