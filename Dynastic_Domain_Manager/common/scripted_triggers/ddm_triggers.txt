# Soft core = holdings the player strongly prefers to keep. In Best-Effort mode these can still be
# granted if that is the only way to reach the actual domain limit.
ddm_title_is_protected_core_trigger = {
    OR = {
        this = scope:ddm_actor.capital_county
        AND = {
            tier = tier_county
            exists = scope:ddm_actor.capital_county.de_jure_liege
            de_jure_liege = scope:ddm_actor.capital_county.de_jure_liege
        }
        AND = {
            tier = tier_county
            title_province = { has_stationed_regiment = yes }
        }
        AND = {
            tier = tier_county
            title_province = { has_special_building = yes }
        }
    }
}

# The realm capital is always hard-protected. With Absolute Core protection, all legacy protected
# counties are hard-protected. With Best-Effort protection, they are preferences rather than blockers.
ddm_title_is_hard_protected_core_trigger = {
    OR = {
        this = scope:ddm_actor.capital_county
        has_variable = ddm_manual_core
    }
}

ddm_county_can_be_distributed_trigger = {
    tier = tier_county
    is_titular = no
    holder = scope:ddm_actor
    NOT = { ddm_title_is_hard_protected_core_trigger = yes }
}

ddm_barony_can_be_distributed_trigger = {
    tier = tier_barony
    holder = scope:ddm_actor
    title_province = { is_county_capital = no }
    NOT = { ddm_title_is_hard_protected_core_trigger = yes }
}

# a holding is "classified" only after the PLAYER has explicitly said Keep, Give Away, or Let the Office
# Decide. Newly acquired holdings therefore force the domain-planning wizard to ask before any automatic
# surplus grant.
ddm_has_unclassified_domain_holding_trigger = {
    any_held_title = {
        OR = {
            tier = tier_county
            AND = { tier = tier_barony title_province = { is_county_capital = no } }
        }
        this != root.capital_county
        NOT = {
            OR = {
                has_variable = ddm_manual_core
                has_variable = ddm_manual_shed
                has_variable = ddm_manual_neutral
            }
        }
    }
}

ddm_domain_plan_limit_changed_trigger = {
    OR = {
        NOT = { has_variable = ddm_domain_plan_limit }
        var:ddm_domain_plan_limit != domain_limit
    }
}

# the player's checked Personal Domain is the actual stopping target. The vanilla domain limit remains
# the maximum number the planner can protect, but selecting fewer holdings deliberately tells DDM to shed
# down to that count.
ddm_above_protected_domain_target_trigger = {
    OR = {
        AND = {
            has_variable = ddm_protected_domain_target
            domain_size > var:ddm_protected_domain_target
        }
        AND = {
            NOT = { has_variable = ddm_protected_domain_target }
            domain_size > domain_limit
        }
    }
}

ddm_domain_plan_target_mismatch_trigger = {
    OR = {
        NOT = { has_variable = ddm_protected_domain_target }
        var:ddm_dash_manual_core_count != var:ddm_protected_domain_target
    }
}

ddm_is_in_actor_realm_trigger = {

        this = scope:ddm_actor
        is_vassal_or_below_of = scope:ddm_actor
    }
}

ddm_is_eligible_land_recipient_trigger = {
    is_alive = yes
    this != scope:ddm_actor
    exists = dynasty
    dynasty = scope:ddm_actor.dynasty
    NOT = { has_character_flag = ddm_never_grant_land }
    NOT = { has_character_flag = ddm_protected_from_automation }
    NOT = { has_trait = incapable }
    NOT = { has_trait = devoted }
    NOT = { has_trait = gallivanter }

    can_be_granted_titles_by = { RULER = scope:ddm_actor }
    NOT = { is_at_war_with = scope:ddm_actor }
    trigger_if = {
        limit = { is_ruler = no }
        is_imprisoned = no
    }

    trigger_if = {
        limit = {
            scope:ddm_actor.var:ddm_gender_alignment = 1
            OR = {
                scope:ddm_actor = { has_realm_law = male_only_law }
                scope:ddm_actor.primary_title = { has_title_law = male_only_law }
                AND = { exists = scope:ddm_title scope:ddm_title = { has_title_law = male_only_law } }
            }
        }
        is_male = yes
    }
    trigger_if = {
        limit = {
            scope:ddm_actor.var:ddm_gender_alignment = 1
            OR = {
                scope:ddm_actor = { has_realm_law = female_only_law }
                scope:ddm_actor.primary_title = { has_title_law = female_only_law }
                AND = { exists = scope:ddm_title scope:ddm_title = { has_title_law = female_only_law } }
            }
        }
        is_female = yes
    }

    OR = {
        is_landed = no
        AND = { is_landed = yes highest_held_title_tier < scope:ddm_actor.highest_held_title_tier }
    }
    OR = { is_landed = no ddm_is_in_actor_realm_trigger = yes }

    # all legally usable dynasts remain candidates. Family-scope settings now change House preference
    # rather than hard-excluding branches.
}

ddm_is_eligible_outsider_recipient_trigger = {
    is_alive = yes
    this != scope:ddm_actor
    OR = {
        is_courtier_of = scope:ddm_actor
        is_vassal_or_below_of = scope:ddm_actor
    }
    OR = { NOT = { exists = dynasty } dynasty != scope:ddm_actor.dynasty }
    NOT = { has_character_flag = ddm_never_grant_land }
    NOT = { has_character_flag = ddm_protected_from_automation }
    NOT = { has_trait = incapable }
    NOT = { has_trait = devoted }
    NOT = { has_trait = gallivanter }
    can_be_granted_titles_by = { RULER = scope:ddm_actor }
    NOT = { is_at_war_with = scope:ddm_actor }

    trigger_if = {
        limit = {
            scope:ddm_actor.var:ddm_gender_alignment = 1
            OR = {
                scope:ddm_actor = { has_realm_law = male_only_law }
                scope:ddm_actor.primary_title = { has_title_law = male_only_law }
                AND = { exists = scope:ddm_title scope:ddm_title = { has_title_law = male_only_law } }
            }
        }
        is_male = yes
    }
    trigger_if = {
        limit = {
            scope:ddm_actor.var:ddm_gender_alignment = 1
            OR = {
                scope:ddm_actor = { has_realm_law = female_only_law }
                scope:ddm_actor.primary_title = { has_title_law = female_only_law }
                AND = { exists = scope:ddm_title scope:ddm_title = { has_title_law = female_only_law } }
            }
        }
        is_female = yes
    }

    OR = {
        is_landed = no
        highest_held_title_tier < scope:ddm_actor.highest_held_title_tier
    }
}

# Explicit title reservations may name either a legal dynast or a legal non-dynastic court/realm
# candidate regardless of the broad recipient-scope policy. The reservation itself is the player's
# one-title exception.
ddm_is_eligible_manual_reservation_recipient_trigger = {
    OR = {
        ddm_is_eligible_land_recipient_trigger = yes
        ddm_is_eligible_outsider_recipient_trigger = yes
    }
}

# Active ordinary pool: dynasts are always eligible under the existing rules; outsiders enter only under
# the player's explicit recipient-scope policy.
ddm_is_eligible_active_recipient_trigger = {
    OR = {
        ddm_is_eligible_land_recipient_trigger = yes
        AND = {
            ddm_is_eligible_outsider_recipient_trigger = yes
            OR = {
                scope:ddm_actor.var:ddm_outsider_recipient_mode = 2
                AND = {
                    scope:ddm_actor.var:ddm_outsider_recipient_mode = 1
                    has_variable = ddm_outsider_favored_for_land
                }
            }
        }
    }
}

# scope:ddm_actor, scope:ddm_transfer_vassal, and scope:ddm_proper_liege_candidate saved.
ddm_is_valid_proper_liege_target_trigger = {
    is_alive = yes
    is_landed = yes
    this != scope:ddm_actor
    liege = scope:ddm_actor
    highest_held_title_tier > scope:ddm_transfer_vassal.highest_held_title_tier
    vassal_count < vassal_limit
    NOT = { has_trait = gallivanter }

    scope:ddm_transfer_vassal = {
        NOT = { is_at_war_with = scope:ddm_proper_liege_candidate }
        NOT = { any_vassal_or_below = { is_at_war_with = scope:ddm_proper_liege_candidate } }
    }

    # Vanilla refuses to hand an administrative vassal to a non-administrative subordinate when the top
    # ruler is administrative.
    NAND = {
        scope:ddm_actor = { government_allows = administrative }
        NOT = { government_allows = administrative }
        scope:ddm_transfer_vassal = { government_allows = administrative }
    }

    # Match the vanilla Ritsuryo/Soryo transfer boundary.
    NAND = {
        scope:ddm_actor = { has_government = japan_administrative_government }
        has_government = japan_administrative_government
        scope:ddm_transfer_vassal = { has_government = japan_feudal_government }
    }

    # Noble-family titles are special administrative objects, not ordinary landed vassals to be shuffled
    # by the Domain Office.
    NAND = {
        scope:ddm_actor = { government_allows = administrative }
        scope:ddm_transfer_vassal = { any_held_title = { is_noble_family_title = yes } }
    }
}
