ddm_initialize_settings_effect = {
    if = { limit = { NOT = { has_variable = ddm_family_scope_mode } } set_variable = { name = ddm_family_scope_mode value = 3 } }
    if = { limit = { NOT = { has_variable = ddm_power_policy } } set_variable = { name = ddm_power_policy value = 2 } }
    if = { limit = { NOT = { has_variable = ddm_purge_method } } set_variable = { name = ddm_purge_method value = 2 } }
    if = { limit = { NOT = { has_variable = ddm_purge_stage } } set_variable = { name = ddm_purge_stage value = 3 } }
    if = { limit = { NOT = { has_variable = ddm_min_dynasty_floor } } set_variable = { name = ddm_min_dynasty_floor value = 8 } }
    if = { limit = { NOT = { has_variable = ddm_regard_mode } } set_variable = { name = ddm_regard_mode value = 1 } }
    if = {
        limit = { NOT = { has_variable = ddm_scoring_model_v14 } }
        if = { limit = { var:ddm_purge_stage >= 5 } set_variable = { name = ddm_purge_stage value = 3 } }
        set_variable = { name = ddm_scoring_model_v14 value = 1 }
    }
    if = { limit = { NOT = { has_variable = ddm_protect_player_heir } } set_variable = { name = ddm_protect_player_heir value = 1 } }
    # 1 = Absolute Core (legacy behavior), 2 = Best-Effort Core (default).
    if = { limit = { NOT = { has_variable = ddm_core_policy } } set_variable = { name = ddm_core_policy value = 2 } }
    # Domain-core mode: 1 = adaptive best-domain core (default), 2 = locked snapshot. The adaptive core
    # explicitly marks the best domain_limit holdings before any surplus grant.
    if = { limit = { NOT = { has_variable = ddm_domain_core_mode } } set_variable = { name = ddm_domain_core_mode value = 1 } }
    # Alliance policy: 0 = off, 1 = all dynastic landed recipients, 2 = direct vassals only, 3 = dukes
    # and above only. Migrate legacy saves that only knew the binary ddm_auto_alliance variable.
    if = {
        limit = { NOT = { has_variable = ddm_alliance_policy } }
        if = {
            limit = { has_variable = ddm_auto_alliance var:ddm_auto_alliance = 0 }
            set_variable = { name = ddm_alliance_policy value = 0 }
        }
        else = { set_variable = { name = ddm_alliance_policy value = 1 } }
    }
    # Succession influence: 1 = Neutral (default), 2 = Respect Current Heir, 3 = Preserve Primary Estate.
    # Succession modifies rightfulness only.
    if = { limit = { NOT = { has_variable = ddm_succession_policy } } set_variable = { name = ddm_succession_policy value = 1 } }
    if = {
        limit = { NOT = { has_variable = ddm_scoring_model_v17 } }
        set_variable = { name = ddm_scoring_model_v17 value = 1 }
    }
    # Align automatic grants with active succession gender law. Male/Female Only hard-filters recipients;
    # preference laws influence ranking rather than ban.
    if = { limit = { NOT = { has_variable = ddm_gender_alignment } } set_variable = { name = ddm_gender_alignment value = 1 } }
    # Realm-integrity heuristic penalizes recipients whose current primary heir is outside the player's
    # realm.
    if = { limit = { NOT = { has_variable = ddm_realm_integrity } } set_variable = { name = ddm_realm_integrity value = 1 } }
    # Construction reserve in gold. Bulk projects stop before crossing this reserve.
    if = { limit = { NOT = { has_variable = ddm_build_reserve } } set_variable = { name = ddm_build_reserve value = 1000 } }
    if = { limit = { NOT = { has_variable = ddm_outsider_policy } } set_variable = { name = ddm_outsider_policy value = 0 } }

    if = { limit = { NOT = { has_variable = ddm_outsider_recipient_mode } } set_variable = { name = ddm_outsider_recipient_mode value = 0 } }
    if = { limit = { NOT = { has_variable = ddm_educator_scope_mode } } set_variable = { name = ddm_educator_scope_mode value = 3 } }
    if = { limit = { NOT = { has_variable = ddm_auto_education_enabled } } set_variable = { name = ddm_auto_education_enabled value = 1 } }
    if = { limit = { NOT = { has_variable = ddm_recipient_page } } set_variable = { name = ddm_recipient_page value = 0 } }
    if = { limit = { NOT = { has_variable = ddm_outsider_page } } set_variable = { name = ddm_outsider_page value = 0 } }
    if = { limit = { NOT = { has_variable = ddm_review_page } } set_variable = { name = ddm_review_page value = 0 } }
    if = { limit = { NOT = { has_variable = ddm_review_sort_mode } } set_variable = { name = ddm_review_sort_mode value = 0 } }
    if = { limit = { NOT = { has_variable = ddm_review_sort_direction } } set_variable = { name = ddm_review_sort_direction value = 1 } }
    # Holding-list organization: 0 = de-jure hierarchy A-Z, 1 = development, 2 = county faith affinity, 3
    # = county culture affinity. Direction 1 is the natural order for the selected mode; -1 reverses it.
    if = { limit = { NOT = { has_variable = ddm_title_sort_mode } } set_variable = { name = ddm_title_sort_mode value = 0 } }
    if = { limit = { NOT = { has_variable = ddm_title_sort_direction } } set_variable = { name = ddm_title_sort_direction value = 1 } }

    if = {
        limit = { NOT = { has_variable = ddm_migrated_explicit_core_v18 } }
        set_variable = { name = ddm_migrated_explicit_core_v18 value = 1 }
        remove_variable = ddm_domain_plan_limit
        clear_variable_list = ddm_domain_review_queue
        every_held_title = {
            limit = {
                OR = {
                    tier = tier_county
                    AND = { tier = tier_barony is_capital_barony = no }
                }
            }
            remove_variable = ddm_manual_core
            remove_variable = ddm_manual_shed
            remove_variable = ddm_manual_neutral
        }
        if = {
            limit = { exists = capital_county }
            capital_county = { set_variable = ddm_manual_core }
        }
    }

    if = {
        limit = { NOT = { has_variable = ddm_succession_model_v19 } }
        set_variable = { name = ddm_succession_model_v19 value = 1 }
        set_variable = { name = ddm_succession_policy value = 1 }
        # This variable remains only for pruning protection compatibility. It no longer participates in
        # land-recipient eligibility or succession policy.
        set_variable = { name = ddm_protect_player_heir value = 1 }
    }

    if = {
        limit = { NOT = { has_variable = ddm_rightfulness_model_v110 } }
        set_variable = { name = ddm_rightfulness_model_v110 value = 1 }
        set_variable = { name = ddm_succession_policy value = 1 }
    }

    if = {
        limit = { NOT = { has_variable = ddm_distribution_target_model_v111 } }
        set_variable = { name = ddm_distribution_target_model_v111 value = 1 }
        set_variable = { name = ddm_protected_domain_target value = 0 }
        every_held_title = {
            limit = {
                has_variable = ddm_manual_core
                OR = {
                    tier = tier_county
                    AND = { tier = tier_barony is_capital_barony = no }
                }
            }
            holder = { change_variable = { name = ddm_protected_domain_target add = 1 } }
        }
        if = {
            limit = { var:ddm_protected_domain_target < 1 }
            set_variable = { name = ddm_protected_domain_target value = domain_limit }
            if = { limit = { domain_size < var:ddm_protected_domain_target } set_variable = { name = ddm_protected_domain_target value = domain_size } }
            if = { limit = { var:ddm_protected_domain_target < 1 } set_variable = { name = ddm_protected_domain_target value = 1 } }
        }
    }

    if = {
        limit = { NOT = { has_variable = ddm_domain_gui_model_v19 } }
        set_variable = { name = ddm_domain_gui_model_v19 value = 1 }
        remove_variable = ddm_domain_plan_limit
        clear_variable_list = ddm_domain_review_queue
        clear_variable_list = ddm_domain_planner_titles
        every_held_title = {
            limit = {
                OR = {
                    tier = tier_county
                    AND = { tier = tier_barony is_capital_barony = no }
                }
            }
            remove_variable = ddm_manual_core
            remove_variable = ddm_manual_shed
            remove_variable = ddm_manual_neutral
            remove_variable = ddm_planner_original_core
        }
        ddm_refresh_adaptive_domain_core_internal_effect = yes
        every_held_title = {
            limit = { has_variable = ddm_auto_core }
            set_variable = ddm_manual_core
        }
        ddm_mark_capital_as_explicit_core_effect = yes
    }
}

ddm_refresh_one_subject_regard_effect = {
    # Run from the subject. scope:ddm_actor must be saved.
    set_variable = { name = ddm_family_regard value = 0 }
    set_variable = { name = ddm_family_regard_count value = 0 }
    save_scope_as = ddm_subject

    if = {
        limit = { scope:ddm_actor.var:ddm_regard_mode = 2 }
        dynasty = {
            every_dynasty_member = {
                limit = { is_alive = yes this != scope:ddm_subject }
                save_temporary_scope_value_as = { name = ddm_observer_opinion value = "opinion(scope:ddm_subject)" }
                scope:ddm_subject = {
                    change_variable = { name = ddm_family_regard add = scope:ddm_observer_opinion }
                    change_variable = { name = ddm_family_regard_count add = 1 }
                }
            }
        }
    }
    else = {
        every_close_or_extended_family_member = {
            limit = { is_alive = yes exists = dynasty dynasty = scope:ddm_actor.dynasty }
            save_temporary_scope_value_as = { name = ddm_observer_opinion value = "opinion(scope:ddm_subject)" }
            scope:ddm_subject = {
                change_variable = { name = ddm_family_regard add = scope:ddm_observer_opinion }
                change_variable = { name = ddm_family_regard_count add = 1 }
            }
        }
        if = {
            limit = { exists = house.house_head house.house_head != this }
            house.house_head = { save_temporary_scope_value_as = { name = ddm_observer_opinion value = "opinion(scope:ddm_subject)" } }
            change_variable = { name = ddm_family_regard add = scope:ddm_observer_opinion }
            change_variable = { name = ddm_family_regard_count add = 1 }
        }
        if = {
            limit = { exists = dynasty.dynasty_head dynasty.dynasty_head != this }
            dynasty.dynasty_head = { save_temporary_scope_value_as = { name = ddm_observer_opinion value = "opinion(scope:ddm_subject)" } }
            change_variable = { name = ddm_family_regard add = scope:ddm_observer_opinion }
            change_variable = { name = ddm_family_regard_count add = 1 }
        }
    }
    if = {
        limit = { var:ddm_family_regard_count > 0 }
        change_variable = { name = ddm_family_regard divide = var:ddm_family_regard_count }
    }
}

ddm_refresh_all_dynasty_regard_effect = {
    save_scope_as = ddm_actor
    dynasty = {
        every_dynasty_member = {
            limit = { is_alive = yes }
            ddm_refresh_one_subject_regard_effect = yes
        }
    }
}

ddm_build_partition_settlement_pool_effect = {
    save_scope_as = ddm_actor
    set_variable = { name = ddm_partition_pool_count value = 0 }
    dynasty = { every_dynasty_member = { remove_character_flag = ddm_partition_settlement_pool } }

    if = {
        limit = { ddm_actor_uses_partition_trigger = yes }

        if = {
            limit = { OR = { var:ddm_succession_policy = 1 var:ddm_succession_policy = 3 } }
            every_child = {
                limit = {
                    ddm_is_eligible_partition_recipient_trigger = yes
                    trigger_if = {
                        limit = {
                            OR = {
                                scope:ddm_actor = { has_realm_law = male_preference_law }
                                scope:ddm_actor.primary_title = { has_title_law = male_preference_law }
                            }
                            is_female = yes
                        }
                        scope:ddm_actor = { NOT = { any_child = { is_male = yes ddm_is_eligible_partition_recipient_trigger = yes } } }
                    }
                    trigger_if = {
                        limit = {
                            OR = {
                                scope:ddm_actor = { has_realm_law = female_preference_law }
                                scope:ddm_actor.primary_title = { has_title_law = female_preference_law }
                            }
                            is_male = yes
                        }
                        scope:ddm_actor = { NOT = { any_child = { is_female = yes ddm_is_eligible_partition_recipient_trigger = yes } } }
                    }
                }
                add_character_flag = ddm_partition_settlement_pool
                scope:ddm_actor = { change_variable = { name = ddm_partition_pool_count add = 1 } }
            }
        }

        # Policy 2: literal current succession. Deduplicate title current heirs.
        else_if = {
            limit = { var:ddm_succession_policy = 2 }
            every_held_title = {
                limit = { tier >= tier_county exists = current_heir }
                current_heir = {
                    if = {
                        limit = {
                            ddm_is_eligible_partition_recipient_trigger = yes
                            NOT = { has_character_flag = ddm_partition_settlement_pool }
                        }
                        add_character_flag = ddm_partition_settlement_pool
                        scope:ddm_actor = { change_variable = { name = ddm_partition_pool_count add = 1 } }
                    }
                }
            }
        }

        # Unusual representation/succession fallback.
        if = {
            limit = { var:ddm_partition_pool_count = 0 }
            every_heir = {
                limit = {
                    ddm_is_eligible_partition_recipient_trigger = yes
                    NOT = { has_character_flag = ddm_partition_settlement_pool }
                }
                add_character_flag = ddm_partition_settlement_pool
                scope:ddm_actor = { change_variable = { name = ddm_partition_pool_count add = 1 } }
            }
        }
    }
}

# Prepare a mass land-grant run. The family-regard cache is refreshed before ranking begins, then a
# per-batch grant counter discourages repeatedly enriching the same branch unless merit, succession, or
# de-jure continuity still wins.
ddm_prepare_land_distribution_batch_effect = {
    ddm_initialize_settings_effect = yes
    save_scope_as = ddm_actor
    # Clean all temporary per-batch fairness state first.
    dynasty = {
        every_dynasty_member = {
            limit = { is_alive = yes }
            remove_variable = ddm_batch_land_grants
        }
    }
    every_vassal_or_below = { remove_variable = ddm_batch_land_grants }

    ddm_build_partition_settlement_pool_effect = yes
    set_variable = { name = ddm_last_partition_heirs_available value = var:ddm_partition_pool_count }
    ddm_refresh_all_dynasty_regard_effect = yes
}

ddm_clear_land_distribution_batch_effect = {
    # Temporary candidate markers must never leak into a later ruler/batch.
    save_scope_as = ddm_actor
    dynasty = {
        every_dynasty_member = {
            remove_character_flag = ddm_partition_settlement_pool
        }
    }
}

# Direct scripted title changes bypass the special Grant Titles interaction's on_accept block. Reapply
# the vanilla recipient-opinion modifier for the title tier and record this recipient for fairness within
# the current mass-grant run.
ddm_apply_grant_social_effect = {
    # Run from actor with scope:ddm_recipient and scope:ddm_title saved.
    if = {
        limit = { exists = scope:ddm_recipient exists = scope:ddm_title }
        scope:ddm_recipient = {
            if = {
                limit = { scope:ddm_title = { tier = tier_barony } }
                hidden_effect = { add_opinion = { target = scope:ddm_actor modifier = received_title_barony } }
            }
            else_if = {
                limit = { scope:ddm_title.tier = tier_county }
                hidden_effect = { add_opinion = { target = scope:ddm_actor modifier = received_title_county } }
            }
            else_if = {
                limit = { scope:ddm_title.tier = tier_duchy }
                hidden_effect = { add_opinion = { target = scope:ddm_actor modifier = received_title_duchy } }
            }
            else_if = {
                limit = { scope:ddm_title.tier = tier_kingdom }
                hidden_effect = { add_opinion = { target = scope:ddm_actor modifier = received_title_kingdom } }
            }
            else_if = {
                limit = { scope:ddm_title.tier = tier_empire }
                hidden_effect = { add_opinion = { target = scope:ddm_actor modifier = received_title_empire } }
            }
            if = { limit = { NOT = { has_variable = ddm_batch_land_grants } } set_variable = { name = ddm_batch_land_grants value = 0 } }
            change_variable = { name = ddm_batch_land_grants add = 1 }
        }
    }
}

ddm_refresh_adaptive_domain_core_internal_effect = {
    save_scope_as = ddm_actor

    every_held_title = {
        limit = {
            OR = {
                tier = tier_county
                AND = { tier = tier_barony title_province = { is_county_capital = no } }
            }
        }
        remove_variable = ddm_auto_core
    }

    if = {
        limit = { domain_limit > 0 }
        ordered_held_title = {
            limit = {
                OR = {
                    tier = tier_county
                    AND = { tier = tier_barony title_province = { is_county_capital = no } }
                }
                NOT = { has_variable = ddm_manual_shed }
            }
            order_by = ddm_holding_retention_priority
            max = ddm_core_last_index
            check_range_bounds = no
            set_variable = ddm_auto_core
        }
    }
}

ddm_refresh_adaptive_domain_core_effect = {
    ddm_initialize_settings_effect = yes
    ddm_refresh_adaptive_domain_core_internal_effect = yes
}

ddm_lock_recommended_domain_core_effect = {
    # Freeze the current adaptive recommendation. Only the top domain_limit holdings are locked, so using
    # this while above the limit does not accidentally protect the whole over-limit domain.
    set_variable = { name = ddm_domain_core_mode value = 1 }
    ddm_refresh_adaptive_domain_core_effect = yes
    every_held_title = {
        limit = { has_variable = ddm_manual_core }
        remove_variable = ddm_manual_core
    }
    every_held_title = {
        limit = { has_variable = ddm_auto_core }
        set_variable = ddm_manual_core
        remove_variable = ddm_manual_shed
        remove_variable = ddm_manual_neutral
        remove_variable = ddm_auto_core
    }
    set_variable = { name = ddm_domain_core_mode value = 2 }
}

}

ddm_continue_domain_choice_review_effect = {
    # Called after one KEEP/GIVE/DELEGATE choice. The current title has already been removed from
    # ddm_domain_review_queue by the event option.
    ddm_prepare_dashboard_effect = yes

    # If the player has filled every personal-domain slot with explicit KEEP choices, everything left in
    # the review queue is necessarily surplus.
    if = {
        limit = {
            domain_size > domain_limit
            var:ddm_dash_manual_core_count >= domain_limit
        }
        every_in_list = {
            variable = ddm_domain_review_queue
            set_variable = ddm_manual_shed
            remove_variable = ddm_manual_core
            remove_variable = ddm_manual_neutral
        }
        clear_variable_list = ddm_domain_review_queue
        ddm_finish_domain_choice_review_effect = yes
        trigger_event = ddm_manager.1022
    }
    else_if = {
        limit = {
            any_in_list = {
                variable = ddm_domain_review_queue
                count >= 1
            }
        }
        trigger_event = ddm_manager.1021
    }
    else = {

        limit = {
            exists = scope:ddm_recipient
            scope:ddm_recipient = { is_alive = yes is_landed = yes this != scope:ddm_actor }
            var:ddm_alliance_policy > 0
        }
        if = {
            limit = { var:ddm_alliance_policy = 1 }
            set_variable = { name = ddm_alliance_allowed value = 1 }

            limit = {

    }

ddm_v18_build_recipient_lists_effect = {
    save_scope_as = ddm_actor
    set_variable = { name = ddm_last_partition_heirs_available value = 0 }
    set_variable = { name = ddm_last_general_recipients_available value = 0 }

    # Ordinary dynasty pool. This retains vanilla's broad Grant Titles gate. Succession does not replace
    # this pool; separately tracked heirs exist only for score modifiers and the narrow vanilla
    # grant-gate bypass.
    dynasty = {
        every_dynasty_member = {
            limit = { ddm_is_eligible_land_recipient_trigger = yes }
            add_to_temporary_list = ddm_v18_recipient_pool
            scope:ddm_actor = { change_variable = { name = ddm_last_general_recipients_available add = 1 } }
        }
    }

    # Optional non-dynastic pool. Courtiers and existing vassals are merged into one concrete list. The
    # transient marker prevents a landed courtier/vassal from being inserted twice by overlapping
    # iterators.
    every_courtier = { remove_character_flag = ddm_recipient_listed_this_pass }
    every_vassal_or_below = { remove_character_flag = ddm_recipient_listed_this_pass }

    if = {
        limit = { var:ddm_outsider_recipient_mode = 1 }
        every_courtier = {
            limit = {
                ddm_is_eligible_outsider_recipient_trigger = yes
                has_variable = ddm_outsider_favored_for_land
                NOT = { has_character_flag = ddm_recipient_listed_this_pass }
            }
            add_character_flag = ddm_recipient_listed_this_pass
            add_to_temporary_list = ddm_v18_recipient_pool
            scope:ddm_actor = { change_variable = { name = ddm_last_general_recipients_available add = 1 } }
        }
        every_vassal_or_below = {
            limit = {
                ddm_is_eligible_outsider_recipient_trigger = yes
                has_variable = ddm_outsider_favored_for_land
                NOT = { has_character_flag = ddm_recipient_listed_this_pass }
            }
            add_character_flag = ddm_recipient_listed_this_pass
            add_to_temporary_list = ddm_v18_recipient_pool
            scope:ddm_actor = { change_variable = { name = ddm_last_general_recipients_available add = 1 } }
        }
    }
    else_if = {
        limit = { var:ddm_outsider_recipient_mode = 2 }
        every_courtier = {
            limit = { ddm_is_eligible_outsider_recipient_trigger = yes NOT = { has_character_flag = ddm_recipient_listed_this_pass } }
            add_character_flag = ddm_recipient_listed_this_pass
            add_to_temporary_list = ddm_v18_recipient_pool
            scope:ddm_actor = { change_variable = { name = ddm_last_general_recipients_available add = 1 } }
        }
        every_vassal_or_below = {
            limit = { ddm_is_eligible_outsider_recipient_trigger = yes NOT = { has_character_flag = ddm_recipient_listed_this_pass } }
            add_character_flag = ddm_recipient_listed_this_pass
            add_to_temporary_list = ddm_v18_recipient_pool
            scope:ddm_actor = { change_variable = { name = ddm_last_general_recipients_available add = 1 } }
        }
    }

    if = {
        limit = { ddm_actor_uses_partition_trigger = yes }

        # Track direct legal children for inheritance context and the narrow vanilla grant-gate bypass.
        # This is never the main allocation universe.
        if = {
            limit = { OR = { var:ddm_succession_policy = 1 var:ddm_succession_policy = 3 } }
            every_child = {
                limit = {
                    ddm_is_eligible_partition_recipient_trigger = yes

                    # Under preference laws, the non-preferred sex enters only if no otherwise-eligible
                    # child of the preferred sex exists.
                    trigger_if = {
                        limit = {
                            NOR = {
                                scope:ddm_actor = { has_realm_law = male_only_law }
                                scope:ddm_actor.primary_title = { has_title_law = male_only_law }
                                scope:ddm_actor = { has_realm_law = female_only_law }
                                scope:ddm_actor.primary_title = { has_title_law = female_only_law }
                            }
                            OR = {
                                scope:ddm_actor = { has_realm_law = male_preference_law }
                                scope:ddm_actor.primary_title = { has_title_law = male_preference_law }
                            }
                            is_female = yes
                        }
                        scope:ddm_actor = {
                            NOT = { any_child = { is_male = yes ddm_is_eligible_partition_recipient_trigger = yes } }
                        }
                    }
                    trigger_if = {
                        limit = {
                            NOR = {
                                scope:ddm_actor = { has_realm_law = male_only_law }
                                scope:ddm_actor.primary_title = { has_title_law = male_only_law }
                                scope:ddm_actor = { has_realm_law = female_only_law }
                                scope:ddm_actor.primary_title = { has_title_law = female_only_law }
                            }
                            OR = {
                                scope:ddm_actor = { has_realm_law = female_preference_law }
                                scope:ddm_actor.primary_title = { has_title_law = female_preference_law }
                            }
                            is_male = yes
                        }
                        scope:ddm_actor = {
                            NOT = { any_child = { is_female = yes ddm_is_eligible_partition_recipient_trigger = yes } }
                        }
                    }
                }
                add_character_flag = ddm_partition_settlement_pool
                add_to_temporary_list = ddm_v18_partition_pool
                scope:ddm_actor = { change_variable = { name = ddm_last_partition_heirs_available add = 1 } }
            }
        }

        # Policy 2: Respect Current Heir. Track each held title's current_heir so that character can
        # receive the modest title-specific legitimacy bonus.
        else_if = {
            limit = { var:ddm_succession_policy = 2 }
            every_held_title = {
                limit = { tier >= tier_county exists = current_heir }
                current_heir = {
                    if = {
                        limit = {
                            ddm_is_eligible_partition_recipient_trigger = yes
                            NOT = { has_character_flag = ddm_partition_settlement_pool }
                        }
                        add_character_flag = ddm_partition_settlement_pool
                        add_to_temporary_list = ddm_v18_partition_pool
                        scope:ddm_actor = { change_variable = { name = ddm_last_partition_heirs_available add = 1 } }
                    }
                }
            }
        }

        # Representation/unusual succession fallback.
        if = {
            limit = { var:ddm_last_partition_heirs_available = 0 }
            every_heir = {
                limit = { ddm_is_eligible_partition_recipient_trigger = yes }
                add_character_flag = ddm_partition_settlement_pool
                add_to_temporary_list = ddm_v18_partition_pool
                scope:ddm_actor = { change_variable = { name = ddm_last_partition_heirs_available add = 1 } }
            }
        }
    }

    # Family regard is refreshed for both pools because a legal partition child may intentionally bypass
    # the ordinary vanilla-recipient gate.
    every_in_list = { list = ddm_v18_recipient_pool ddm_refresh_one_subject_regard_effect = yes }
    every_in_list = { list = ddm_v18_partition_pool ddm_refresh_one_subject_regard_effect = yes }
    every_courtier = { remove_character_flag = ddm_recipient_listed_this_pass }
    every_vassal_or_below = { remove_character_flag = ddm_recipient_listed_this_pass }
}

ddm_rank_dynasty_effect = {
    save_scope_as = ddm_actor
    ddm_refresh_all_dynasty_regard_effect = yes

    dynasty = {
        every_dynasty_member = {
            limit = { is_alive = yes }
            remove_variable = ddm_rank
            remove_variable = ddm_rank_quota
            remove_character_flag = ddm_rightful_pool
        }
    }
    every_courtier = { remove_character_flag = ddm_rightful_pool }
    every_vassal_or_below = { remove_character_flag = ddm_rightful_pool }

    # every eligible dynast is ranked. No bottom-quarter purge.
    set_variable = { name = ddm_ranked_total value = 0 }
    dynasty = {
        every_dynasty_member = {
            limit = {
                is_alive = yes
                this != scope:ddm_actor
                ddm_is_rankable_dynast_trigger = yes
            }
            scope:ddm_actor = { change_variable = { name = ddm_ranked_total add = 1 } }
        }
    }
    set_variable = { name = ddm_ranked_kept value = var:ddm_ranked_total }

    set_variable = { name = ddm_rank_cursor value = 0 }
    while = {
        limit = { var:ddm_rank_cursor < var:ddm_ranked_total }
        change_variable = { name = ddm_rank_cursor add = 1 }
        remove_variable = ddm_rank_assigned
        dynasty = {
            ordered_dynasty_member = {
                limit = {
                    is_alive = yes
                    this != scope:ddm_actor
                    ddm_is_rankable_dynast_trigger = yes
                    NOT = { has_variable = ddm_rank }
                }
                order_by = ddm_rightful_ruler_value
                position = 0
                check_range_bounds = no
                set_variable = { name = ddm_rank value = scope:ddm_actor.var:ddm_rank_cursor }
                add_character_flag = ddm_rightful_pool
                scope:ddm_actor = { set_variable = { name = ddm_rank_assigned value = 1 } }
            }
        }
        break_if = { NOT = { has_variable = ddm_rank_assigned } }
    }

    set_variable = { name = ddm_ranked_kept value = var:ddm_rank_cursor }

    # Non-dynasty candidates do not receive a dynastic rank number, but they do enter the same rightful
    # pool so title-specific scoring and hierarchy can compare them directly with dynasts.
    every_in_list = {
        list = ddm_v18_recipient_pool
        limit = { OR = { NOT = { exists = dynasty } dynasty != scope:ddm_actor.dynasty } }
        add_character_flag = ddm_rightful_pool
    }

    remove_variable = ddm_rank_cursor
    remove_variable = ddm_rank_assigned
}

ddm_v18_choose_recipient_effect = {
    remove_variable = ddm_found_recipient
    set_variable = { name = ddm_last_recipient_search_stage value = 0 }

    # A direct title reservation is processed before every scored path. It is an explicit one-title
    # instruction, so it may name a legal dynast, courtier or existing vassal even when the broad
    # recipient policy is Dynasty Only.
    if = {
        limit = {
            exists = scope:ddm_title
            scope:ddm_title = {
                exists = var:ddm_reserved_recipient
                var:ddm_reserved_recipient = { ddm_is_eligible_manual_reservation_recipient_trigger = yes }
            }
        }
        scope:ddm_title = { var:ddm_reserved_recipient = { save_scope_as = ddm_recipient } }
        set_variable = { name = ddm_found_recipient value = 1 }
        set_variable = { name = ddm_last_recipient_search_stage value = 1 }
    }
    else_if = {
        limit = { exists = scope:ddm_title scope:ddm_title = { exists = var:ddm_reserved_recipient } }
        # Stale reservations should never strand settlement. If the named person is dead, has left
        # court/realm, or is no longer a legal grantee, clear the stale instruction and continue through
        # ordinary scoring.
        scope:ddm_title = { remove_variable = ddm_reserved_recipient }
    }

    # Primary path: rank the complete active recipient pool for THIS title. The score is recalculated
    # after every grant, so territorial state and the player's selected-outsider first-pass priorities
    # update immediately.
    if = {
        limit = { NOT = { has_variable = ddm_found_recipient } any_in_list = { list = ddm_v18_recipient_pool count >= 1 } }
        ordered_in_list = {
            list = ddm_v18_recipient_pool
            limit = { ddm_rank_quota_remaining_trigger = yes }
            order_by = ddm_rightful_title_recipient_value
            position = 0
            check_range_bounds = no
            save_scope_as = ddm_recipient
            scope:ddm_actor = {
                set_variable = { name = ddm_found_recipient value = 1 }
                set_variable = { name = ddm_last_recipient_search_stage value = 5 }
            }
        }
    }

    # Conservative whole-dynasty legal fallback.
    if = {
        limit = { NOT = { has_variable = ddm_found_recipient } }
        ordered_in_list = {
            list = ddm_v18_recipient_pool
            limit = {
                trigger_if = {
                    limit = { scope:ddm_title = { tier = tier_county } }
                    ddm_safe_for_additional_county_trigger = yes
                }
                trigger_if = {
                    limit = { scope:ddm_title = { tier = tier_barony } }
                    ddm_is_eligible_active_recipient_trigger = yes
                }
            }
            order_by = ddm_rightful_title_recipient_value
            position = 0
            check_range_bounds = no
            save_scope_as = ddm_recipient
            scope:ddm_actor = {
                set_variable = { name = ddm_found_recipient value = 1 }
                set_variable = { name = ddm_last_recipient_search_stage value = 2 }
            }
        }
    }

    # Partition bypass comes LAST. It exists only for heirs hidden by vanilla's grant-interaction
    # restriction; it never turns succession into the pool.
    if = {
        limit = {
            NOT = { has_variable = ddm_found_recipient }
            any_in_list = { list = ddm_v18_partition_pool count >= 1 }
        }
        ordered_in_list = {
            list = ddm_v18_partition_pool
            limit = { ddm_safe_partition_candidate_trigger = yes }
            order_by = ddm_rightful_title_recipient_value
            position = 0
            check_range_bounds = no
            save_scope_as = ddm_recipient
            scope:ddm_actor = {
                set_variable = { name = ddm_found_recipient value = 1 }
                set_variable = { name = ddm_last_recipient_search_stage value = 3 }
            }
        }
    }

    # Mechanical last resort from the already validated dynasty pool.
    if = {
        limit = { NOT = { has_variable = ddm_found_recipient } }
        random_in_list = {
            list = ddm_v18_recipient_pool
            limit = {
                trigger_if = {
                    limit = { scope:ddm_title = { tier = tier_county } }
                    ddm_safe_for_additional_county_trigger = yes
                }
                trigger_if = {
                    limit = { scope:ddm_title = { tier = tier_barony } }
                    ddm_is_eligible_active_recipient_trigger = yes
                }
            }
            save_scope_as = ddm_recipient
            scope:ddm_actor = {
                set_variable = { name = ddm_found_recipient value = 1 }
                set_variable = { name = ddm_last_recipient_search_stage value = 4 }
            }
        }
    }
}

ddm_v18_transfer_selected_title_effect = {
    # Run from actor with scope:ddm_title and scope:ddm_recipient saved.
    remove_variable = ddm_grant_success

    if = {
        limit = {
            exists = scope:ddm_title
            exists = scope:ddm_recipient
            scope:ddm_title = { holder = scope:ddm_actor }
        }

        clear_saved_scope = ddm_v18_title_change
        create_title_and_vassal_change = {
            type = granted
            save_scope_as = ddm_v18_title_change
            add_claim_on_loss = yes
        }
        scope:ddm_title = {
            change_title_holder = {
                holder = scope:ddm_recipient
                change = scope:ddm_v18_title_change
                government_base = scope:ddm_actor
            }
        }
        resolve_title_and_vassal_change = scope:ddm_v18_title_change

        if = {
            limit = {
                scope:ddm_title = { holder = scope:ddm_recipient }
            }

            # If the new holder did not automatically become a vassal below the actor, perform a separate
            # liege-change transaction.
            if = {
                limit = {
                    scope:ddm_recipient = {
                        is_landed = yes
                        NOT = { is_vassal_or_below_of = scope:ddm_actor }
                        highest_held_title_tier < scope:ddm_actor.highest_held_title_tier
                    }
                }
                clear_saved_scope = ddm_v18_liege_change
                create_title_and_vassal_change = {
                    type = granted
                    save_scope_as = ddm_v18_liege_change
                    add_claim_on_loss = yes
                }
                scope:ddm_recipient = {
                    change_liege = {
                        liege = scope:ddm_actor
                        change = scope:ddm_v18_liege_change
                    }
                }
                resolve_title_and_vassal_change = scope:ddm_v18_liege_change
            }

            scope:ddm_title = {
                remove_variable = ddm_manual_shed
                remove_variable = ddm_manual_neutral
                remove_variable = ddm_auto_core
                remove_variable = ddm_reserved_recipient
            }

            if = {
                limit = { scope:ddm_title = { tier = tier_barony } }
                change_variable = { name = ddm_last_baronies_granted add = 1 }
            }
            else_if = {
                limit = { scope:ddm_title = { tier = tier_county } }
                change_variable = { name = ddm_last_counties_granted add = 1 }
            }

            set_variable = { name = ddm_grant_success value = 1 }
            ddm_apply_grant_social_effect = yes
            ddm_make_dynastic_alliance_effect = yes
        }
    }
}

ddm_distribute_surplus_counties_effect = {
    ddm_initialize_settings_effect = yes
    save_scope_as = ddm_actor
    clear_saved_scope = ddm_title
    clear_saved_scope = ddm_recipient

    ddm_refresh_adaptive_domain_core_effect = yes

    dynasty = {
        every_dynasty_member = {
            limit = { is_alive = yes }
            remove_variable = ddm_batch_land_grants
            remove_character_flag = ddm_partition_settlement_pool
        }
    }
    every_courtier = { remove_variable = ddm_batch_land_grants }
    every_vassal_or_below = { remove_variable = ddm_batch_land_grants }

    set_variable = { name = ddm_last_baronies_granted value = 0 }
    set_variable = { name = ddm_last_counties_granted value = 0 }
    set_variable = { name = ddm_last_titles_left value = 0 }
    set_variable = { name = ddm_last_baronies_left value = 0 }
    set_variable = { name = ddm_last_distribution_stalled value = 0 }
    set_variable = { name = ddm_last_recipient_search_stage value = 0 }
    set_variable = { name = ddm_last_distribution_start_surplus value = domain_size }
    change_variable = { name = ddm_last_distribution_start_surplus subtract = var:ddm_protected_domain_target }
    if = {
        limit = { var:ddm_last_distribution_start_surplus < 0 }
        set_variable = { name = ddm_last_distribution_start_surplus value = 0 }
    }

    # These effects populate temporary lists in the SAME unbroken effect chain.
    ddm_v18_build_recipient_lists_effect = yes
    ddm_v18_build_surplus_title_list_effect = yes

    # Rank the entire eligible dynasty before any title moves. the current model has no artificial 15%
    # cap or equal-share quota; concentration is controlled by live domain/power state and title-specific
    # rightfulness.
    ddm_rank_dynasty_effect = yes

    remove_variable = ddm_v18_loop_exhausted

    while = {
        limit = {
            ddm_above_protected_domain_target_trigger = yes
            any_in_list = {
                list = ddm_v18_surplus_titles
                count >= 1
            }
        }

        remove_variable = ddm_grant_success
        remove_variable = ddm_found_title
        remove_variable = ddm_found_recipient
        clear_saved_scope = ddm_title
        clear_saved_scope = ddm_recipient

        # Prefer the player's explicit GIVE AWAY orders, then secondary baronies, then the least valuable
        # delegated county.
        ordered_in_list = {
            list = ddm_v18_surplus_titles
            limit = {
                holder = scope:ddm_actor
                OR = {
                    AND = {
                        tier = tier_barony
                        is_capital_barony = no
                        ddm_barony_can_be_distributed_trigger = yes
                    }
                    AND = {
                        tier = tier_county
                        is_titular = no
                        ddm_county_can_be_distributed_trigger = yes
                    }
                }
            }
            order_by = ddm_v18_title_shed_priority
            position = 0
            check_range_bounds = no
            save_scope_as = ddm_title
            scope:ddm_actor = { set_variable = { name = ddm_found_title value = 1 } }
        }

        # Mechanical fallback: if weighted ordering returns nothing, take a literal member of the already
        # validated list.
        if = {
            limit = { NOT = { has_variable = ddm_found_title } }
            random_in_list = {
                list = ddm_v18_surplus_titles
                limit = {
                    holder = scope:ddm_actor
                    OR = {
                        AND = {
                            tier = tier_barony
                            is_capital_barony = no
                            ddm_barony_can_be_distributed_trigger = yes
                        }
                        AND = {
                            tier = tier_county
                            is_titular = no
                            ddm_county_can_be_distributed_trigger = yes
                        }
                    }
                }
                save_scope_as = ddm_title
                scope:ddm_actor = { set_variable = { name = ddm_found_title value = 1 } }
            }
        }

        if = {
            limit = {
                has_variable = ddm_found_title
                exists = scope:ddm_title
            }
            ddm_v18_choose_recipient_effect = yes
        }

        if = {
            limit = {
                has_variable = ddm_found_recipient
                exists = scope:ddm_recipient
                exists = scope:ddm_title
            }
            ddm_v18_transfer_selected_title_effect = yes
        }

        if = {
            limit = { has_variable = ddm_grant_success }
            # remove_from_list removes CURRENT scope, so deliberately switch back to the exact title that
            # was just transferred.
            scope:ddm_title = {
                remove_from_list = ddm_v18_surplus_titles
            }
        }
        else = {
            set_variable = { name = ddm_last_distribution_stalled value = 1 }
            # A title that could not be placed this iteration is retired from the queue so the run
            # continues with the next title instead of aborting the whole distribution.
            if = {
                limit = {
                    has_variable = ddm_found_title
                    exists = scope:ddm_title
                }
                scope:ddm_title = {
                    remove_from_list = ddm_v18_surplus_titles
                }
            }
        }

        if = {
            limit = { NOT = { has_variable = ddm_found_title } }
            set_variable = { name = ddm_v18_loop_exhausted value = 1 }
        }

        break_if = { has_variable = ddm_v18_loop_exhausted }
    }

    # A transient rejected title is not a stalled settlement if the player still reached the chosen
    # protected-domain target through other grants.
    if = { limit = { NOT = { ddm_above_protected_domain_target_trigger = yes } } set_variable = { name = ddm_last_distribution_stalled value = 0 } }

    if = {
        limit = { ddm_above_protected_domain_target_trigger = yes }
        every_in_list = {
            list = ddm_v18_surplus_titles
            limit = { holder = scope:ddm_actor }
            if = {
                limit = { tier = tier_barony }
                scope:ddm_actor = { change_variable = { name = ddm_last_baronies_left add = 1 } }
            }
            else_if = {
                limit = { tier = tier_county }
                scope:ddm_actor = { change_variable = { name = ddm_last_titles_left add = 1 } }
            }
        }
    }

    dynasty = {
        every_dynasty_member = {
            remove_character_flag = ddm_partition_settlement_pool
            remove_variable = ddm_batch_land_grants
        }
    }
    every_vassal_or_below = { remove_variable = ddm_batch_land_grants }

    remove_variable = ddm_grant_success
    remove_variable = ddm_found_title
    remove_variable = ddm_found_recipient
    remove_variable = ddm_v18_loop_exhausted
    remove_variable = ddm_batch_grant_cap
    ddm_clear_dynasty_ranking_effect = yes

    # "Distribute" is a complete settlement command, not county-only. Once the personal domain reaches
    # its selected target, place duchies and kingdoms into the dynastic hierarchy and then queue one
    # alliance event.
    ddm_organize_dynastic_hierarchy_effect = yes
    ddm_schedule_post_settlement_alliance_event_effect = yes

    clear_saved_scope = ddm_title
    clear_saved_scope = ddm_recipient
    clear_saved_scope = ddm_v18_title_change
    clear_saved_scope = ddm_v18_liege_change
}


ddm_transfer_one_direct_vassal_to_proper_liege_effect = {
    save_scope_as = ddm_transfer_vassal
    clear_saved_scope = ddm_proper_liege
    clear_saved_scope = ddm_proper_liege_candidate
    clear_saved_scope = ddm_dejure_level_1
    clear_saved_scope = ddm_dejure_level_2
    clear_saved_scope = ddm_dejure_level_3
    scope:ddm_actor = { remove_variable = ddm_proper_liege_resolved }

    # De-jure level 1 (county -> duchy, duchy -> kingdom, kingdom -> empire).
    if = {
        limit = { primary_title = { exists = de_jure_liege } }
        primary_title.de_jure_liege = { save_scope_as = ddm_dejure_level_1 }
        if = {
            limit = { scope:ddm_dejure_level_1 = { exists = holder holder = scope:ddm_actor } }
            scope:ddm_actor = { set_variable = { name = ddm_proper_liege_resolved value = 1 } }
        }
        else_if = {
            limit = { scope:ddm_dejure_level_1 = { exists = holder } }
            scope:ddm_dejure_level_1.holder = { save_scope_as = ddm_proper_liege_candidate }
            if = {
                limit = { scope:ddm_proper_liege_candidate = { ddm_is_valid_proper_liege_target_trigger = yes } }
                scope:ddm_proper_liege_candidate = { save_scope_as = ddm_proper_liege }
                scope:ddm_actor = { set_variable = { name = ddm_proper_liege_resolved value = 1 } }
            }
            clear_saved_scope = ddm_proper_liege_candidate
        }
    }

    # De-jure level 2. Only search higher when the nearer held title did not already establish the proper
    # reporting line.
    if = {
        limit = {
            scope:ddm_actor = { NOT = { has_variable = ddm_proper_liege_resolved } }
            exists = scope:ddm_dejure_level_1
            scope:ddm_dejure_level_1 = { exists = de_jure_liege }
        }
        scope:ddm_dejure_level_1.de_jure_liege = { save_scope_as = ddm_dejure_level_2 }
        if = {
            limit = { scope:ddm_dejure_level_2 = { exists = holder holder = scope:ddm_actor } }
            scope:ddm_actor = { set_variable = { name = ddm_proper_liege_resolved value = 1 } }
        }
        else_if = {
            limit = { scope:ddm_dejure_level_2 = { exists = holder } }
            scope:ddm_dejure_level_2.holder = { save_scope_as = ddm_proper_liege_candidate }
            if = {
                limit = { scope:ddm_proper_liege_candidate = { ddm_is_valid_proper_liege_target_trigger = yes } }
                scope:ddm_proper_liege_candidate = { save_scope_as = ddm_proper_liege }
                scope:ddm_actor = { set_variable = { name = ddm_proper_liege_resolved value = 1 } }
            }
            clear_saved_scope = ddm_proper_liege_candidate
        }
    }

    # De-jure level 3 (the farthest possible step for a county).
    if = {
        limit = {
            scope:ddm_actor = { NOT = { has_variable = ddm_proper_liege_resolved } }
            exists = scope:ddm_dejure_level_2
            scope:ddm_dejure_level_2 = { exists = de_jure_liege }
        }
        scope:ddm_dejure_level_2.de_jure_liege = { save_scope_as = ddm_dejure_level_3 }
        if = {
            limit = { scope:ddm_dejure_level_3 = { exists = holder holder = scope:ddm_actor } }
            scope:ddm_actor = { set_variable = { name = ddm_proper_liege_resolved value = 1 } }
        }
        else_if = {
            limit = { scope:ddm_dejure_level_3 = { exists = holder } }
            scope:ddm_dejure_level_3.holder = { save_scope_as = ddm_proper_liege_candidate }
            if = {
                limit = { scope:ddm_proper_liege_candidate = { ddm_is_valid_proper_liege_target_trigger = yes } }
                scope:ddm_proper_liege_candidate = { save_scope_as = ddm_proper_liege }
                scope:ddm_actor = { set_variable = { name = ddm_proper_liege_resolved value = 1 } }
            }
            clear_saved_scope = ddm_proper_liege_candidate
        }
    }

    if = {
        limit = { exists = scope:ddm_proper_liege }
        create_title_and_vassal_change = {
            type = granted
            save_scope_as = ddm_proper_liege_change
            add_claim_on_loss = no
        }
        scope:ddm_transfer_vassal = {
            change_liege = {
                liege = scope:ddm_proper_liege
                change = scope:ddm_proper_liege_change
            }
        }
        resolve_title_and_vassal_change = scope:ddm_proper_liege_change
        scope:ddm_actor = { change_variable = { name = ddm_last_proper_liege_transfers add = 1 } }
        clear_saved_scope = ddm_proper_liege_change
    }

    scope:ddm_actor = { remove_variable = ddm_proper_liege_resolved }
    clear_saved_scope = ddm_proper_liege
    clear_saved_scope = ddm_proper_liege_candidate
    clear_saved_scope = ddm_dejure_level_1
    clear_saved_scope = ddm_dejure_level_2
    clear_saved_scope = ddm_dejure_level_3
    clear_saved_scope = ddm_transfer_vassal
}

ddm_prepare_top_rank_retention_effect = {
    save_scope_as = ddm_actor
    set_variable = { name = ddm_top_rank_total value = 0 }
    set_variable = { name = ddm_top_rank_keep_target value = 0 }
    set_variable = { name = ddm_top_rank_retained_count value = 0 }
    set_variable = { name = ddm_last_top_rank_retained value = 0 }

    every_held_title = {
        remove_variable = ddm_top_rank_retained
        remove_variable = ddm_top_rank_military_density
        remove_variable = ddm_top_rank_counties
    }

    scope:ddm_actor.dynasty = { save_scope_as = ddm_retention_dynasty }

    if = {
        limit = { highest_held_title_tier >= tier_empire }
        scope:ddm_retention_dynasty = {
            if = { limit = { NOT = { has_variable = ddm_managed_kingdom_total } } set_variable = { name = ddm_managed_kingdom_total value = 0 } }
        }
        every_held_title = {
            limit = {
                tier = tier_kingdom
                ddm_title_not_faith_protected_trigger = yes
            }
            if = {
                limit = { NOT = { has_variable = ddm_top_rank_managed_title } }
                set_variable = ddm_top_rank_managed_title
                scope:ddm_retention_dynasty = { change_variable = { name = ddm_managed_kingdom_total add = 1 } }
            }
            save_scope_as = ddm_retention_title
            set_variable = { name = ddm_top_rank_military_density value = 0 }
            set_variable = { name = ddm_top_rank_counties value = 0 }
            every_de_jure_county = {
                if = {
                    limit = {
                        exists = holder
                        holder = { ddm_is_in_actor_realm_trigger = yes }
                    }
                    holder = {
                        save_scope_as = ddm_retention_holder
                        set_variable = { name = ddm_retention_strength_sample value = current_military_strength }
                        if = { limit = { domain_size > 0 } change_variable = { name = ddm_retention_strength_sample divide = domain_size } }
                    }
                    scope:ddm_retention_title = {
                        change_variable = { name = ddm_top_rank_military_density add = scope:ddm_retention_holder.var:ddm_retention_strength_sample }
                        change_variable = { name = ddm_top_rank_counties add = 1 }
                    }
                    scope:ddm_retention_holder = { remove_variable = ddm_retention_strength_sample }
                }
            }
            if = {
                limit = { var:ddm_top_rank_counties > 0 }
                change_variable = { name = ddm_top_rank_military_density divide = var:ddm_top_rank_counties }
            }
            add_to_temporary_list = ddm_top_rank_titles
        }
        set_variable = { name = ddm_top_rank_total value = scope:ddm_retention_dynasty.var:ddm_managed_kingdom_total }
    }
    else_if = {
        limit = { highest_held_title_tier = tier_kingdom }
        scope:ddm_retention_dynasty = {
            if = { limit = { NOT = { has_variable = ddm_managed_duchy_total } } set_variable = { name = ddm_managed_duchy_total value = 0 } }
        }
        every_held_title = {
            limit = {
                tier = tier_duchy
                ddm_title_not_faith_protected_trigger = yes
            }
            if = {
                limit = { NOT = { has_variable = ddm_top_rank_managed_title } }
                set_variable = ddm_top_rank_managed_title
                scope:ddm_retention_dynasty = { change_variable = { name = ddm_managed_duchy_total add = 1 } }
            }
            save_scope_as = ddm_retention_title
            set_variable = { name = ddm_top_rank_military_density value = 0 }
            set_variable = { name = ddm_top_rank_counties value = 0 }
            every_de_jure_county = {
                if = {
                    limit = {
                        exists = holder
                        holder = { ddm_is_in_actor_realm_trigger = yes }
                    }
                    holder = {
                        save_scope_as = ddm_retention_holder
                        set_variable = { name = ddm_retention_strength_sample value = current_military_strength }
                        if = { limit = { domain_size > 0 } change_variable = { name = ddm_retention_strength_sample divide = domain_size } }
                    }
                    scope:ddm_retention_title = {
                        change_variable = { name = ddm_top_rank_military_density add = scope:ddm_retention_holder.var:ddm_retention_strength_sample }
                        change_variable = { name = ddm_top_rank_counties add = 1 }
                    }
                    scope:ddm_retention_holder = { remove_variable = ddm_retention_strength_sample }
                }
            }
            if = {
                limit = { var:ddm_top_rank_counties > 0 }
                change_variable = { name = ddm_top_rank_military_density divide = var:ddm_top_rank_counties }
            }
            add_to_temporary_list = ddm_top_rank_titles
        }
        set_variable = { name = ddm_top_rank_total value = scope:ddm_retention_dynasty.var:ddm_managed_duchy_total }
    }

    if = {
        limit = { var:ddm_top_rank_total > 0 }
        set_variable = { name = ddm_top_rank_keep_target value = var:ddm_top_rank_total }
        change_variable = { name = ddm_top_rank_keep_target multiply = 0.40 }
        if = { limit = { var:ddm_top_rank_keep_target < 1 } set_variable = { name = ddm_top_rank_keep_target value = 1 } }

        # Keep the capital duchy as part of (not in addition to) a king's 40% retained tier. This
        # preserves the personal core without silently making a king retain 40% + one extra duchy.
        if = {
            limit = {
                highest_held_title_tier = tier_kingdom
                exists = capital_county.de_jure_liege
                capital_county.de_jure_liege = {
                    holder = scope:ddm_actor
                    tier = tier_duchy
                    ddm_title_not_faith_protected_trigger = yes
                }
            }
            capital_county.de_jure_liege = {
                set_variable = ddm_top_rank_retained
            }
            change_variable = { name = ddm_top_rank_retained_count add = 1 }
            change_variable = { name = ddm_last_top_rank_retained add = 1 }
        }

        while = {
            limit = { var:ddm_top_rank_retained_count < var:ddm_top_rank_keep_target }
            remove_variable = ddm_top_rank_found
            ordered_in_list = {
                list = ddm_top_rank_titles
                limit = {
                    holder = scope:ddm_actor
                    NOT = { has_variable = ddm_top_rank_retained }
                    NOT = { has_variable = ddm_reserved_recipient }
                }
                order_by = ddm_top_rank_retention_value
                position = 0
                check_range_bounds = no
                set_variable = ddm_top_rank_retained
                scope:ddm_actor = {
                    change_variable = { name = ddm_top_rank_retained_count add = 1 }
                    change_variable = { name = ddm_last_top_rank_retained add = 1 }
                    set_variable = { name = ddm_top_rank_found value = 1 }
                }
            }
            break_if = { NOT = { has_variable = ddm_top_rank_found } }
        }
    }

    every_in_list = {
        list = ddm_top_rank_titles
        remove_variable = ddm_top_rank_military_density
        remove_variable = ddm_top_rank_counties
    }
    remove_variable = ddm_top_rank_found
    clear_saved_scope = ddm_retention_dynasty
}

ddm_organize_dynastic_hierarchy_effect = {
    ddm_initialize_settings_effect = yes
    save_scope_as = ddm_actor

    ddm_v18_build_recipient_lists_effect = yes
    ddm_rank_dynasty_effect = yes

    # Kings decide which 40% of duchies to retain before the duchy pass. Empires wait until duchies are
    # organized, then evaluate kingdom military density from the finished lower-level structure.
    if = { limit = { highest_held_title_tier = tier_kingdom } ddm_prepare_top_rank_retention_effect = yes }

    set_variable = { name = ddm_last_duchies_granted value = 0 }
    set_variable = { name = ddm_last_kingdoms_granted value = 0 }
    set_variable = { name = ddm_last_top_rank_retained value = 0 }

    if = {
        limit = { highest_held_title_tier >= tier_kingdom }
        every_held_title = {
            limit = {
                tier = tier_duchy
                this != scope:ddm_actor.primary_title
                ddm_title_not_faith_protected_trigger = yes
                trigger_if = {
                    limit = { exists = scope:ddm_actor.capital_county.de_jure_liege }
                    this != scope:ddm_actor.capital_county.de_jure_liege
                }
                trigger_if = {
                    limit = { scope:ddm_actor.highest_held_title_tier = tier_kingdom }
                    NOT = { has_variable = ddm_top_rank_retained }
                }
            }
            add_to_temporary_list = ddm_hier_duchy_list
        }

        every_in_list = {
            list = ddm_hier_duchy_list
            limit = { holder ?= scope:ddm_actor }
            save_scope_as = ddm_hier_title
            save_scope_as = ddm_title
            ddm_hier_reset_counts_effect = yes

            # Count de jure counties, and how many each dynast holds.
            every_de_jure_county = {
                scope:ddm_actor = { change_variable = { name = ddm_hier_total add = 1 } }
                if = {
                    limit = {
                        exists = holder
                        holder = { is_alive = yes has_character_flag = ddm_rightful_pool }
                    }
                    holder = { change_variable = { name = ddm_hier_held add = 1 } }
                }
            }
            scope:ddm_actor = {
                set_variable = { name = ddm_hier_half value = var:ddm_hier_total }
                change_variable = { name = ddm_hier_half multiply = 0.5 }
            }

            if = {
                limit = { exists = var:ddm_reserved_recipient }
                if = {
                    limit = {
                        var:ddm_reserved_recipient = {
                            ddm_is_eligible_manual_reservation_recipient_trigger = yes
                            is_landed = yes
                            ddm_is_in_actor_realm_trigger = yes
                        }
                    }
                    var:ddm_reserved_recipient = { save_scope_as = ddm_hier_recipient }
                    scope:ddm_actor = { set_variable = { name = ddm_hier_found value = 1 } }
                }
                else_if = {
                    # Still a legal candidate: hold the reservation for a later settlement after that
                    # person receives qualifying land.
                    limit = { var:ddm_reserved_recipient = { ddm_is_eligible_manual_reservation_recipient_trigger = yes } }
                }
                else = {
                    # Truly stale reservation: clear it and return to scored allocation.
                    remove_variable = ddm_reserved_recipient
                ordered_in_list = {
                    list = ddm_v18_recipient_pool
                    limit = {
                        ddm_is_ranked_title_candidate_trigger = yes
                        has_variable = ddm_hier_held
                        var:ddm_hier_held > 0
                    }
                    order_by = ddm_hier_duchy_rightful_value
                    position = 0
                    check_range_bounds = no
                    save_scope_as = ddm_hier_recipient
                    scope:ddm_actor = { set_variable = { name = ddm_hier_found value = 1 } }
                }
                }
            }
            else = {
                # Local territorial control is the dominant ordinary input; the rightfulness judgment
                # resolves which plausible ruler becomes duke.
                ordered_in_list = {
                    list = ddm_v18_recipient_pool
                    limit = {
                        ddm_is_ranked_title_candidate_trigger = yes
                        has_variable = ddm_hier_held
                        var:ddm_hier_held > 0
                    }
                    order_by = ddm_hier_duchy_rightful_value
                    position = 0
                    check_range_bounds = no
                    save_scope_as = ddm_hier_recipient
                    scope:ddm_actor = { set_variable = { name = ddm_hier_found value = 1 } }
                }
            }

            if = {
                limit = {
                    scope:ddm_actor = { has_variable = ddm_hier_found }
                    exists = scope:ddm_hier_recipient
                }
                scope:ddm_actor = {
                    clear_saved_scope = ddm_hier_change
                    create_title_and_vassal_change = {
                        type = granted
                        save_scope_as = ddm_hier_change
                        add_claim_on_loss = yes
                    }
                }
                change_title_holder_include_vassals = {
                    holder = scope:ddm_hier_recipient
                    change = scope:ddm_hier_change
                    government_base = scope:ddm_actor
                }
                scope:ddm_actor = {
                    resolve_title_and_vassal_change = scope:ddm_hier_change
                    change_variable = { name = ddm_last_duchies_granted add = 1 }
                }
                scope:ddm_hier_title = { remove_variable = ddm_reserved_recipient }
                scope:ddm_hier_recipient = { save_scope_as = ddm_recipient }
                scope:ddm_hier_title = { save_scope_as = ddm_title }
                scope:ddm_actor = {
                    ddm_apply_grant_social_effect = yes
                    ddm_make_dynastic_alliance_effect = yes
                }
            }
            clear_saved_scope = ddm_hier_recipient
        }
    }

    # ---- Kingdoms ----
    if = { limit = { highest_held_title_tier >= tier_empire } ddm_prepare_top_rank_retention_effect = yes }

    # "those who rank maybe second or third can get a title above that." The majority test is over de
    # jure duchy titles now held by the candidate or by a vassal of the candidate, so a king is built on
    # top of his own dukes.
    if = {
        limit = { highest_held_title_tier >= tier_empire }
        every_held_title = {
            limit = {
                tier = tier_kingdom
                this != scope:ddm_actor.primary_title
                ddm_title_not_faith_protected_trigger = yes
                NOT = { has_variable = ddm_top_rank_retained }
            }
            add_to_temporary_list = ddm_hier_kingdom_list
        }

        every_in_list = {
            list = ddm_hier_kingdom_list
            limit = { holder ?= scope:ddm_actor }
            save_scope_as = ddm_hier_title
            save_scope_as = ddm_title
            ddm_hier_reset_counts_effect = yes

            every_de_jure_county = {
                scope:ddm_actor = { change_variable = { name = ddm_hier_total add = 1 } }
                if = {
                    limit = {
                        exists = holder
                        holder = {
                            is_alive = yes
                            has_character_flag = ddm_rightful_pool
                        }
                    }
                    holder = { change_variable = { name = ddm_hier_held add = 1 } }
                }
                # A county under one of the candidate's own vassals counts toward that candidate, which
                # is what makes the pyramid work.
                if = {
                    limit = {
                        exists = holder
                        exists = holder.liege
                        holder.liege = {
                            is_alive = yes
                            has_character_flag = ddm_rightful_pool
                            this != scope:ddm_actor
                        }
                    }
                    holder.liege = { change_variable = { name = ddm_hier_held add = 1 } }
                }
            }
            scope:ddm_actor = {
                set_variable = { name = ddm_hier_half value = var:ddm_hier_total }
                change_variable = { name = ddm_hier_half multiply = 0.5 }
            }

            if = {
                limit = { exists = var:ddm_reserved_recipient }
                if = {
                    limit = {
                        var:ddm_reserved_recipient = {
                            ddm_is_eligible_manual_reservation_recipient_trigger = yes
                            is_landed = yes
                            ddm_is_in_actor_realm_trigger = yes
                            highest_held_title_tier >= tier_duchy
                        }
                    }
                    var:ddm_reserved_recipient = { save_scope_as = ddm_hier_recipient }
                    scope:ddm_actor = { set_variable = { name = ddm_hier_found value = 1 } }
                }
                else_if = {
                    # A living legal candidate who has not yet reached ducal rank keeps this kingdom
                    # reserved for a future hierarchy pass.
                    limit = { var:ddm_reserved_recipient = { ddm_is_eligible_manual_reservation_recipient_trigger = yes } }
                }
                else = {
                    remove_variable = ddm_reserved_recipient
                ordered_in_list = {
                    list = ddm_v18_recipient_pool
                    limit = {
                        ddm_is_ranked_title_candidate_trigger = yes
                        has_variable = ddm_hier_held
                        var:ddm_hier_held > 0
                        highest_held_title_tier >= tier_duchy
                    }
                    order_by = ddm_hier_kingdom_rightful_value
                    position = 0
                    check_range_bounds = no
                    save_scope_as = ddm_hier_recipient
                    scope:ddm_actor = { set_variable = { name = ddm_hier_found value = 1 } }
                }
                }
            }
            else = {
                ordered_in_list = {
                    list = ddm_v18_recipient_pool
                    limit = {
                        ddm_is_ranked_title_candidate_trigger = yes
                        has_variable = ddm_hier_held
                        var:ddm_hier_held > 0
                        highest_held_title_tier >= tier_duchy
                    }
                    order_by = ddm_hier_kingdom_rightful_value
                    position = 0
                    check_range_bounds = no
                    save_scope_as = ddm_hier_recipient
                    scope:ddm_actor = { set_variable = { name = ddm_hier_found value = 1 } }
                }
            }

            if = {
                limit = {
                    scope:ddm_actor = { has_variable = ddm_hier_found }
                    exists = scope:ddm_hier_recipient
                }
                scope:ddm_actor = {
                    clear_saved_scope = ddm_hier_change
                    create_title_and_vassal_change = {
                        type = granted
                        save_scope_as = ddm_hier_change
                        add_claim_on_loss = yes
                    }
                }
                change_title_holder_include_vassals = {
                    holder = scope:ddm_hier_recipient
                    change = scope:ddm_hier_change
                    government_base = scope:ddm_actor
                }
                scope:ddm_actor = {
                    resolve_title_and_vassal_change = scope:ddm_hier_change
                    change_variable = { name = ddm_last_kingdoms_granted add = 1 }
                }
                scope:ddm_hier_title = { remove_variable = ddm_reserved_recipient }
                scope:ddm_hier_recipient = { save_scope_as = ddm_recipient }
                scope:ddm_hier_title = { save_scope_as = ddm_title }
                scope:ddm_actor = {
                    ddm_apply_grant_social_effect = yes
                    ddm_make_dynastic_alliance_effect = yes
                }
            }
            clear_saved_scope = ddm_hier_recipient
        }
    }

    every_in_list = { list = ddm_v18_recipient_pool remove_variable = ddm_hier_held }
    every_held_title = { remove_variable = ddm_top_rank_retained }
    remove_variable = ddm_top_rank_total
    remove_variable = ddm_top_rank_keep_target
    remove_variable = ddm_top_rank_retained_count
    remove_variable = ddm_hier_total
    remove_variable = ddm_hier_half
    remove_variable = ddm_hier_found
    clear_saved_scope = ddm_hier_title
    clear_saved_scope = ddm_title
    clear_saved_scope = ddm_hier_recipient
    clear_saved_scope = ddm_hier_change
    ddm_clear_dynasty_ranking_effect = yes
    dynasty = {
        every_dynasty_member = {
            remove_character_flag = ddm_partition_settlement_pool
            remove_variable = ddm_batch_land_grants
        }
    }
    every_courtier = { remove_variable = ddm_batch_land_grants remove_variable = ddm_hier_held }
    every_vassal_or_below = { remove_variable = ddm_batch_land_grants remove_variable = ddm_hier_held }
    ddm_schedule_post_settlement_alliance_event_effect = yes
}

ddm_build_purge_register_effect = {
    # Build one exact, stable list. The preview and the execution use the same character flags, so
    # family-regard recalculation cannot make the displayed count silently vanish when the player
    # confirms the order.
    ddm_initialize_settings_effect = yes
    save_scope_as = ddm_actor

    # Family regard is intentionally refreshed here, once, before the register is built. The execution
    # does not recalculate it again mid-purge.
    ddm_refresh_all_dynasty_regard_effect = yes

    set_variable = { name = ddm_purge_living_total value = 0 }
    set_variable = { name = ddm_purge_candidate_count value = 0 }
    set_variable = { name = ddm_purge_standard_survivor_count value = 0 }
    set_variable = { name = ddm_purge_protected_count value = 0 }
    set_variable = { name = ddm_purge_already_satisfied_count value = 0 }

    dynasty = {
        every_dynasty_member = {
            limit = { is_alive = yes }
            remove_character_flag = ddm_selected_for_review
            remove_character_flag = ddm_processed_this_review
            scope:ddm_actor = { change_variable = { name = ddm_purge_living_total add = 1 } }

            # The standard is a score band, not a named-trait gate. Count everyone who clears it so the
            # audit can be reconciled with the candidate list.
            if = {
                limit = { NOT = { ddm_fails_current_dynastic_standard_trigger = yes } }
                scope:ddm_actor = { change_variable = { name = ddm_purge_standard_survivor_count add = 1 } }
            }

            # Track exclusions separately from people who actually meet the standard, so the UI can
            # explain a surprising total.
            if = {
                limit = {
                    OR = {
                        this = scope:ddm_actor
                        has_character_flag = ddm_protected_from_automation
                        AND = {
                            scope:ddm_actor.var:ddm_protect_player_heir = 1
                            exists = scope:ddm_actor.player_heir
                            this = scope:ddm_actor.player_heir
                        }
                    }
                }
                scope:ddm_actor = { change_variable = { name = ddm_purge_protected_count add = 1 } }
            }
            else_if = {
                limit = {
                    ddm_fails_current_dynastic_standard_trigger = yes
                    NOT = { ddm_purge_needs_action_trigger = yes }
                }
                scope:ddm_actor = { change_variable = { name = ddm_purge_already_satisfied_count add = 1 } }
            }

            if = {
                limit = { ddm_purge_needs_action_trigger = yes }
                add_character_flag = ddm_selected_for_review
                scope:ddm_actor = { change_variable = { name = ddm_purge_candidate_count add = 1 } }
            }
        }
    }
    set_variable = { name = ddm_purge_register_ready value = 1 }
}

ddm_build_outsider_page_effect = {
    if = { limit = { NOT = { has_variable = ddm_outsider_page } } set_variable = { name = ddm_outsider_page value = 0 } }
    clear_variable_list = ddm_outsider_page_list

    set_variable = { name = ddm_outsider_page_size value = 20 }
    set_variable = { name = ddm_outsider_page_total value = 0 }
    set_variable = { name = ddm_outsider_last_page value = 0 }
    set_variable = { name = ddm_outsider_page_fill value = 0 }

    if = {
        limit = { has_variable_list = ddm_outsider_counties }
        every_in_list = {
            variable = ddm_outsider_counties
            prev = {
                if = {
                    limit = { var:ddm_outsider_page_fill >= var:ddm_outsider_page_size }
                    change_variable = { name = ddm_outsider_last_page add = 1 }
                    set_variable = { name = ddm_outsider_page_fill value = 0 }
                }
                change_variable = { name = ddm_outsider_page_fill add = 1 }
                change_variable = { name = ddm_outsider_page_total add = 1 }
            }
        }
    }

    if = { limit = { var:ddm_outsider_page < 0 } set_variable = { name = ddm_outsider_page value = 0 } }
    if = { limit = { var:ddm_outsider_page > var:ddm_outsider_last_page } set_variable = { name = ddm_outsider_page value = var:ddm_outsider_last_page } }

    set_variable = { name = ddm_outsider_first value = var:ddm_outsider_page }
    change_variable = { name = ddm_outsider_first multiply = var:ddm_outsider_page_size }
    set_variable = { name = ddm_outsider_stop value = var:ddm_outsider_first }
    change_variable = { name = ddm_outsider_stop add = var:ddm_outsider_page_size }
    set_variable = { name = ddm_outsider_index value = 0 }

    if = {
        limit = { has_variable_list = ddm_outsider_counties }
        every_in_list = {
            variable = ddm_outsider_counties
            save_temporary_scope_as = ddm_outsider_page_row
            prev = {
                if = {
                    limit = { var:ddm_outsider_index >= var:ddm_outsider_first var:ddm_outsider_index < var:ddm_outsider_stop }
                    add_to_variable_list = { name = ddm_outsider_page_list target = scope:ddm_outsider_page_row }
                }
                change_variable = { name = ddm_outsider_index add = 1 }
            }
        }
    }

    set_variable = { name = ddm_outsider_shown_from value = var:ddm_outsider_first }
    change_variable = { name = ddm_outsider_shown_from add = 1 }
    set_variable = { name = ddm_outsider_shown_to value = var:ddm_outsider_stop }
    if = { limit = { var:ddm_outsider_shown_to > var:ddm_outsider_page_total } set_variable = { name = ddm_outsider_shown_to value = var:ddm_outsider_page_total } }
    if = { limit = { var:ddm_outsider_page_total = 0 } set_variable = { name = ddm_outsider_shown_from value = 0 } }
    set_variable = { name = ddm_outsider_page_display value = var:ddm_outsider_page }
    change_variable = { name = ddm_outsider_page_display add = 1 }
    set_variable = { name = ddm_outsider_pages_display value = var:ddm_outsider_last_page }
    change_variable = { name = ddm_outsider_pages_display add = 1 }
}

            }
        }
    }

        every_in_list = {
            variable = ddm_outsider_recipient_candidates
            save_temporary_scope_as = ddm_recipient_page_row
            prev = {
                if = {
                    limit = {
                        var:ddm_recipient_index >= var:ddm_recipient_first
                        var:ddm_recipient_index < var:ddm_recipient_stop
                    }
                    add_to_variable_list = { name = ddm_recipient_page_list target = scope:ddm_recipient_page_row }
                }
                change_variable = { name = ddm_recipient_index add = 1 }
            }
        }
    }

    set_variable = { name = ddm_recipient_shown_from value = var:ddm_recipient_first }
    change_variable = { name = ddm_recipient_shown_from add = 1 }
    set_variable = { name = ddm_recipient_shown_to value = var:ddm_recipient_stop }
    if = { limit = { var:ddm_recipient_shown_to > var:ddm_recipient_page_total } set_variable = { name = ddm_recipient_shown_to value = var:ddm_recipient_page_total } }
    if = { limit = { var:ddm_recipient_page_total = 0 } set_variable = { name = ddm_recipient_shown_from value = 0 } }
    set_variable = { name = ddm_recipient_page_display value = var:ddm_recipient_page }
    change_variable = { name = ddm_recipient_page_display add = 1 }
    set_variable = { name = ddm_recipient_pages_display value = var:ddm_recipient_last_page }
    change_variable = { name = ddm_recipient_pages_display add = 1 }
