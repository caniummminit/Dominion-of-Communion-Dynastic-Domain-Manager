    type = character_event

        first_valid = {
            triggered_desc = { trigger = { var:ddm_succession_policy = 1 } desc = ddm_manager.1000.policy_balanced }
            triggered_desc = { trigger = { var:ddm_succession_policy = 2 } desc = ddm_manager.1000.policy_follow }
            triggered_desc = { trigger = { var:ddm_succession_policy = 3 } desc = ddm_manager.1000.policy_estate }
        }
    }
    theme = crown
    left_portrait = { character = root animation = personality_rational }
    immediate = { ddm_prepare_dashboard_effect = yes }

    option = {
        name = ddm_manager.1000.v19_edit
        flavor = ddm_manager.1000.v19_edit.flavor
        set_variable = { name = ddm_pending_domain_action value = 0 }
        ddm_prepare_domain_planner_effect = yes
    }
    option = {
        name = ddm_manager.1000.v19_distribute
        flavor = ddm_manager.1000.v19_distribute.flavor
        # Always available: even at the protected-domain target, Distribute also settles duchies/kingdoms
        # and can refresh the hierarchy after conquest.
        if = {
            limit = {
                OR = {
                    ddm_domain_plan_limit_changed_trigger = yes
                    ddm_domain_plan_target_mismatch_trigger = yes
                }
            }
            set_variable = { name = ddm_pending_domain_action value = 1 }
            ddm_prepare_domain_planner_effect = yes
        }
        else = {
            ddm_run_full_settlement_effect = yes
            trigger_event = ddm_manager.0101
        }
    }
    option = {
        name = ddm_manager.1000.v19_preview
        flavor = ddm_manager.1000.v19_preview.flavor
        if = {
            limit = {
                ddm_above_protected_domain_target_trigger = yes
                OR = {
                    ddm_domain_plan_limit_changed_trigger = yes
                    ddm_domain_plan_target_mismatch_trigger = yes
                }
            }
            set_variable = { name = ddm_pending_domain_action value = 0 }
            ddm_prepare_domain_planner_effect = yes
        }
        else = { trigger_event = ddm_manager.1010 }
    }
    option = {
        name = ddm_manager.1000.v19_policy
        flavor = ddm_manager.1000.v19_policy.flavor
        trigger_event = ddm_manager.1050
    }
    option = { name = ddm_manager.1000.v19_more trigger_event = ddm_manager.1030 }
    option = { name = ddm_manager.main set_variable = { name = ddm_ui_page value = flag:overview } set_variable = { name = ddm_ui_open value = yes } }

ddm_manager.0112 = {
    type = character_event
    hidden = yes
    immediate = {
        if = {
            limit = { has_variable = ddm_settlement_queued is_at_war = yes }
            trigger_event = { id = ddm_manager.0112 days = 7 }
        }
        else_if = {
            limit = { has_variable = ddm_settlement_queued is_at_war = no }
            remove_variable = ddm_settlement_queued
            ddm_initialize_settings_effect = yes
            ddm_prepare_dashboard_effect = yes
            if = {
                limit = { OR = { ddm_domain_plan_limit_changed_trigger = yes ddm_domain_plan_target_mismatch_trigger = yes } }
                set_variable = { name = ddm_pending_domain_action value = 1 }
                ddm_prepare_domain_planner_effect = yes
                set_variable = { name = ddm_ui_page value = flag:estate }
                set_variable = { name = ddm_ui_open value = yes }
            }
            else = {
                set_variable = { name = ddm_ui_processing_settlement value = 1 }
                ddm_run_full_settlement_effect = yes
                set_variable = { name = ddm_ui_records_mode value = 2 }
                set_variable = { name = ddm_ui_page value = flag:records }
                trigger_event = { id = ddm_manager.0111 days = 1 }
            }
        }
        else = { remove_variable = ddm_settlement_queued }
    }
}

        else = {
            ddm_finish_domain_choice_review_effect = yes
            trigger_event = ddm_manager.1022
        }
    }
    option = {
        name = ddm_manager.1020.b
        flavor = ddm_manager.1020.b.flavor
        ddm_reset_all_domain_choices_effect = yes
        ddm_lock_recommended_domain_core_effect = yes
        # Accepting the recommendation is an explicit player command. Classify every remaining holding as
        # "office decides" so distribution will not force another manual review immediately.
        every_held_title = {
            limit = {
                OR = {
                    tier = tier_county
                    AND = { tier = tier_barony title_province = { is_county_capital = no } }
                }
                NOT = {
                    OR = {
                        has_variable = ddm_manual_core
                        has_variable = ddm_manual_shed
                        has_variable = ddm_manual_neutral
                    }
                }
            }
            set_variable = ddm_manual_neutral
        }
        set_variable = { name = ddm_domain_plan_limit value = domain_limit }
        trigger_event = ddm_manager.1020
    }
    option = {
        name = ddm_manager.1020.c
